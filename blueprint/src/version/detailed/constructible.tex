\chapter{Constructible sheaves}\label{chap:constructible}

This chapter sets up the constructible sheaves on the small \'etale site of
a scheme: the definitions, their stability under pullback, pushforward along
a closed immersion and extension by zero along an open immersion, the
open--closed d\'evissage, and the closure under kernels, cokernels and
extensions.  Everything is classical; the references are
\cite[Exp.~IX, \S2]{SGA4} and the Stacks Project (Tags \texttt{03RW}--\texttt{03RZ},
\texttt{05BE}, \texttt{095K}, \texttt{09YJ}).  The statements are arranged
in the order in which they are formalised, and the proofs are written at
the level of detail of the Stacks Project.

\section{Setting}\label{sec:constructible-setting}

We keep the conventions of \Cref{sec:openclosed-setting}: schemes live in a
universe $u$; the coefficient category $\mathcal A$ is Grothendieck abelian
with morphisms in $\mathrm{Type}\,(u+1)$; $\et X$ is the small \'etale site
of $X$ and $\shv{\et X}$ the category of $\mathcal A$-valued sheaves on it;
$g^*\dashv g_*$ is the pullback--pushforward adjunction along a morphism $g$
and $j_!$ the extension by zero along an open immersion.  Whenever
finiteness of stalks is mentioned, $\mathcal A$ is moreover concrete over
$\mathrm{Type}\,(u+1)$, and finiteness refers to the underlying set.  The
case of interest is $\mathcal A=\mathrm{Mod}_\Lambda$ for a finite ring
$\Lambda$ (in the $\ell$-adic formalism $\Lambda=\Z/\ell^n\Z$); some
statements are only claimed in this case and say so.

A geometric point of $X$ is a morphism $\bar x\colon\spec\Omega\to X$ with
$\Omega$ separably closed; the stalk $\mathcal F_{\bar x}$ is the fibre of
$\mathcal F$ at the corresponding point of the site $\et X$.

\begin{definition}[Constant sheaves]\label{def:constructible-constant}
\lean{CategoryTheory.Sheaf.IsConstant}
\leanok
A sheaf $\mathcal F\in\shv{\et X}$ is \emph{constant} if it lies in the
essential image of the constant-sheaf functor
$\mathcal A\to\shv{\et X}$, $M\mapsto M_X$, the left adjoint of global
sections (sheafification of the constant presheaf).
\end{definition}

\begin{definition}[Locally constant sheaves]
\label{def:constructible-locally-constant}
\uses{def:constructible-constant}
\lean{AlgebraicGeometry.Scheme.LocallyConstantDatum,
      AlgebraicGeometry.Scheme.IsLocallyConstantSheaf}
\leanok
A \emph{trivialising datum} for $\mathcal F\in\shv{\et X}$ is an \'etale
cover $(U_i\to X)_{i\in I}$ --- a jointly surjective family of \'etale
morphisms --- such that for every $i$ the restriction of $\mathcal F$ to
$\et{U_i}$, that is, the sheaf induced by $\mathcal F$ on the site
$\et X$ localised at $U_i$, is constant.  The sheaf $\mathcal F$ is
\emph{locally constant} if a trivialising datum exists.
\end{definition}

\begin{definition}[Finite sheaves]\label{def:constructible-finite}
\lean{AlgebraicGeometry.Scheme.IsFiniteSheaf}
\leanok
A sheaf $\mathcal F\in\shv{\et X}$ (with $\mathcal A$ concrete) is
\emph{finite} if for every geometric point $\bar x$ of $X$ the underlying
set of the stalk $\mathcal F_{\bar x}$ is finite.
\end{definition}

\begin{definition}[Finite locally constant sheaves]
\label{def:constructible-finite-locally-constant}
\uses{def:constructible-locally-constant,def:constructible-finite}
\lean{AlgebraicGeometry.Scheme.IsFiniteLocallyConstantSheaf}
\leanok
A sheaf is \emph{finite locally constant} if it is locally constant and
finite.
\end{definition}

\begin{definition}[Stratifications]\label{def:constructible-stratification}
\lean{Topology.Stratification, AlgebraicGeometry.Scheme.Stratification}
\leanok
A \emph{stratification} of a topological space $T$ is a finite family
$(T_k)_{k<n}$ of pairwise disjoint locally closed subsets covering $T$
(Stacks Project, Tag \texttt{09XZ}, with finiteness added).  A
\emph{stratification} of a scheme $X$ is a stratification $(X_k)_{k<n}$ of
its underlying space together with, for each $k$, a scheme $Z_k$ and an
immersion $\iota_k\colon Z_k\to X$ with set-theoretic image $X_k$.
\end{definition}

\begin{definition}[Constructible sheaves]\label{def:constructible-constructible}
\uses{def:constructible-finite-locally-constant,def:constructible-stratification}
\lean{AlgebraicGeometry.Scheme.IsConstructibleSheaf}
\leanok
A sheaf $\mathcal F\in\shv{\et X}$ is \emph{constructible} if there is a
stratification $(\iota_k\colon Z_k\to X)_{k<n}$ of $X$ such that each
pullback $\iota_k^*\mathcal F$ is finite locally constant.
\end{definition}

The scheme structures $Z_k$ on the strata are part of the datum; by
\Cref{lemma:constructible-immersion-independence} below the condition does
not depend on them.

\begin{lemma}[Constant sheaves are locally constant]
\label{lemma:constructible-constant-locally-constant}
\uses{def:constructible-constant,def:constructible-locally-constant}
\lean{CategoryTheory.Sheaf.IsConstant.isLocallyConstantSheaf}
\leanok
A constant sheaf is locally constant.
\end{lemma}
\begin{proof}
\leanok
The identity cover $(X\to X)$ is a trivialising datum: the restriction of
$M_X$ to $\et X$ localised at $X$ is $M_X$ again, which is constant.
\end{proof}

\begin{lemma}[Pullback of a constant sheaf]
\label{lemma:constructible-pullback-constant}
\uses{def:constructible-constant}
\lean{AlgebraicGeometry.Scheme.upperStarConstantSheafIso}
\leanok
For $g\colon Y\to X$ and $M\in\mathcal A$ there is a natural isomorphism
$g^*(M_X)\cong M_Y$.  In particular $g^*$ sends constant sheaves to constant
sheaves.
\end{lemma}
\begin{proof}
\leanok
Both $M\mapsto g^*(M_X)$ and $M\mapsto M_Y$ are left adjoint to
$\mathcal G\mapsto\Gamma(Y,\mathcal G)=\Gamma(X,g_*\mathcal G)$, because
base change along $g$ preserves the terminal object of the site
($Y\times_XX\cong Y$); left adjoints are unique up to unique isomorphism.
\end{proof}

\section{Pullback stability}\label{sec:constructible-pullback}

\begin{lemma}[Invariance under isomorphism]
\label{lemma:constructible-iso-invariance}
\uses{def:constructible-locally-constant,def:constructible-finite,def:constructible-finite-locally-constant,def:constructible-constructible}
\lean{AlgebraicGeometry.Scheme.IsLocallyConstantSheaf.of_iso,
  AlgebraicGeometry.Scheme.IsFiniteSheaf.of_iso,
  AlgebraicGeometry.Scheme.IsFiniteLocallyConstantSheaf.of_iso,
  AlgebraicGeometry.Scheme.IsConstructibleSheaf.of_iso}
\leanok
If $\mathcal F\cong\mathcal F'$ in $\shv{\et X}$, then $\mathcal F$ is
locally constant (finite, finite locally constant, constructible) if and
only if $\mathcal F'$ is.
\end{lemma}
\begin{proof}
\leanok
An isomorphism of sheaves induces isomorphisms of the restrictions to every
localised site, of all stalks, and of all pullbacks, and each of the four
notions is a condition on these.  A trivialising datum, respectively a
stratification, for $\mathcal F$ is therefore one for $\mathcal F'$.
\end{proof}

\begin{lemma}[Pullback of locally constant sheaves]
\label{lemma:constructible-pullback-locally-constant}
\uses{def:constructible-locally-constant,lemma:constructible-pullback-constant}
\lean{AlgebraicGeometry.Scheme.IsLocallyConstantSheaf.upperStar,
  AlgebraicGeometry.Scheme.isConstant_over_iff_upperStar,
  AlgebraicGeometry.Scheme.Etale.overEquiv}
\leanok
Let $g\colon Y\to X$ be a morphism and let $\mathcal F\in\shv{\et X}$ be
locally constant.  Then $g^*\mathcal F$ is locally constant.
\end{lemma}
\begin{proof}
\leanok
Let $(U_i\to X)_i$ be a trivialising datum for $\mathcal F$.  The base
change $(U_i\times_XY\to Y)_i$ is an \'etale cover of $Y$, since \'etale
morphisms and jointly surjective families are stable under base change.
The restriction of $g^*\mathcal F$ to $\et{U_i\times_XY}$ is the pullback,
along the projection $g_i\colon U_i\times_XY\to U_i$, of the restriction
of $\mathcal F$ to $\et{U_i}$: pullback along $g$ followed by restriction to
the localised site equals restriction followed by pullback along $g_i$,
because both are the sheaf pullback along the composite of the base change
functors, which agree.  The restriction of $\mathcal F$ to $\et{U_i}$ is
constant, and pullbacks of constant sheaves are constant by
\Cref{lemma:constructible-pullback-constant}.  Hence
$(U_i\times_XY\to Y)_i$ is a trivialising datum for $g^*\mathcal F$.
\end{proof}

\begin{lemma}[Pullback of finite sheaves]
\label{lemma:constructible-pullback-finite}
\uses{def:constructible-finite,lemma:openclosed-stalk-upperStar}
\lean{AlgebraicGeometry.Scheme.IsFiniteSheaf.upperStar}
\leanok
Let $g\colon Y\to X$ be a morphism and let $\mathcal F\in\shv{\et X}$ be
finite.  Then $g^*\mathcal F$ is finite.
\end{lemma}
\begin{proof}
\leanok
For a geometric point $\bar y$ of $Y$, the stalk $(g^*\mathcal F)_{\bar y}$
is the stalk $\mathcal F_{g\circ\bar y}$ by
\Cref{lemma:openclosed-stalk-upperStar}, which is finite by hypothesis.
\end{proof}

\begin{lemma}[Pullback of a stratification]
\label{lemma:constructible-pullback-stratification}
\uses{def:constructible-stratification}
\lean{AlgebraicGeometry.Scheme.Stratification.pullback}
\leanok
Let $g\colon Y\to X$ be a morphism and let $(\iota_k\colon Z_k\to X)_{k<n}$
be a stratification of $X$.  Then the base changes
$(\iota'_k\colon Z_k\times_XY\to Y)_{k<n}$ form a stratification of $Y$,
with strata $g^{-1}(X_k)$.
\end{lemma}
\begin{proof}
\leanok
Immersions are stable under base change, and the image of the base change
$Z_k\times_XY\to Y$ of $\iota_k$ is the preimage $g^{-1}(\iota_k(Z_k))$:
the underlying set of a fibre product of schemes surjects onto the
set-theoretic fibre product, so the image of the projection to $Y$ is
exactly the set of points of $Y$ whose image in $X$ lies in the image of
$\iota_k$.  The sets $g^{-1}(X_k)$ are locally closed, pairwise disjoint and
cover $Y$, as preimages of the locally closed, pairwise disjoint sets $X_k$
covering $X$.
\end{proof}

\begin{proposition}[Pullback of constructible sheaves]
\label{prop:constructible-pullback}
\uses{def:constructible-constructible,lemma:constructible-pullback-locally-constant,lemma:constructible-pullback-finite,lemma:constructible-pullback-stratification,lemma:constructible-iso-invariance}
\lean{AlgebraicGeometry.Scheme.IsConstructibleSheaf.upperStar,
  AlgebraicGeometry.Scheme.IsFiniteLocallyConstantSheaf.upperStar}
\leanok
Let $g\colon Y\to X$ be a morphism and let $\mathcal F\in\shv{\et X}$ be
constructible.  Then $g^*\mathcal F$ is constructible.  In particular
constructibility is preserved by restriction to an open, to a closed
subscheme, and to any locally closed subscheme.
\end{proposition}
\begin{proof}
\leanok
Let $(\iota_k\colon Z_k\to X)_k$ be a stratification of $X$ with all
$\iota_k^*\mathcal F$ finite locally constant, and let
$(\iota'_k\colon Z_k\times_XY\to Y)_k$ be its pullback
(\Cref{lemma:constructible-pullback-stratification}), with projections
$g_k\colon Z_k\times_XY\to Z_k$.  Since $\iota_k\circ g_k=g\circ\iota'_k$
and pullback is functorial up to natural isomorphism,
$\iota_k'^*g^*\mathcal F\cong g_k^*\iota_k^*\mathcal F$, which is finite
locally constant by
\Cref{lemma:constructible-pullback-locally-constant,lemma:constructible-pullback-finite}
and \Cref{lemma:constructible-iso-invariance}.
\end{proof}

\section{Independence of the scheme structure on the strata}
\label{sec:constructible-invariance}

\begin{lemma}[Invariance under universal homeomorphisms]
\label{lemma:constructible-universal-homeomorphism-invariance}
\uses{def:constructible-finite-locally-constant,lemma:weil-pbc-universal-homeomorphism-site-equivalence,prop:weil-pbc-universal-homeomorphism-topos,lemma:constructible-pullback-locally-constant,lemma:constructible-pullback-finite}
\lean{CategoryTheory.Sheaf.isConstant_upperStar_iff,
  AlgebraicGeometry.Scheme.IsLocallyConstantSheaf.of_upperStar,
  AlgebraicGeometry.Scheme.exists_comp_eq_of_surjective,
  AlgebraicGeometry.Scheme.IsFiniteSheaf.of_upperStar,
  AlgebraicGeometry.Scheme.IsFiniteLocallyConstantSheaf.of_upperStar,
  AlgebraicGeometry.Scheme.isFiniteLocallyConstantSheaf_upperStar_iff}
\leanok
Let $u\colon Z'\to Z$ be a universal homeomorphism.  Then
$\mathcal G\in\shv{\et Z}$ is constant (locally constant) if and only if
$u^*\mathcal G$ is.  If moreover $u$ is a surjective closed immersion, then
$\mathcal G$ is finite (finite locally constant) if and only if
$u^*\mathcal G$ is.
\end{lemma}
\begin{proof}
\leanok
The direction ``only if'' is
\Cref{lemma:constructible-pullback-locally-constant,lemma:constructible-pullback-finite}.
For the converse, base change along $u$ is an equivalence of sites
$\et Z\simeq\et{Z'}$ by
\Cref{lemma:weil-pbc-universal-homeomorphism-site-equivalence}
(Tag \texttt{04DZ}), hence $u^*$ is an equivalence of categories of sheaves
(\Cref{prop:weil-pbc-universal-homeomorphism-topos}).  Constancy: $u^*$ is
fully faithful and carries constant sheaves to constant sheaves, so from
$u^*\mathcal G\cong u^*M_{Z}$ we get $\mathcal G\cong M_Z$.  Local
constancy: let $(V'_i\to Z')_i$ be a trivialising \'etale cover of
$u^*\mathcal G$.  Each $V'_i$ is isomorphic over $Z'$ to $V_i\times_ZZ'$
for an \'etale $V_i\to Z$, and the $V_i$ cover $Z$ since $u$ is
surjective.  The projection $u_i\colon V_i\times_ZZ'\to V_i$ is a
universal homeomorphism, and $u_i^*(\mathcal G|_{V_i})\cong
(u^*\mathcal G)|_{V'_i}$ is constant, so $\mathcal G|_{V_i}$ is constant
by the first part applied to $u_i$.  Finiteness, for $u$ a surjective closed
immersion: the residue field maps of a closed immersion are surjective,
hence isomorphisms, so every geometric point $\bar z$ of $Z$ lifts to a
geometric point $\bar z'$ of $Z'$ with $u\circ\bar z'=\bar z$, and the
stalk of $\mathcal G$ at $\bar z$ is the stalk of $u^*\mathcal G$ at
$\bar z'$.
\end{proof}

\begin{lemma}[Independence of the scheme structure]
\label{lemma:constructible-immersion-independence}
\uses{lemma:constructible-universal-homeomorphism-invariance,lemma:constructible-pullback-stratification}
\lean{AlgebraicGeometry.Scheme.IsFiniteLocallyConstantSheaf.upperStar_of_range_subset,
  AlgebraicGeometry.Scheme.isFiniteLocallyConstantSheaf_upperStar_iff_of_range_eq}
\leanok
Let $a\colon Z\to X$ and $b\colon Z'\to X$ be immersions with the same
image.  Then $a^*\mathcal F$ is finite locally constant if and only if
$b^*\mathcal F$ is.
\end{lemma}
\begin{proof}
\leanok
Let $W=Z\times_XZ'$ with projections $p\colon W\to Z$ and $q\colon W\to Z'$.
The projection $p$ is an immersion, as a base change of $b$, and it is
surjective, its image being $a^{-1}(b(Z'))=a^{-1}(a(Z))=Z$.  A surjective
immersion is a surjective closed immersion (an immersion is a closed
immersion into an open subscheme, and surjectivity forces that open to be
everything), hence a universal homeomorphism.  The same holds for $q$.  Now
$p^*a^*\mathcal F\cong q^*b^*\mathcal F$, both being the pullback along
$a\circ p=b\circ q$, so by
\Cref{lemma:constructible-universal-homeomorphism-invariance} applied to
$p$ and to $q$, $a^*\mathcal F$ is finite locally constant if and only if
$p^*a^*\mathcal F$ is, if and only if $q^*b^*\mathcal F$ is, if and only if
$b^*\mathcal F$ is.
\end{proof}

\section{Refinement and generic constancy}\label{sec:constructible-refinement}

\begin{definition}[Refinement]\label{def:constructible-refines}
\uses{def:constructible-stratification}
\lean{AlgebraicGeometry.Scheme.Stratification.Refines}
\leanok
A stratification $(T_l)_{l<m}$ of $T$ \emph{refines} a stratification
$(T_k)_{k<n}$ if every $T_l$ is contained in some $T_k$.
\end{definition}

\begin{lemma}[Common refinement]\label{lemma:constructible-common-refinement}
\uses{def:constructible-refines}
\lean{AlgebraicGeometry.Scheme.Stratification.inter,
  AlgebraicGeometry.Scheme.Stratification.exists_refines}
\leanok
Two stratifications $(\iota_k\colon Z_k\to X)_{k<n}$ and
$(\kappa_l\colon W_l\to X)_{l<m}$ of a scheme $X$ have a common refinement.
\end{lemma}
\begin{proof}
\leanok
Index the pairs $(k,l)$ by $k+nl<nm$.  The intersections $X_k\cap X'_l$ are
locally closed (an intersection of two locally closed sets is locally
closed), pairwise disjoint, and cover $X$.  Take as scheme on the stratum
$X_k\cap X'_l$ the fibre product $Z_k\times_XW_l$ with the immersion
$Z_k\times_XW_l\to X$; it is an immersion, as a composite of the base
change of $\kappa_l$ with $\iota_k$, and its image is
$\iota_k(\iota_k^{-1}(X'_l))=X_k\cap X'_l$.  This stratification refines
both given ones.
\end{proof}

\begin{lemma}[Constructibility passes to refinements]
\label{lemma:constructible-of-refines}
\uses{def:constructible-refines,lemma:constructible-immersion-independence,prop:constructible-pullback}
\lean{AlgebraicGeometry.Scheme.Stratification.Refines.isFiniteLocallyConstantSheaf_upperStar,
  AlgebraicGeometry.Scheme.IsConstructibleSheaf.of_refines}
\leanok
Let $(\iota_k\colon Z_k\to X)_k$ be a stratification with all
$\iota_k^*\mathcal F$ finite locally constant, and let
$(\kappa_l\colon W_l\to X)_l$ refine it.  Then all $\kappa_l^*\mathcal F$
are finite locally constant.
\end{lemma}
\begin{proof}
\leanok
Fix $l$ and choose $k$ with $X'_l\subseteq X_k$.  The base change
$\kappa'_l\colon W_l\times_XZ_k\to Z_k$ of $\kappa_l$ is an immersion, and
the composite $\iota_k\circ\kappa'_l\colon W_l\times_XZ_k\to X$ is an
immersion with image $\iota_k(\iota_k^{-1}(X'_l))=X'_l$, the same image as
$\kappa_l$.  Its pullback of $\mathcal F$ is
$\kappa_l'^*\iota_k^*\mathcal F$, finite locally constant by
\Cref{prop:constructible-pullback} (its lemmas for locally constant and
finite sheaves).  By \Cref{lemma:constructible-immersion-independence},
$\kappa_l^*\mathcal F$ is finite locally constant.
\end{proof}

\begin{proposition}[Generic constancy]
\label{prop:constructible-generic-constancy}
\uses{def:constructible-constructible}
Let $X$ be a Noetherian scheme and let $\mathcal F\in\shv{\et X}$ be
constructible.  Then there is a dense open $U\subseteq X$ such that
$\mathcal F|_U$ is finite locally constant.
\end{proposition}
\begin{proof}
Let $(X_k)_{k<n}$ be a stratification of $X$ witnessing constructibility.
Let $X=X^{(1)}\cup\dots\cup X^{(r)}$ be the decomposition into irreducible
components, with generic points $\eta_1,\dots,\eta_r$.  Each $\eta_i$ lies
in exactly one stratum $X_{k(i)}$; since $X_{k(i)}$ is locally closed and
contains the generic point of $X^{(i)}$, it contains a nonempty open
subset of $X^{(i)}$, namely $V\cap X^{(i)}$ for an open $V\subseteq X$ with
$X_{k(i)}=V\cap\overline{X_{k(i)}}$ and $\eta_i\in V$.  Let $U_i$ be the
open subset of $X$ obtained by removing from $V\cap X^{(i)}$ the other
components, so that $U_i$ is a nonempty open of $X$ contained in
$X_{k(i)}$, and let $U=\bigcup_iU_i$.  Then $U$ is open and dense, meeting
every irreducible component, and $U\cap X_k$ is open in $X_k$ for each $k$,
with $U=\bigsqcup_k(U\cap X_k)$ a decomposition into finitely many open
subsets, each of which is a locally closed subscheme of a stratum.  On the
open $U\cap X_k$ of $U$ the sheaf $\mathcal F|_U$ is the restriction of
$\iota_k^*\mathcal F$ along an open immersion (up to the identification of
\Cref{lemma:constructible-immersion-independence}), hence finite locally
constant by \Cref{prop:constructible-pullback}; a sheaf which is finite
locally constant on each member of a finite open cover is finite locally
constant, the union of the trivialising covers being a trivialising cover.
\end{proof}

\section{Finite locally constant sheaves}\label{sec:constructible-flc}

\begin{lemma}[Finite direct sums]\label{lemma:constructible-flc-biproduct}
\uses{def:constructible-finite-locally-constant,lemma:constructible-common-refinement}
Finite direct sums of finite locally constant sheaves are finite locally
constant; the same holds for locally constant sheaves and for finite
sheaves.
\end{lemma}
\begin{proof}
For two sheaves with trivialising data $(U_i\to X)_i$ and $(V_j\to X)_j$,
the family $(U_i\times_XV_j\to X)_{i,j}$ is an \'etale cover trivialising
both.  Restriction to a localised site is exact and preserves finite direct
sums, and a finite direct sum $M_X\oplus N_X$ of constant sheaves is the
constant sheaf $(M\oplus N)_X$, because sheafification and the constant
presheaf functor are additive.  Stalks commute with finite direct sums,
and a finite direct sum of finite sets is finite.
\end{proof}

\begin{lemma}[Morphisms between constant sheaves]
\label{lemma:constructible-hom-constant-locally}
\uses{def:constructible-constant}
Let $\mathcal A=\mathrm{Mod}_\Lambda$ with $\Lambda$ a finite ring, let
$M,N$ be finite $\Lambda$-modules and let $\varphi\colon M_X\to N_X$ be a
morphism of constant sheaves on $\et X$.  Then there is an \'etale cover
$(U_i\to X)_i$ such that $\varphi|_{U_i}$ is induced by a $\Lambda$-linear
map $f_i\colon M\to N$, i.e.\ equals $(f_i)_{U_i}\colon M_{U_i}\to N_{U_i}$.
Consequently the kernel and the cokernel of $\varphi$ are locally constant,
with restrictions $(\ker f_i)_{U_i}$ and $(\operatorname{coker}f_i)_{U_i}$.
\end{lemma}
\begin{proof}
By the adjunction between constant sheaves and global sections,
$\Hom(M_X,N_X)=\Hom_\Lambda(M,\Gamma(X,N_X))$.  A global section of $N_X$
over $X$ is, on an \'etale cover, a section of the constant presheaf, that
is, an element of $N$; hence a $\Lambda$-linear map
$M\to\Gamma(X,N_X)$ is, on an \'etale cover $(U_i\to X)_i$ obtained by
combining the covers for a finite generating set of $M$
(\Cref{lemma:constructible-flc-biproduct}, the cover for finitely many
sections), given by a $\Lambda$-linear map $f_i\colon M\to N$, and
$\varphi|_{U_i}$ is the morphism of constant sheaves induced by $f_i$.
Since restriction to $\et{U_i}$ is exact and the constant sheaf functor is
exact (a composite of the exact constant presheaf functor and
sheafification), the kernel and cokernel of $\varphi|_{U_i}$ are the
constant sheaves on $\ker f_i$ and $\operatorname{coker}f_i$.
\end{proof}

\begin{lemma}[Torsors under finite locally constant sheaves]
\label{lemma:constructible-torsor-trivialises}
\uses{def:constructible-finite-locally-constant}
Let $\mathcal G$ be a finite locally constant sheaf of groups on $\et X$
and let $P$ be a $\mathcal G$-torsor.  Then $P$ is trivial on an \'etale
cover of $X$.
\end{lemma}
\begin{proof}
Working on the members of a trivialising cover of $\mathcal G$ we may
assume $\mathcal G=G_X$ for a finite group $G$.  A $G_X$-torsor is
representable by a scheme $P\to X$ finite \'etale (locally on $X$ it is
$\bigsqcup_{g\in G}X$, and the transition functions are locally constant
permutations; see \cite[Exp.~IX, 2.1]{SGA4}); since $P\to X$ is \'etale and
surjective it is an \'etale cover, and $P$ restricted to $P$ has the
tautological section, so it is trivial there.
\end{proof}

\begin{lemma}[Kernels, cokernels and extensions of finite locally constant sheaves]
\label{lemma:constructible-flc-serre}
\uses{def:constructible-finite-locally-constant,lemma:constructible-hom-constant-locally,lemma:constructible-torsor-trivialises,lemma:constructible-flc-biproduct}
Let $\mathcal A=\mathrm{Mod}_\Lambda$ with $\Lambda$ a finite ring.  The
kernel and the cokernel of a morphism between finite locally constant
sheaves, and the middle term of a short exact sequence whose outer terms
are finite locally constant, are finite locally constant.
\end{lemma}
\begin{proof}
Restricting to an \'etale cover trivialising both sheaves
(\Cref{lemma:constructible-flc-biproduct}), we may assume the morphism is
$\varphi\colon M_X\to N_X$ with $M$, $N$ finite (the stalks of the constant
sheaf $M_X$ are $M$).  By \Cref{lemma:constructible-hom-constant-locally}
kernel and cokernel are locally constant with finite stalks.

For an extension $0\to M_X\to\mathcal E\to N_X\to0$, again on a
trivialising cover, consider the sheaf $P$ of sections of
$\mathcal E\to N_X$: over $U\to X$ its sections are the morphisms
$s\colon N_U\to\mathcal E|_U$ with $\pi\circ s=\mathrm{id}$.  It is a torsor
under the sheaf $\mathcal{H}om(N_X,M_X)=\Hom_\Lambda(N,M)_X$ (two sections
differ by a morphism $N_U\to M_U$, and locally sections exist since the
sequence is locally split in the \'etale topology: the stalks are finite
modules and a surjection of sheaves is locally surjective on sections).
The group $\Hom_\Lambda(N,M)$ is finite, so by
\Cref{lemma:constructible-torsor-trivialises} $P$ has a section over an
\'etale cover $(U_i\to X)$, and over $U_i$ the sequence splits:
$\mathcal E|_{U_i}\cong M_{U_i}\oplus N_{U_i}=(M\oplus N)_{U_i}$.  Hence
$\mathcal E$ is locally constant, with finite stalks $M\oplus N$.
\end{proof}

\section{The open--closed d\'evissage}\label{sec:constructible-devissage}

Let $(j\colon U\to X,\ i\colon Z\to X)$ be a complementary pair on $X$
(\Cref{sec:openclosed-setting}).

\begin{proposition}[Open--closed criterion]
\label{prop:constructible-open-closed}
\uses{def:constructible-constructible,prop:constructible-pullback,lemma:constructible-iso-invariance}
\lean{AlgebraicGeometry.Scheme.isConstructibleSheaf_iff_upperStar,
  AlgebraicGeometry.Scheme.Stratification.openClosed}
\leanok
A sheaf $\mathcal F\in\shv{\et X}$ is constructible if and only if
$j^*\mathcal F$ and $i^*\mathcal F$ are constructible.
\end{proposition}
\begin{proof}
\leanok
``Only if'' is \Cref{prop:constructible-pullback}.  Conversely let
$(\kappa_l\colon W_l\to U)_{l<m}$ and $(\lambda_k\colon V_k\to Z)_{k<n}$ be
stratifications witnessing the constructibility of $j^*\mathcal F$ and
$i^*\mathcal F$.  The family of the $j\circ\kappa_l$ and the
$i\circ\lambda_k$ is a stratification of $X$ with $m+n$ strata: each
composite is an immersion; the images $j(U_l)$ are locally closed in $X$,
being locally closed in the open $U$; the images $i(Z_k)$ are locally
closed in $X$, being locally closed in the closed $Z$; and the images are
pairwise disjoint and cover $X$ because the $U_l$ partition $U$, the $Z_k$
partition $Z$, and $X=U\sqcup Z$.  Finally
$(j\circ\kappa_l)^*\mathcal F\cong\kappa_l^*j^*\mathcal F$ and
$(i\circ\lambda_k)^*\mathcal F\cong\lambda_k^*i^*\mathcal F$ are finite
locally constant by construction and \Cref{lemma:constructible-iso-invariance}.
\end{proof}

\begin{lemma}[Pushforward along a closed immersion]
\label{lemma:constructible-lowerStar-closed}
\uses{prop:constructible-open-closed,lemma:openclosed-vanishing-lowerStar,prop:openclosed-lowerStar-fully-faithful}
\lean{AlgebraicGeometry.Scheme.IsConstructibleSheaf.lowerStar_of_isClosedImmersion}
\leanok
If $\mathcal G\in\shv{\et Z}$ is constructible, then so is
$i_*\mathcal G$.
\end{lemma}
\begin{proof}
\leanok
By \Cref{prop:constructible-open-closed} it suffices to see that
$j^*i_*\mathcal G$ and $i^*i_*\mathcal G$ are constructible.  The first is
zero by \Cref{lemma:openclosed-vanishing-lowerStar}, and the zero sheaf is
constructible (the trivial stratification with one stratum; the zero sheaf
is constant with finite stalks).  The second is isomorphic to $\mathcal G$
by the counit isomorphism of
\Cref{prop:openclosed-lowerStar-fully-faithful}, hence constructible by
\Cref{lemma:constructible-iso-invariance}.
\end{proof}

\begin{lemma}[Extension by zero along an open immersion]
\label{lemma:constructible-lowerShriek-open}
\uses{prop:constructible-open-closed,lemma:openclosed-vanishing-lowerShriek}
\lean{AlgebraicGeometry.Scheme.IsConstructibleSheaf.openImmersionLowerShriek}
\leanok
If $\mathcal G\in\shv{\et U}$ is constructible, then so is
$j_!\mathcal G$.
\end{lemma}
\begin{proof}
\leanok
As before: $i^*j_!\mathcal G=0$ by
\Cref{lemma:openclosed-vanishing-lowerShriek}, and $j^*j_!\mathcal G\cong
\mathcal G$ by the unit isomorphism of the adjunction $j_!\dashv j^*$
(\Cref{sec:openclosed-setting}).
\end{proof}

\begin{theorem}[Constructible sheaves form a weak Serre subcategory]
\label{thm:constructible-serre}
\uses{def:constructible-constructible,lemma:constructible-common-refinement,lemma:constructible-of-refines,lemma:constructible-flc-serre,prop:constructible-pullback}
Let $\mathcal A=\mathrm{Mod}_\Lambda$ with $\Lambda$ a finite ring.  The
kernel and the cokernel of a morphism between constructible sheaves on
$\et X$, finite direct sums of constructible sheaves, and the middle term of
a short exact sequence whose outer terms are constructible, are
constructible.  Consequently the constructible sheaves form a weak Serre
subcategory of $\shv{\et X}$.
\end{theorem}
\begin{proof}
Let $\varphi\colon\mathcal F\to\mathcal F'$ be a morphism of constructible
sheaves.  By \Cref{lemma:constructible-common-refinement} and
\Cref{lemma:constructible-of-refines} there is one stratification
$(\iota_k\colon Z_k\to X)_k$ such that all $\iota_k^*\mathcal F$ and
$\iota_k^*\mathcal F'$ are finite locally constant.  Since $\iota_k^*$ is
exact, $\iota_k^*\ker\varphi=\ker(\iota_k^*\varphi)$ and
$\iota_k^*\operatorname{coker}\varphi=\operatorname{coker}(\iota_k^*\varphi)$
are finite locally constant by \Cref{lemma:constructible-flc-serre}, so
$\ker\varphi$ and $\operatorname{coker}\varphi$ are constructible with
respect to the same stratification.  The same argument, using exactness of
$\iota_k^*$ and the extension part of \Cref{lemma:constructible-flc-serre},
handles extensions, and \Cref{lemma:constructible-flc-biproduct} handles
finite direct sums.  A full subcategory of an abelian category closed under
kernels, cokernels and extensions is a weak Serre subcategory.
\end{proof}

\begin{proposition}[The constructible derived category]
\label{prop:constructible-derived-subcategory}
\uses{thm:constructible-serre}
Let $\mathcal A=\mathrm{Mod}_\Lambda$ with $\Lambda$ a finite ring.  The
full subcategory $D^b_c(\et X)$ of the bounded derived category
$D^b(\shv{\et X})$ consisting of the complexes all of whose cohomology
sheaves are constructible is a triangulated subcategory.
\end{proposition}
\begin{proof}
For a weak Serre subcategory $\mathcal C$ of an abelian category, the
complexes with cohomology in $\mathcal C$ form a triangulated subcategory
of the derived category: the zero object and shifts are clear, and for a
distinguished triangle $K\to L\to M\to K[1]$ with $K$ and $M$ having
cohomology in $\mathcal C$, the long exact cohomology sequence
$H^{n}(K)\to H^n(L)\to H^n(M)$ exhibits $H^n(L)$ as an extension of a
subobject of $H^n(M)$ (a kernel) by a quotient of $H^n(K)$ (a cokernel),
both in $\mathcal C$, hence $H^n(L)\in\mathcal C$.  Apply this to the weak
Serre subcategory of \Cref{thm:constructible-serre}.
\end{proof}
