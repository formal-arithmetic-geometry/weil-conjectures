\begin{center}
{\large\bfseries Finiteness of Derived Direct Image\par}
\medskip
{\normalsize From ambient pushforward to bounded constructibility\par}
\end{center}

\begin{abstract}
For finite torsion coefficients, ordinary direct image is first derived on
the ambient sheaf category by $K$-injective replacement.  Exceptional direct
image is first constructed on $D^+$ from compactifications and is independent
of every compactification choice.  Finite morphisms, curve finiteness,
affine-space reduction, noetherian devissage, and finite Cech assembly then
give bounded constructible and finite-Tor outputs.  Coefficient change and
etale--pro-etale comparison preserve these outputs.  All derived categories
are ordinary Verdier-localized triangulated categories.
\end{abstract}

\section{Finite coefficients and constructible categories}

\begin{definition}[Finite torsion coefficient datum]
Fix an integer $N>0$ and a finite commutative ring $\Lambda$ such that
$N\Lambda=0$.  A scheme is \emph{$\Lambda$-admissible} if $N$ is invertible
in every local ring.  Throughout the finite-level theory, schemes are
noetherian and $\Lambda$-admissible unless a different hypothesis is stated.

Let $E/\mathbf Q_\ell$ be a finite extension, let $\mathcal O$ be its ring
of integers, and let $\varpi$ be a uniformizer.  The principal tower is
\[
 \Lambda_n=\mathcal O/\varpi^n\mathcal O,\qquad n\geq1,
\]
Its residue field is finite, and every $\Lambda_n$ is finite.  The standing
condition is that $\ell$ is invertible on every scheme under consideration.
\deps{Only the elementary notions of a finite ring, an annihilating integer,
and invertibility of a global integer on a scheme.}
\end{definition}

\begin{definition}[The selected sites and the coefficient sheaf]
The small etale site $X_{\mathrm{et}}$ has etale morphisms $U\to X$ as
objects and jointly surjective families of etale morphisms as coverings.
The small pro-etale site $X_{\mathrm{proet}}$ has weakly etale morphisms
$U\to X$ as objects.  A covering family consists of weakly etale arrows
$U_i\to U$ over $X$ which are jointly surjective and have the following
quasi-compact refinement property: for every affine open $V\subset U$,
finitely many affine opens contained in the $U_i\times_UV$ have images
covering $V$.  In each case choose
a small skeleton closed under finite fibre products and containing the final
object $X$.

For $\tau\in\{\mathrm{et},\mathrm{proet}\}$, the coefficient sheaf
$\Lambda_{X,\tau}$ is the sheafification of the constant
$\Lambda$-presheaf.  Its sections are locally constant maps to the finite
discrete ring $\Lambda$; on a pro-etale object $U$ they may be written as
continuous maps $|U|\to\Lambda$.

The inclusion of etale objects into pro-etale objects defines a morphism of
topoi
\[
 \epsilon_X:\Sh(X_{\mathrm{proet}})\longrightarrow
             \Sh(X_{\mathrm{et}}),
\]
whose inverse image is denoted $\epsilon_X^{-1}$.
\deps{The finite coefficient datum and the definitions of the small etale
and small pro-etale topologies.}
\end{definition}

\begin{definition}[Ambient module-sheaf and complex categories]
Set
\[
 \mathcal A_\tau(X,\Lambda)
   :=\Mod(X_\tau,\Lambda_{X,\tau}).
\]
Its objects are sheaves of $\Lambda_{X,\tau}$-modules and its arrows are
$\Lambda_{X,\tau}$-linear sheaf morphisms.  Write
\[
 \operatorname{Ch}(X_\tau,\Lambda),\qquad
 K(X_\tau,\Lambda),\qquad
 D(X_\tau,\Lambda)
\]
for the categories of cochain complexes, the homotopy category, and the
Verdier localization of the homotopy category at quasi-isomorphisms.  The
localization functor is $Q_X:K(X_\tau,\Lambda)\to D(X_\tau,\Lambda)$.
\deps{The selected site and its coefficient sheaf.}
\end{definition}

\begin{theorem}[Grothendieck and triangulated structures]
The category $\mathcal A_\tau(X,\Lambda)$ is Grothendieck abelian.  In
particular it has exact filtered colimits, a generator, enough injectives,
and all small limits and colimits.  The category $D(X_\tau,\Lambda)$ is an
ordinary triangulated category: its translation is the cohomological shift,
and its distinguished triangles are the triangles isomorphic to localized
mapping-cone triangles.
\deps{The module-sheaf category on a small ringed site and ordinary Verdier
localization.}
\end{theorem}

\begin{definition}[Finite locally constant and constructible sheaves]
A sheaf $F\in\mathcal A_\tau(X,\Lambda)$ is \emph{finite locally constant}
if a covering $\{U_a\to X\}$ exists such that
\[
 F|_{U_a}\cong (M_a)_{U_a}
\]
for finite $\Lambda$-modules $M_a$.

A \emph{finite constructible stratification} is a finite partition
\[
 X=\coprod_{\alpha\in A}X_\alpha
\]
by reduced locally closed subschemes with constructible underlying subsets.
The sheaf $F$ is \emph{constructible} if it is finite locally constant on
every stratum of some such stratification.  Let
\[
 \Cons_\tau(X,\Lambda)\subset\mathcal A_\tau(X,\Lambda)
\]
be the full subcategory of constructible sheaves.
\deps{The ambient module-sheaf category and finite stratifications of a
noetherian scheme.}
\end{definition}

\begin{proposition}[The constructible Serre subcategory]
The full subcategory $\Cons_\tau(X,\Lambda)$ is closed under kernels,
cokernels, images, finite direct sums, subquotients, and extensions.  It is
an abelian Serre subcategory of $\mathcal A_\tau(X,\Lambda)$, and
its inclusion is exact.
\deps{Common refinements of finitely many constructible stratifications and
closure of finite $\Lambda$-modules under kernels, cokernels, and extensions.}
\end{proposition}

\begin{definition}[Bounded constructible and finite-Tor complexes]
For $K\in D(X_\tau,\Lambda)$, let $\mathcal H^i(K)$ be its cohomology sheaf.
Define
\[
 \Dbc(X_\tau,\Lambda)
 =\left\{K:\begin{array}{l}
       \mathcal H^i(K)\in\Cons_\tau(X,\Lambda)\text{ for every }i,\\
       \mathcal H^i(K)=0\text{ for all but finitely many }i
   \end{array}\right\}.
\]
For integers $a\leq b$, the notation $D_c^{[a,b]}$ means that the nonzero
cohomology sheaves occur only in degrees $a,\ldots,b$.

The complex $K$ has \emph{Tor-amplitude in $[r,s]$} if, for every
$M\in\mathcal A_\tau(X,\Lambda)$ placed in degree zero,
\[
 \mathcal H^i(K\otimes^L_{\Lambda_X}M)=0
 \qquad(i\notin[r,s]).
\]
The category $\Dctf(X_\tau,\Lambda)$ is the full subcategory of
$\Dbc(X_\tau,\Lambda)$ on objects of finite Tor-amplitude.  Finite
Tor-amplitude is not part of the definition of constructibility.
\deps{The ambient derived category, its standard cohomology functors, and the
constructible Serre subcategory.}
\end{definition}

\begin{proposition}[Triangulated closure]
The category $\Dbc(X_\tau,\Lambda)$ is a strictly full triangulated
subcategory of $D(X_\tau,\Lambda)$.  It is stable under standard truncations.
If $u:K\to L$ is a morphism between bounded constructible complexes, the
constructibility of its cone follows degreewise from
\[
 0\longrightarrow\operatorname{coker}\mathcal H^{i}(u)
 \longrightarrow\mathcal H^i(\operatorname{Cone}u)
 \longrightarrow\ker\mathcal H^{i+1}(u)\longrightarrow0.
\]
The finite-Tor subcategory is stable under shifts, cones, and direct
summands.
\deps{The constructible Serre property, the long exact cohomology sequence
of a cone triangle, and the corresponding Tor-amplitude inequalities.}
\end{proposition}

\begin{theorem}[Finite etale--pro-etale comparison]
Inverse image is exact and induces $t$-exact triangulated equivalences
\[
 \epsilon_X^{-1}:\Dbc(X_{\mathrm{et}},\Lambda)
   \xrightarrow{\ \sim\ }
   \Dbc(X_{\mathrm{proet}},\Lambda),
 \qquad
 \epsilon_X^{-1}:\Dctf(X_{\mathrm{et}},\Lambda)
   \xrightarrow{\ \sim\ }
   \Dctf(X_{\mathrm{proet}},\Lambda).
\]
They preserve cohomology sheaves, shifts, cones, finite Tor-amplitudes, and
same-coefficient derived pullback.
\deps{The section formula for $\epsilon_X^{-1}$, full faithfulness on etale
sheaves, descent of finite locally constant pro-etale sheaves, finite
stratum induction, and bounded derived truncation.}
\end{theorem}

\section{The ambient derived direct image}

Let $f:X\to Y$ be a morphism of $\Lambda$-admissible schemes.  This section
does not assume that $f$ is proper or of finite type.

\begin{definition}[Underived pullback and direct image]
On the same finite constant coefficient ring, inverse image is exact and
gives
\[
 f^*=f^{-1}:\mathcal A_\tau(Y,\Lambda)\longrightarrow
              \mathcal A_\tau(X,\Lambda).
\]
Its right adjoint is the left-exact functor
\[
 f_*:\mathcal A_\tau(X,\Lambda)\longrightarrow
       \mathcal A_\tau(Y,\Lambda),
 \qquad
 (f_*F)(V)=F(V\times_YX).
\]
The adjunction unit and counit are
\[
 \eta:\operatorname{id}\longrightarrow f_*f^*,
 \qquad
 \varepsilon:f^*f_*\longrightarrow\operatorname{id}.
\]
\deps{The morphism of ringed sites induced by $f$ and the equality of the
pulled-back finite constant coefficient sheaves.}
\end{definition}

\begin{definition}[$K$-injective complexes]
A complex $I\in\operatorname{Ch}(X_\tau,\Lambda)$ is \emph{$K$-injective}
if the complex of abelian groups $\Hom^\bullet(A,I)$ is acyclic for every
acyclic complex $A$.  Equivalently,
\[
 \Hom_{K(X_\tau,\Lambda)}(A,I)=0
\]
for every acyclic $A$.  A $K$-injective replacement of $K$ is a
quasi-isomorphism $\iota_K:K\to I_X(K)$ with $I_X(K)$ $K$-injective.
\deps{The complex and homotopy categories of the Grothendieck abelian
category $\mathcal A_\tau(X,\Lambda)$.}
\end{definition}

\begin{theorem}[Functorial $K$-injective replacement]
There is a functor $I_X$ on the complex category and a natural
quasi-isomorphism
\[
 \iota:\operatorname{id}\longrightarrow I_X
\]
such that every $I_X(K)$ is $K$-injective.  A quasi-isomorphism between
$K$-injective complexes is a homotopy equivalence.  Any two such replacement
functors determine canonically naturally isomorphic functors on the derived
category.
\deps{The Grothendieck property of $\mathcal A_\tau(X,\Lambda)$ and the
unbounded $K$-injective resolution theorem.}
\end{theorem}

\begin{proposition}[Descent of the resolved direct image]
The complex functor
\[
 \operatorname{Ch}(X_\tau,\Lambda)\longrightarrow
 \operatorname{Ch}(Y_\tau,\Lambda),
 \qquad B\longmapsto f_*I_X(B),
\]
sends every quasi-isomorphism to a quasi-isomorphism.  The unique
factorization of $Q_Yf_*I_X$ through $Q_X$ is an exact triangulated functor
$Rf_*$ equipped with a natural isomorphism
\[
 Rf_*Q_X\xrightarrow{\ \sim\ }Q_Yf_*I_X.
\]
\deps{The two-out-of-three property for $\iota:\operatorname{id}\to I_X$,
the fact that a quasi-isomorphism between $K$-injectives is a homotopy
equivalence, preservation of chain homotopies by $f_*$, and the universal
property of Verdier localization.}
\end{proposition}

\begin{definition}[Ambient derived direct image]
Define the ambient derived direct image to be the unique factor in the
preceding proposition:
\[
 Rf_*:D(X_\tau,\Lambda)\longrightarrow D(Y_\tau,\Lambda),
 \qquad
 Rf_*Q_X(B)\cong Q_Y\bigl(f_*I_X(B)\bigr).
\]
On a localized morphism represented after $K$-injective replacement by a
chain map $u$, its value is
\[
 Rf_*(u):=Q_Y\bigl(f_*I_X(u)\bigr).
\]
For a sheaf $F$ in degree zero, define
\[
 R^qf_*F:=\mathcal H^q(Rf_*F).
\]
The displayed functor is ambient: its definition contains no boundedness or
constructibility conclusion.
\deps{Descent of the resolved direct image and the factorization isomorphism
through Verdier localization.}
\end{definition}

\begin{proposition}[Derived pullback--direct-image adjunction]
Exact pullback applied termwise defines an exact triangulated functor
\[
 Lf^*=f^*:D(Y_\tau,\Lambda)\longrightarrow D(X_\tau,\Lambda).
\]
There are natural bijections
\[
 \Hom_{D(X_\tau,\Lambda)}(Lf^*L,K)
 \cong
 \Hom_{D(Y_\tau,\Lambda)}(L,Rf_*K),
\]
with derived unit $\operatorname{id}\to Rf_*Lf^*$ and derived counit
$Lf^*Rf_*\to\operatorname{id}$.  Their two triangle identities hold.
\deps{The exact underived adjunction $f^*\dashv f_*$, $K$-injective
replacement of the right argument, and descent through Verdier
localization.}
\end{proposition}

\begin{lemma}[Preservation of $K$-injectives]
The functor $f_*$ sends $K$-injective complexes to $K$-injective complexes.
For every complex $A$ and every complex $I$, there is a natural isomorphism
of Hom complexes
\[
 \Hom^\bullet(A,f_*I)\cong\Hom^\bullet(f^*A,I),
\]
compatible with differentials and composition.
\deps{The exact adjunction $f^*\dashv f_*$, preservation of acyclic
complexes by the exact functor $f^*$, and the definition of a $K$-injective
complex.}
\end{lemma}

\begin{proposition}[Strict-local stalk formula]
Assume that $f$ is quasi-compact and quasi-separated.  Let
$\bar y\to Y$ be a geometric point, let
\[
 Y_{(\bar y)}=\Spec(\mathcal O^{\mathrm{sh}}_{Y,\bar y}),
 \qquad
 X_{(\bar y)}=X\times_YY_{(\bar y)},
\]
and let $p:X_{(\bar y)}\to X$ be the projection.  For every etale sheaf $F$
and every $q\geq0$, there is a natural isomorphism
\[
 (R^qf_*F)_{\bar y}
 \xrightarrow{\ \sim\ }
 H^q(X_{(\bar y),\mathrm{et}},p^{-1}F).
\]
For $K\in D^+(X_{\mathrm{et}},\Lambda)$ the corresponding derived statement
is
\[
 (Rf_*K)_{\bar y}
 \xrightarrow{\ \sim\ }
 R\Gamma(X_{(\bar y),\mathrm{et}},p^{-1}K).
\]
\deps{Cofinality of affine pointed etale neighbourhoods, the inverse-limit
description of strict henselization, continuity of etale cohomology, and the
$K$-injective definition of $Rf_*$.}
\end{proposition}

\section{Finite morphisms and composition}

\begin{theorem}[Geometric-stalk formula for a finite morphism]
\label{thm:finite-stalk}
\textbf{Hypotheses.} Let $v:Z\to T$ be finite, let $G$ be an etale
$\Lambda$-module sheaf on $Z$, and let $\bar t\to T$ be a geometric point.

\textbf{Map.} Restriction of a section to the finite geometric fibre defines
\[
 (v_*G)_{\bar t}\longrightarrow
 \prod_{\bar z\in |Z_{\bar t}|}G_{\bar z}.
\]

\textbf{Conclusion.} This map is an isomorphism, and its target is also the
corresponding finite direct sum.
\deps{The strict-local stalk formula, invariance of the etale topos under
nilpotent thickenings and universal homeomorphisms, and the decomposition of
a finite scheme over a strictly henselian local base into its local factors.}
\end{theorem}

\begin{corollary}[Exact finite direct image]
\label{cor:finite-exact}
\textbf{Hypotheses.} Let $v:Z\to T$ be finite.

\textbf{Source and target.}
\[
 v_*:\mathcal A_{\mathrm{et}}(Z,\Lambda)\longrightarrow
      \mathcal A_{\mathrm{et}}(T,\Lambda).
\]

\textbf{Conclusion.} The functor $v_*$ is exact and
\[
 R^qv_*=0\qquad(q>0).
\]
If $G$ is constructible, then $v_*G$ is constructible.  The exact functor
induces a $t$-exact triangulated functor
\[
 Rv_*:\Dbc(Z_{\mathrm{et}},\Lambda)\longrightarrow
       \Dbc(T_{\mathrm{et}},\Lambda)
\]
and maps $\Dctf$ to $\Dctf$ without enlarging
Tor-amplitude.
\deps{The finite-morphism stalk formula, exactness and conservativity of
geometric stalks, finite stratification of a finite morphism, the
constructible Serre property, and the finite-morphism projection formula for
the finite-Tor assertion.}
\end{corollary}

\begin{theorem}[Derived composition]
\label{thm:composition}
\textbf{Hypotheses.} Let $X\xrightarrow{f}Y\xrightarrow{g}Z$ be composable
morphisms of $\Lambda$-admissible schemes.

\textbf{Source and target.} Both sides below are exact triangulated functors
from $D(X_\tau,\Lambda)$ to $D(Z_\tau,\Lambda)$.

\textbf{Comparison.} The equality $(g\circ f)_*=g_*f_*$ before deriving
induces a canonical natural isomorphism
\[
 c_{g,f}:R(g\circ f)_*\xrightarrow{\ \sim\ }Rg_*\,Rf_*.
\]

\textbf{Conclusion.} The comparison is the identity when either morphism is
an identity, and for three composable morphisms its two iterated composites
agree.
\deps{Preservation of $K$-injectives by direct image, functorial
$K$-injective replacement, and associativity of underived direct image.}
\end{theorem}

\begin{proposition}[Addition of cohomological bounds]
\label{prop:add-cd}
Suppose $f$ has cohomological dimension at most $d$ on a class of sheaves
$\mathcal C_X$, $g$ has cohomological dimension at most $e$ on a class
$\mathcal C_Y$, and every $R^qf_*F$ with $F\in\mathcal C_X$ belongs to
$\mathcal C_Y$.  Then $g\circ f$ has cohomological dimension at most $d+e$
on $\mathcal C_X$.  The comparison is computed by the convergent
Grothendieck spectral sequence
\[
 E_2^{p,q}=R^pg_*R^qf_*F
 \Longrightarrow R^{p+q}(g\circ f)_*F.
\]
\deps{Derived composition and the Grothendieck spectral sequence for the
composite of two left-exact functors.}
\end{proposition}

\section{Ambient compactifiable direct image}

\begin{definition}[Open extension by zero]
Let $j:U\hookrightarrow\overline X$ be a quasi-compact open immersion and
let $F\in\mathcal A_{\mathrm{et}}(U,\Lambda)$.  For an etale map
$V\to\overline X$, set $V_U=V\times_{\overline X}U$.  For
$s\in F(V_U)$, define its support in $V$ by
\[
 \operatorname{Supp}_V(s)
 :=V_U\setminus
 \bigcup_{\substack{W\subseteq V_U\text{ open}\\ s|_W=0}}W
 \subseteq V.
\]
The restriction maps of $F$ make
\[
 (j^{\mathrm{pre}}_!F)(V)
 :=\{s\in F(V_U):\operatorname{Supp}_V(s)\text{ is closed in }V\}
\]
a presheaf of $\Lambda$-modules on $\overline X_{\mathrm{et}}$.  If
$a_{\overline X}$ denotes associated sheaf, define
\[
 j_!F:=a_{\overline X}(j^{\mathrm{pre}}_!F).
\]
Restriction of morphisms of sheaves defines
$j_!:\mathcal A_{\mathrm{et}}(U,\Lambda)\to
\mathcal A_{\mathrm{et}}(\overline X,\Lambda)$.
\deps{The etale sites, the coefficient sheaves, support of a sheaf section,
and sheafification of a presheaf of modules.}
\end{definition}

\begin{proposition}[Exactness and stalks of open extension]
The functor $j_!$ is left adjoint to restriction
$j^*=j^{-1}$ and is exact.  For every geometric point
$\bar x\to\overline X$ there is a natural identification
\[
 (j_!F)_{\bar x}= 
 \begin{cases}
   F_{\bar x},&\bar x\text{ factors through }U,\\
   0,&\bar x\text{ factors through }\overline X\setminus U.
 \end{cases}
\]
Termwise application sends quasi-isomorphisms to quasi-isomorphisms and
induces an exact triangulated functor
\[
 j_!:D^+(U_{\mathrm{et}},\Lambda)\longrightarrow
      D^+(\overline X_{\mathrm{et}},\Lambda).
\]
\deps{The support-closed presheaf defining $j_!$, exactness detected on
geometric stalks, and exactness and conservativity of geometric stalks.}
\end{proposition}

\begin{theorem}[Bounded-below finite-torsion proper base change]
\label{thm:ambient-proper-bc}
Let
\[
\begin{CD}
 X'@>{g'}>>X\\
 @V{p'}VV @VV{p}V\\
 Y'@>{g}>>Y
\end{CD}
\]
be Cartesian, with $p$ proper.  For every
$K\in D^+(X_{\mathrm{et}},\Lambda)$, the canonical exchange morphism is an
isomorphism
\[
 g^*Rp_*K\xrightarrow{\ \sim\ }Rp'_*g'^*K.
\]
It is natural in $K$, is the identity for an identity square, and satisfies
horizontal and vertical pasting.  For every sheaf $F$ of $\Lambda$-modules
and every geometric point $\bar y\to Y$, the case $K=F[0]$ gives
\[
 (R^qp_*F)_{\bar y}\xrightarrow{\ \sim\ }
 H^q(X_{\bar y},F|_{X_{\bar y}}),
 \qquad q\geq0.
\]
The theorem for torsion abelian sheaves transports to $\Lambda$-module
sheaves through the exact conservative forgetful functor.  This functor
commutes with $g^*$ and derived direct image and carries the $\Lambda$-linear
exchange map to the underlying abelian exchange map.
\deps{Finite-torsion proper base change for arbitrary torsion sheaves, its
bounded-below hypercohomology extension, and the exact conservative forgetful
functor from $\Lambda$-module sheaves.}
\end{theorem}

\begin{theorem}[Open--proper interchange on $D^+$]
\label{thm:open-proper-interchange}
Let
\[
\begin{CD}
 U'@>{j'}>>X'\\
 @V{p_U}VV @VV{p}V\\
 U@>{j}>>X
\end{CD}
\]
be Cartesian, where $j$ is quasi-compact open and $p$ is proper.  For every
$K\in D^+(U'_{\mathrm{et}},\Lambda)$, the stalkwise proper exchange map and
extension by zero define a natural isomorphism
\[
 j_!Rp_{U,*}K\xrightarrow{\ \sim\ }Rp_*j'_!K.
\]
The isomorphism is compatible with identity squares, horizontal and vertical
pasting, and proper composition.  If $p_U$ is the identity, it specializes to
$j_!K\simeq Rp_*j'_!K$.
\deps{Theorem~\ref{thm:ambient-proper-bc}, exact open extension by zero, its
geometric-stalk formula, and conservativity of geometric stalks.}
\end{theorem}

\begin{definition}[Compactification candidate on $D^+$]
Let $f:X\to Y$ be separated and of finite type.  For a Nagata
compactification
\[
 c=\bigl(X\xrightarrow{j_c}\overline X_c\xrightarrow{p_c}Y\bigr),
 \qquad j_c\text{ open},\quad p_c\text{ proper},\quad p_cj_c=f,
\]
define the ambient compactification candidate
\[
 Rf_{!,c}:=Rp_{c,*}j_{c,!}:
 D^+(X_{\mathrm{et}},\Lambda)\longrightarrow
 D^+(Y_{\mathrm{et}},\Lambda).
\]
\deps{Exact open extension by zero, bounded-below proper direct image, and a
Nagata compactification.}
\end{definition}

\begin{theorem}[Choice-independent ambient exceptional direct image]
\label{thm:ambient-shriek}
Let $c_1,c_2$ be compactifications of $f$.  A common schematically dense
refinement $d$ and its proper refinement maps define a natural isomorphism
\[
 \kappa_{c_1,c_2}:Rf_{!,c_1}\xrightarrow{\ \sim\ }Rf_{!,c_2}
\]
of functors $D^+(X_{\mathrm{et}},\Lambda)\to
D^+(Y_{\mathrm{et}},\Lambda)$.  It is independent of $d$ and satisfies
\[
 \kappa_{c,c}=\operatorname{id},\qquad
 \kappa_{c_2,c_3}\kappa_{c_1,c_2}=\kappa_{c_1,c_3}.
\]
After choosing one compactification $c_f$, set
\[
 Rf_!:=Rf_{!,c_f}:
 D^+(X_{\mathrm{et}},\Lambda)\longrightarrow
 D^+(Y_{\mathrm{et}},\Lambda).
\]
The maps $\kappa$ identify the functors obtained from all other choices.
\deps{Nagata compactification, product-closure common dense refinements,
open-proper interchange on $D^+$, proper direct-image composition, and the
identity and composition equations for refinement transformations.}
\end{theorem}

\begin{theorem}[Ambient composition, units, and proper normalization]
\label{thm:ambient-shriek-composition}
For separated finite-type maps $X\xrightarrow{f}Y\xrightarrow{g}Z$, there
are canonical natural isomorphisms
\[
 c^!_{g,f}:Rg_!Rf_!\xrightarrow{\ \sim\ }R(gf)_!,
 \qquad
 u_X:R(\operatorname{id}_X)_!\xrightarrow{\ \sim\ }
      \operatorname{id}_{D^+(X_{\mathrm{et}},\Lambda)}.
\]
The left and right unit equations hold.  For three composable maps,
\[
 c^!_{h,gf}\circ(Rh_!c^!_{g,f})
 =c^!_{hg,f}\circ(c^!_{h,g}Rf_!).
\]
If $f$ is proper, the identity compactification gives a canonical
normalization
\[
 Rf_!\xrightarrow{\ \sim\ }Rf_*
\]
on $D^+(X_{\mathrm{et}},\Lambda)$, compatible with the units and
compositors.
\deps{Choice-independent ambient exceptional direct image, successive
compactifications, open-proper interchange, and associativity of proper
direct image.}
\end{theorem}

\begin{theorem}[Ambient finite-torsion compact-support base change]
\label{thm:ambient-shriek-bc}
For a Cartesian square
\[
\begin{CD}
 X'@>{g'}>>X\\
 @V{f'}VV @VV{f}V\\
 Y'@>{g}>>Y
\end{CD}
\]
with $f$ separated and of finite type, the proper and open exchange maps
define a canonical natural isomorphism
\[
 \operatorname{BC}_{!,\square}:
 g^*Rf_!\xrightarrow{\ \sim\ }Rf'_!g'^*
\]
of functors
$D^+(X_{\mathrm{et}},\Lambda)\to D^+(Y'_{\mathrm{et}},\Lambda)$.
It is independent of compactifications and satisfies identity, horizontal
pasting, vertical pasting, and compatibility with the compositors
$c^!_{g,f}$.
\deps{Choice-independent ambient exceptional direct image,
Theorem~\ref{thm:ambient-proper-bc}, base change for open extension, dense
replacement after base change, and refinement naturality.}
\end{theorem}

\begin{proposition}[Bounded hypercohomology spectral sequence]
\label{prop:hypercohomology}
Let $f:X\to Y$ be quasi-compact and quasi-separated.  Let $Rf_\diamond$
denote the ambient $Rf_*$; if $f$ is separated and of finite type, it may
instead denote the ambient $Rf_!$.  For a sheaf $F$, put
\[
 R^rf_\diamond F:=\mathcal H^r(Rf_\diamond(F[0])).
\]
Let $K\in D_c^{[a,b]}(X_{\mathrm{et}},\Lambda)$ and assume that
$R^rf_\diamond F=0$ for $r>c$ for every constructible cohomology sheaf
$F=\mathcal H^s(K)$.  The truncation filtration gives a functorial
convergent spectral sequence
\[
 E_2^{r,s}=R^rf_\diamond\mathcal H^s(K)
 \Longrightarrow \mathcal H^{r+s}(Rf_\diamond K),
\]
supported in the finite rectangle
\[
 0\leq r\leq c,\qquad a\leq s\leq b.
\]
If every $E_2^{r,s}$ is constructible, every abutment sheaf is
constructible and
\[
 Rf_\diamond K\in D_c^{[a,b+c]}(Y_{\mathrm{et}},\Lambda).
\]
\deps{The standard truncation filtration, the constructible Serre property,
the ambient functor $Rf_\diamond:D^+\to D^+$, and finite convergence in a
bounded rectangle.}
\end{proposition}

\section{Affine torsion cohomological dimension}

\begin{definition}[Field torsion cohomological dimension]
\label{def:field-cd}
For a field $K$ and an integer $c\geq0$, write
\[
 \operatorname{FieldCD}_N(K)\leq c
\]
if
\[
 H^r(\Spec K,F)=0\qquad(r>c)
\]
for every etale sheaf of abelian groups killed by $N$.  The same notation is
used for sheaves of $\Lambda$-modules after forgetting the module structure.
\deps{Etale cohomology of a field and the fixed invertible torsion exponent
$N$.}
\end{definition}

\begin{definition}[Strong affine cohomological dimension]
\label{def:strong-affine-cd}
For a field $K$ and integers $c,d\geq0$, the assertion
\[
 \operatorname{AffCD}_N(K;c,d)
\]
means that, whenever $\operatorname{FieldCD}_N(K)\leq c$, $X$ is affine and
of finite type over $K$ with $\dim X\leq d$, and $F$ is an arbitrary etale
sheaf killed by $N$, one has
\[
 H^r(X,F)=0\qquad(r>c+d).
\]
\deps{The field cohomological-dimension predicate and ordinary etale
cohomology.}
\end{definition}

\begin{theorem}[Continuity for arbitrary torsion sheaves]
\label{thm:torsion-continuity}
Let $X$ be noetherian and let $(F_\lambda)$ be a filtered system of torsion
abelian sheaves on $X_{\mathrm{et}}$.  For every $q$ the canonical map
\[
 \varinjlim_\lambda H^q(X,F_\lambda)
 \xrightarrow{\ \sim\ }
 H^q\!\left(X,\varinjlim_\lambda F_\lambda\right)
\]
is an isomorphism.  If $X=\varprojlim_iX_i$ is a cofiltered inverse limit of
qcqs schemes with affine transition maps and every $X_i$ has torsion
cohomological dimension at most $r$, then $X$ also has torsion
cohomological dimension at most $r$.
\deps{Finite-skeleton etale hypercover descent, exactness of filtered
colimits of sheaves, and continuity of the small etale site under affine
inverse limits.}
\end{theorem}

\begin{theorem}[Absolute torsion cohomological dimension of curves]
\label{thm:absolute-curve-cd}
Let $k$ be separably closed, let $C$ be of finite type over $k$ with
$\dim C\leq1$, and let $F$ be an arbitrary etale sheaf killed by $N$.
If $C$ is affine, then
\[
 H^q(C,F)=0\qquad(q>1).
\]
If $C$ is proper, then
\[
 H^q(C,F)=0\qquad(q>2).
\]
Both zero identifications commute with extensions between separably closed
fields and with automorphisms of the base field.
\deps{Reduction and normalization of curves, finite direct-image exactness,
open--closed localization, the strict-henselian trait calculation, primary
coefficient devissage, constructible approximation, and
Theorem~\ref{thm:torsion-continuity}.}
\end{theorem}

\begin{theorem}[Affine torsion cohomological dimension in dimensions zero
and one]
\label{thm:affine-cd-base}
For every field $K$ and every $c\geq0$,
\[
 \operatorname{FieldCD}_N(K)\leq c
 \quad\Longrightarrow\quad
 \operatorname{AffCD}_N(K;c,0),\quad
 \operatorname{AffCD}_N(K;c,1).
\]
\deps{Finite direct-image exactness, Theorem~\ref{thm:absolute-curve-cd},
the geometric-stalk formula, field cohomological dimension, and the Leray
spectral sequence.}
\end{theorem}

\begin{theorem}[Transcendence degree and field cohomological dimension]
\label{thm:field-transcendence-cd}
Suppose $\operatorname{FieldCD}_N(K)\leq c$ and let $L/K$ have finite
transcendence degree $s$.  Then
\[
 \operatorname{FieldCD}_N(L)\leq c+s.
\]
The assertion is compatible with towers of extensions and with passage to a
separable algebraic extension.
\deps{The dimension-one affine theorem, presentation of a
transcendence-degree-one field as an affine inverse limit, and
Theorem~\ref{thm:torsion-continuity}, together with topological invariance of
the etale topos under purely inseparable field extensions.}
\end{theorem}

\begin{definition}[Transcendence-degree support filtration]
\label{def:transcendence-support}
If $Z$ is of finite type over $K$ and $a\in\mathbf Z$, put
\[
 E_a(Z/K)=
 \{z\in Z:\operatorname{trdeg}_K\kappa(z)\leq a\}.
\]
For $a<0$ this set is empty.  Each $E_a(Z/K)$ is stable under
specialization.  A torsion sheaf is supported on $E_a(Z/K)$ when all its
geometric stalks outside this subset vanish.
\deps{The dimension formula for closures of points on a finite-type scheme.}
\end{definition}

\begin{theorem}[Approximation with controlled support]
\label{thm:support-approximation}
Let $Z$ be noetherian, let $E\subseteq Z$ be stable under specialization,
and let $G$ be a torsion sheaf supported on $E$.  There is a filtered system
of constructible torsion subsheaves $G_i\subseteq G$ and reduced closed
immersions $z_i:Z_i\hookrightarrow Z$ such that
\[
 G=\varinjlim_iG_i,
 \qquad
 \operatorname{Supp}(G_i)\subseteq |Z_i|\subseteq E.
\]
If $E=E_a(Z/K)$, every $Z_i$ may be chosen with $\dim Z_i\leq a$.
\deps{Finite-generator images in a torsion sheaf, noetherianity, generic
points of constructible supports, and specialization-closedness of $E$.}
\end{theorem}

\begin{theorem}[Cohomology with bounded transcendence support]
\label{thm:supported-affine-cd}
Let $d\geq1$, and let $Z$ be smooth, affine, and of finite type over $K$.
Suppose that
$\operatorname{FieldCD}_N(K)\leq c$, that $G$ is an arbitrary $N$-torsion
sheaf supported on $E_a(Z/K)$, that $a<d$, and that
$\operatorname{AffCD}_N(K;c,e)$ holds for every $e<d$.  Then
\[
 H^b(Z,G)=0\qquad(b>a+c).
\]
\deps{Theorem~\ref{thm:support-approximation}, exact direct image for a
closed immersion, the lower-dimensional strong affine assertions, and
Theorem~\ref{thm:torsion-continuity}.}
\end{theorem}

\begin{definition}[Standard affine-line compactification]
\label{def:standard-a1-compactification}
Let $K$ be a field on which $N$ is invertible and let $d>1$.  Put
$B=\mathbf A^{d-1}_K$, $\overline X=\mathbf P^1_K\times_KB$, and use the
diagram
\[
\begin{CD}
 \mathbf A^d_K=\mathbf A^1_K\times_KB @>{j}>>
 \mathbf P^1_K\times_KB\\
 @V{f}VV @VV{g}V\\
 B @= B.
\end{CD}
\]
The complement of $j$ is the effective Cartier divisor
$D=\{\infty\}\times_KB$, and $g|_D:D\xrightarrow{\sim}B$.
\deps{The standard open immersion $\mathbf A^1\subset\mathbf P^1$ and
formation of products over $K$.}
\end{definition}

\begin{definition}[Punctured strict-local dimension-drop datum]
\label{def:punctured-strict-local}
Let $K$ be a field on which $N$ is invertible, let $d\geq1$, let $Y$ be
smooth of dimension $d$ over $K$, and let $D\subset Y$ be an
effective Cartier divisor, and let $y\in D$ with
$a=\operatorname{trdeg}_K\kappa(y)$.  Fix a geometric lift $\bar y$, put
$A=\mathcal O^{\mathrm{sh}}_{Y,\bar y}$, and let $t\in A$ be a local equation
of $D$.  A dimension-drop datum consists of
\begin{enumerate}
\item a map $L=K(t_1,\ldots,t_a)^{\mathrm{sep}}\to A$ inducing a purely
      inseparable residue-field extension;
\item an injection $L(x)\to A[1/t]$ carrying $x$ to $t$;
\item a filtered presentation
      $A[1/t]=\varinjlim_\lambda B_\lambda$ in which every $B_\lambda$ is
      of finite type over $L(x)$ and has dimension at most $d-a-1$;
\item the induced affine inverse-limit and etale-cohomology continuity maps.
\end{enumerate}
\deps{Strict henselization, coefficient fields at a geometric point, and
affine inverse limits.}
\end{definition}

\begin{theorem}[Existence of the strict-local dimension drop]
\label{thm:punctured-strict-local-exists}
Every tuple $(Y,D,y,\bar y,t)$ in
Definition~\ref{def:punctured-strict-local} admits such a datum.  The
finite-type stages and their comparison maps may be chosen functorially after
passing to a cofinal subsystem.
\deps{Noether normalization around $y$, finite-type models of a strict
henselization, regularity of $Y$, inversion of a Cartier equation, the
finite-type dimension formula, and affine-limit continuity.}
\end{theorem}

\begin{theorem}[Punctured strict-local Artin bound]
\label{thm:punctured-strict-local-bound}
Assume the strong affine theorem in dimensions $<d$ over every field.  In the
notation of Definition~\ref{def:punctured-strict-local}, every $N$-torsion
sheaf $F$ on $\Spec A[1/t]$ satisfies
\[
 H^q(\Spec A[1/t],F)=0\qquad(q>d-a).
\]
Here $L$ is separably closed, $\operatorname{FieldCD}_N(L(x))\leq1$, and the
finite-type stages have dimension at most $d-a-1$.
\deps{Theorem~\ref{thm:punctured-strict-local-exists},
Theorem~\ref{thm:field-transcendence-cd}, the lower-dimensional strong affine
assertions, and Theorem~\ref{thm:torsion-continuity}.}
\end{theorem}

\begin{theorem}[Support at infinity]
\label{thm:open-higher-support}
Let $d>1$, let $K$ be a field on which $N$ is invertible, and let
$B$, $\overline X$, $j$, and $D$ be as in
Definition~\ref{def:standard-a1-compactification}.  Assume
$\operatorname{AffCD}_N(L;c',e)$ for every field $L$, every $c'\geq0$, and
every $e<d$.  Let $F$ be an arbitrary $N$-torsion sheaf on
$\mathbf A^d_K$.  For every $q>0$,
\[
 \operatorname{Supp}(R^qj_*F)\subseteq E_{d-q}(D/K),
\]
and $R^qj_*F=0$ for $q>d$.  More precisely, at a point
$y\in D$ of transcendence degree $a$, the geometric stalk is
\[
 (R^qj_*F)_{\bar y}
 \cong H^q(\Spec\mathcal O^{\mathrm{sh}}_{\overline X,\bar y}[1/t],F)
\]
and is zero for $q>d-a$.
\deps{The strict-local stalk formula for an open immersion,
Theorem~\ref{thm:punctured-strict-local-bound}, and conservativity of
geometric stalks.}
\end{theorem}

\begin{theorem}[Relative projective-line amplitude]
\label{thm:relative-p1-cd}
Let $d>1$, let $K$ be a field on which $N$ is invertible, and let
$B$, $g$, and $D$ be as in
Definition~\ref{def:standard-a1-compactification}.
For every $N$-torsion sheaf $H$ on
$\mathbf P^1_K\times_KB$,
\[
 R^pg_*H=0\qquad(p>2).
\]
This assertion is valid for arbitrary torsion sheaves and is compatible with
strict henselization at points of $B$.
\deps{Theorem~\ref{thm:ambient-proper-bc}, its arbitrary-sheaf geometric
stalk formula, Theorem~\ref{thm:absolute-curve-cd}, and geometric-stalk
conservativity.}
\end{theorem}

\begin{theorem}[The relative Leray rows at infinity]
\label{thm:a1-relative-rows}
Let $d>1$, let $K$ be a field on which $N$ is invertible, let $F$ be an
arbitrary $N$-torsion sheaf on $\mathbf A^d_K$, and let $f$, $g$, $j$, $B$,
and $D$ be as in Definition~\ref{def:standard-a1-compactification}.
Assume $\operatorname{AffCD}_N(L;c',e)$ for every field $L$, every
$c'\geq0$, and every $e<d$.
For $f=gj$ as in Definition~\ref{def:standard-a1-compactification},
\[
 R^pg_*R^qj_*F=0\quad(p>0,\ q>0),
 \qquad
 R^pg_*j_*F=0\quad(p>2).
\]
There are canonical isomorphisms
\[
 R^qf_*F\xrightarrow{\ \sim\ }g_*R^qj_*F\qquad(q>2).
\]
\deps{Theorem~\ref{thm:open-higher-support}, the identity $g|_D\simeq
\operatorname{id}_B$, reconstruction of a sheaf supported on $D$ as the
exact direct image from the closed immersion $D\hookrightarrow
\mathbf P^1_K\times_KB$, relative projective-line amplitude, and the finite
Leray filtration.}
\end{theorem}

\begin{theorem}[The second affine-line row]
\label{thm:a1-second-row}
Let $d>1$, let $K$ be a field on which $N$ is invertible, let $F$ be an
arbitrary $N$-torsion sheaf on $\mathbf A^d_K$, and let
$f:\mathbf A^d_K\to B=\mathbf A^{d-1}_K$ be the projection in
Definition~\ref{def:standard-a1-compactification}.
Let $\eta$ be the generic point of $B$.  Then
\[
 (R^2f_*F)_{\bar\eta}
 \cong H^2(\mathbf A^1_{\bar\eta},F|)=0,
\]
and therefore
\[
 \operatorname{Supp}(R^2f_*F)\subseteq E_{d-2}(B/K).
\]
\deps{The geometric-stalk formula, affine-curve torsion cohomological
dimension from Theorem~\ref{thm:absolute-curve-cd}, and the fact that $B$ is
integral of dimension $d-1$.}
\end{theorem}

\begin{theorem}[Support of higher affine-line direct images]
\label{thm:a1-higher-support}
Let $d>1$, let $K$ be a field on which $N$ is invertible, and put
$B=\mathbf A^{d-1}_K$.  Assume
$\operatorname{AffCD}_N(L;c',e)$ for every field $L$, every $c'\geq0$, and
every $e<d$.  For every $N$-torsion sheaf $F$ on $\mathbf A^d_K$, the
projection $f:\mathbf A^d_K\to B$ satisfies
\[
 \operatorname{Supp}(R^qf_*F)\subseteq E_{d-q}(B/K)\quad(q\geq2),
 \qquad
 R^qf_*F=0\quad(q>d).
\]
\deps{Theorems~\ref{thm:open-higher-support},
\ref{thm:a1-relative-rows}, and \ref{thm:a1-second-row}.}
\end{theorem}

\begin{proposition}[Affine-space induction]
\label{prop:affine-space-cd}
Under the hypotheses of Theorem~\ref{thm:a1-higher-support}, if
$\operatorname{FieldCD}_N(K)\leq c$, then every $N$-torsion sheaf $F$ on
$\mathbf A^d_K$ satisfies
\[
 H^r(\mathbf A^d_K,F)=0\qquad(r>d+c).
\]
\deps{The higher-direct-image support theorem, bounded-support affine
cohomology, the lower-dimensional strong affine assertions, and the finite
Leray filtration.}
\end{proposition}

\begin{theorem}[Strong affine torsion cohomological dimension]
\label{thm:strong-affine-cd}
For every field $K$ and all $c,d\geq0$,
\[
 \operatorname{FieldCD}_N(K)\leq c
 \quad\Longrightarrow\quad
 \operatorname{AffCD}_N(K;c,d).
\]
\deps{The dimension-zero and dimension-one cases, Noether normalization,
finite-morphism exactness, the affine-space induction, and simultaneous
induction over all fields and smaller dimensions.}
\end{theorem}

\begin{corollary}[Affine Artin vanishing]
\label{thm:affine-artin}
Let $k$ be separably closed, let $U$ be affine of finite type over $k$, and
let $F$ be an arbitrary $N$-torsion sheaf on $U_{\mathrm{et}}$.  Then
\[
 H^q(U,F)=0\qquad(q>\dim U).
\]
No finiteness assertion is included in this corollary.
\deps{A separably closed field has torsion cohomological dimension zero and
Theorem~\ref{thm:strong-affine-cd}.}
\end{corollary}

\section{The geometric constructibility chain}

For proper $f$, ambient proper normalization identifies $Rf_*$ with $Rf_!$.
The statements below first establish constructibility and boundedness for
the already defined ambient exceptional direct image.

\begin{theorem}[Quasi-finite seed]
\label{thm:qf-seed}
\textbf{Hypotheses.} Let $f:X\to Y$ be separated, locally quasi-finite, and
of finite presentation.

\textbf{Source and target.}
\[
 f_!: \Cons_{\mathrm{et}}(X,\Lambda)\longrightarrow
       \Cons_{\mathrm{et}}(Y,\Lambda).
\]

\textbf{Conclusion.} The functor $f_!$ is exact and preserves constructible
finite sheaves.  Its derived functor is represented termwise and is
$t$-exact on $\Dbc$.
\deps{Zariski's Main Theorem in the form $f=pj$ with $j$ open and $p$ finite,
exact constructible open extension, and exact constructible finite direct
image from Corollary~\ref{cor:finite-exact}.}
\end{theorem}

\begin{theorem}[Compactly supported finiteness for curves]
\label{thm:curve-finite}
\textbf{Hypotheses.} Let $k$ be separably closed, let $C$ be a separated
finite-type $k$-scheme of dimension at most one, and let $F$ be a
constructible finite $\Lambda$-sheaf on $C_{\mathrm{et}}$.

\textbf{Source and target.}
\[
 H_c^q(C,F)=\mathcal H^q\bigl(Ra_!F\bigr),
 \qquad a:C\to\Spec k.
\]

\textbf{Conclusion.} Every $H_c^q(C,F)$ is a finite $\Lambda$-module and
all but finitely many of them vanish.  If $C$ is affine, then
$H^q(C,F)=0$ for $q>1$ and every $H^q(C,F)$ is finite.
\deps{Normalization of a reduced curve, open--closed localization, the
quasi-finite seed, finite-etale descent of lisse sheaves, Kummer theory on a
smooth compactification, and the cohomological-dimension-one theorem for a
strict geometric trait.}
\end{theorem}

\begin{theorem}[Prime-coefficient proper smooth curves]
\label{thm:prime-smooth-proper-curve}
Let $r$ be a prime dividing $N$, and let
$v:\overline C\to S$ be proper and smooth with geometrically connected
fibres of dimension at most one.  Then
\[
 R^qv_*\mathbf F_r
\]
is finite locally constant for $0\leq q\leq2$, is zero for $q>2$, and its
formation commutes with every base change on $S$.
\deps{Smooth local acyclicity for torsion prime to the residue
characteristics, bounded-below proper base change, finite cohomology of a
smooth proper geometric curve, Theorem~\ref{thm:absolute-curve-cd}, and the
specialization criterion for a finite locally constant sheaf.}
\end{theorem}

\begin{theorem}[Proper smooth curves with constant finite coefficients]
\label{thm:smooth-proper-curve}
Let $v:\overline C\to S$ be proper and smooth with geometrically connected
fibres of dimension at most one.  For every finite $\Lambda$-module $M$,
\[
 R^qv_*\underline M
\]
is constructible for every $q$ and is zero for $q>2$.  These objects satisfy
the proper-base-change comparison after every Cartesian base change.
\deps{Theorem~\ref{thm:prime-smooth-proper-curve}, primary decomposition of
a finite module, a finite composition series whose simple factors are
finite-dimensional vector spaces over prime fields, long exact higher direct
images, and the constructible Serre property.}
\end{theorem}

\begin{theorem}[Geometric improvement in relative dimension one]
\label{thm:stratified-curve-compactification}
Let $u:C\to S$ be separated and of finite presentation, let $\eta$ be a
generic point of an irreducible component of $S$, and suppose
$\dim C_\eta\leq1$.  After replacing $S$ by an open neighbourhood $U$ of
$\eta$, there are data
\[
 V\longrightarrow U,
 \qquad
 \coprod_iY_i\xrightarrow{\nu}C_V,
 \qquad
 Y_i\xhookrightarrow{j_i}\overline Y_i\longrightarrow T_i\longrightarrow V
\]
with the following properties:
\begin{enumerate}
\item $V\to U$ is finite, surjective, and universally injective;
\item $\nu$ is finite and surjective, and over an open $W\subset C_V$ dense
      in every fibre, $\nu^{-1}(W)\to W$ is a thickening; the reduced
      complement $C_V\setminus W\to V$ is quasi-finite;
\item $T_i\to V$ is finite etale and $Y_i\to T_i$ is smooth and surjective;
\item $\overline Y_i\to T_i$ is proper and smooth with geometrically
      connected fibres of dimension at most one;
\item $Y_i\hookrightarrow\overline Y_i$ is open and
      $\overline Y_i\setminus j_i(\nu^{-1}(W)\cap Y_i)$ is quasi-finite over
      $V$ after shrinking $U$ once more.
\end{enumerate}
\deps{Normalization of the generic curve, a finite purely inseparable
extension of its ground field, spreading out, universal-homeomorphism
invariance, and openness and generic fibre-density for smooth morphisms.}
\end{theorem}

\begin{theorem}[Finite-surjective detection and thickening invariance]
\label{thm:finite-surjective-detection}
Let $a:V\to U$ be finite and surjective.  A sheaf or bounded complex on
$U_{\mathrm{et}}$ is constructible if its pullback to $V$ is constructible.
If $r:Z_{\mathrm{red}}\to Z$ is a thickening, then $r^{-1}$ is an equivalence
of etale topoi, compatible with extension by zero and compactly supported
direct image.  Both assertions are stable under arbitrary base change.
\deps{Finite-surjective descent of constructible sheaves, topological
invariance of the etale site, and compact-support base change.}
\end{theorem}

\begin{proposition}[Generic-neighbourhood reduction for a constant curve]
\label{prop:constant-curve-generic}
In the situation of
Theorem~\ref{thm:stratified-curve-compactification}, let $M$ be a finite
$\Lambda$-module.  On the open neighbourhood $U$ furnished there,
\[
 R(u_U)_!\underline M\in D_c^{[0,2]}(U_{\mathrm{et}},\Lambda).
\]
After pullback to $V$, this complex has a finite filtration by open--closed
triangles whose nonzero terms are quasi-finite direct images and finite direct
images of the complexes $R(\overline Y_i\to T_i)_*\underline M$.
\deps{Theorems~\ref{thm:stratified-curve-compactification},
\ref{thm:finite-surjective-detection}, \ref{thm:smooth-proper-curve}, and
\ref{thm:qf-seed}, together with compact-support composition and the
constructible Serre property.}
\end{proposition}

\begin{theorem}[Constant sheaf on an affine smooth relative curve]
\label{thm:relative-curve}
Let $S$ be noetherian, affine, and finite-dimensional, let
$u:C\to S$ be separated, affine, and smooth of relative dimension one, and
let $M$ be a finite $\Lambda$-module.  Then
\[
 Ru_!\underline M\in D_c^{[0,2]}(S_{\mathrm{et}},\Lambda).
\]
The assertion is compatible with the compact-support base-change
isomorphism.
\deps{Proposition~\ref{prop:constant-curve-generic}, ambient compact-support
base change, the finite set of generic points and the union of their
neighbourhoods, open--closed localization on the base, and noetherian
induction.}
\end{theorem}

\begin{corollary}[Constant sheaf on a separated smooth relative curve]
\label{cor:constant-separated-curve}
Let $S$ be noetherian, affine, and finite-dimensional, let $u:C\to S$ be
separated, smooth, of finite presentation, and of relative dimension one,
and let $M$ be a finite $\Lambda$-module.  Then
\[
 Ru_!\underline M\in D_c^b(S_{\mathrm{et}},\Lambda).
\]
\deps{Theorem~\ref{thm:relative-curve}, finite affine coverings, affineness
of intersections under separatedness, exactness of the finite augmented
Cech resolution on geometric stalks, compact-support composition, and the
constructible Serre property.}
\end{corollary}

\begin{theorem}[Prime-primary finite-monodromy splitting]
\label{thm:curve-lisse-descent}
Let $\ell$ be a prime and $n\geq1$.  Let $U$ be connected and locally
noetherian, assume $\ell^n\Lambda=0$, and let $L$ be a finite locally constant
$\Lambda$-sheaf on $U$.  There is a
finite etale map $a:U'\to U$ of degree prime to $\ell$ such that $a^*L$ has a
finite filtration with constant finite successive quotients.  The unit and
trace maps
\[
 L\xrightarrow{\eta_a}a_*a^*L
 \xrightarrow{\operatorname{Tr}_a}L
\]
have composite $\deg(a)\operatorname{id}_L$.  The pair
$(\eta_a,\deg(a)^{-1}\operatorname{Tr}_a)$ exhibits $L$ as a direct summand
of $a_*a^*L$.
\deps{The finite-etale Galois category of a connected locally noetherian
scheme, finite monodromy of a finite locally constant sheaf, an $\ell$-Sylow
subgroup of its monodromy group, the invariant filtration for a finite
$\ell$-group, and the finite-etale trace identity.}
\end{theorem}

\begin{theorem}[Lisse sheaves on open relative affine lines]
\label{thm:lisse-open-a1}
Let $S$ be noetherian, affine, and finite-dimensional, let
$U\subset\mathbf A^1_S$ be open, let $f:U\to S$, and let $L$ be a finite
locally constant $\Lambda$-sheaf.  Then
\[
 Rf_!L\in D_c^b(S_{\mathrm{et}},\Lambda).
\]
The conclusion holds separately on the primary coefficient factors.
\deps{Primary decomposition of $\Lambda$, finiteness of the connected
components of a noetherian scheme, Theorem~\ref{thm:curve-lisse-descent},
Corollary~\ref{cor:constant-separated-curve}, compact-support composition,
and stability of constructibility under finite filtrations and direct
summands.}
\end{theorem}

\begin{proposition}[Affine-line open--closed devissage]
\label{prop:noeth-devissage}
Let $S$ be noetherian, affine, and finite-dimensional, let
$p:\mathbf A^1_S\to S$, and let $F$ be a constructible finite sheaf.  There
are a dense open $V\subset S$, a dense open
$U\subset\mathbf A^1_V$, and a reduced closed complement
$i:Z\hookrightarrow\mathbf A^1_V$ such that
\begin{enumerate}
\item $F|_U$ is finite locally constant;
\item after shrinking $V$, the map $Z\to V$ is finite;
\item the localization triangle
\[
 R(U\to V)_!F|_U\longrightarrow
 R(\mathbf A^1_V\to V)_!F|_{\mathbf A^1_V}\longrightarrow
 R(Z\to V)_!i^*F\longrightarrow
\]
has first and third terms in $D_c^b(V_{\mathrm{et}},\Lambda)$;
\item on $S\setminus V$, compact-support base change identifies the remaining
term with the same assertion over a base of smaller dimension.
\end{enumerate}
\deps{A dense lisse open for a constructible sheaf, generic finiteness of
$Z\to S$, Theorems~\ref{thm:lisse-open-a1} and \ref{thm:qf-seed},
compact-support base change, localization, and noetherian induction on
$\dim S$.}
\end{proposition}

\begin{theorem}[Affine-line constructibility]
\label{thm:a1-constructible}
\textbf{Hypotheses.} Let $S$ be noetherian and finite-dimensional, let
$p:\mathbf A^1_S\to S$, and let $F$ be a constructible finite
$\Lambda$-sheaf on $\mathbf A^1_S$.

\textbf{Conclusion.} Every $R^qp_!F$ is constructible and
$R^qp_!F=0$ outside a finite interval independent of points of $S$.
\deps{Proposition~\ref{prop:noeth-devissage},

\end{theorem}

\begin{corollary}[Affine-space constructibility]

\deps{Affine-line constructibility, compact-support composition, the

property.}
\end{corollary}

\begin{lemma}[Affine factorization over an affine target]

 g=\pi_ri,
 \qquad

for every constructible finite sheaf $G$ on $U$.
\deps{A finite set of $A$-algebra generators of $B$, the corresponding

\end{lemma}

\begin{theorem}[Affine finite-presentation constructibility]

\deps{Finite affine covering of a noetherian scheme, finite presentation of
the local algebras, Lemma~\ref{lem:affine-factorization}, ambient
compact-support base change, affine-space constructibility, target-locality
of constructibility, and the maximum of finitely many local amplitude
bounds.}
\end{theorem}

\begin{lemma}[Finite extension-by-zero Cech resolution]
\label{lem:finite-cech-resolution}
Let $p:Z\to T$ be separated and qcqs with $T$ affine, let
$Z=\bigcup_{\nu=1}^mU_\nu$ be a finite affine open cover, and let $G$ be a
sheaf on $Z_{\mathrm{et}}$.  For nonempty
$I\subseteq\{1,\ldots,m\}$, put
$U_I=\bigcap_{\nu\in I}U_\nu$ and let $j_I:U_I\hookrightarrow Z$.  The
augmented complex
\[
 0\to \bigoplus_{|I|=m}j_{I,!}G|_{U_I}\to\cdots\to
 \bigoplus_{|I|=2}j_{I,!}G|_{U_I}\to
 \bigoplus_{|I|=1}j_{I,!}G|_{U_I}\to G\to0
\]
is exact.  Its differentials are the alternating sums of the canonical maps
$j_{I,!}G|_{U_I}\to j_{J,!}G|_{U_J}$ for $J\subset I$ and
$|J|=|I|-1$; its augmentation is the sum of the adjunction maps.
Every $U_I$ is affine.
\deps{The closed relative diagonal of $p$, affineness of the target,
affineness of finite intersections of affine opens under relative
separatedness, and the stalkwise contraction of the augmented finite simplex.}
\end{lemma}

\textbf{Conclusion.} Every $R^qp_*G$ is constructible.
\deps{Target-locality of constructibility, finite affine localization on
$T$, Lemma~\ref{lem:finite-cech-resolution}, proper normalization
$Rp_*\simeq Rp_!$, compact-support composition for each $pj_I$,
affine-morphism constructibility, the finite Cech spectral sequence, and the
constructible Serre property.}
\end{theorem}

\section{Sharp cohomological bounds for proper morphisms}

\begin{definition}[Constructible cohomological dimension]
Let $f:X\to Y$ and let $c\geq0$.  Write
\[
 \operatorname{ccd}_\Lambda(f)\leq c
\]
if $R^qf_*F=0$ for every constructible finite $\Lambda$-sheaf $F$ and every
$q>c$.  The bound is \emph{universal} if the same $c$ works after every base
change $Y'\to Y$ for which the pulled-back schemes remain in the stated
geometric class.
\deps{The higher direct-image sheaves of the ambient $Rf_*$.}
\end{definition}

\begin{definition}[Strong geometric proper cohomological dimension]
\label{def:proper-cd}
For $d\geq0$, let $\operatorname{ProperCD}_N(d)$ denote the assertion that,
for every separably closed field $k$ on which $N$ is invertible, every proper
finite-type $k$-scheme $X$ with $\dim X\leq d$, and every etale sheaf $F$ of
$\Lambda$-modules on $X$, one has
\[
 H^r(X,F)=0\qquad(r>2d).
\]
No constructibility hypothesis is imposed on $F$.
\deps{Proper schemes over a separably closed field, arbitrary
$N$-torsion coefficient sheaves, and ordinary etale cohomology.}
\end{definition}

\begin{theorem}[Invariance under reduction]
\label{thm:reduction-etale}
For the nilpotent closed immersion
$r:X_{\mathrm{red}}\hookrightarrow X$, pullback and pushforward are inverse
exact equivalences of small etale topoi.  For every $q$ and every sheaf of
$\Lambda$-modules $F$, the induced cohomology map is
\[
 H^q(X,F)\xrightarrow{\ \sim\ }
 H^q(X_{\mathrm{red}},r^*F)
\].
\deps{Topological invariance of the etale site under a universal
homeomorphism and exactness of an equivalence of topoi.}
\end{theorem}

\begin{definition}[Normalization descent datum]
\label{def:normalization-descent}
Let $k$ be a field on which $N$ is invertible, let $X$ be reduced, proper,
and of finite type over $k$, and let $F$ be an etale sheaf of
$\Lambda$-modules on $X$.  A normalization descent datum for $(X,F)$ consists
of a finite surjection
\[
 \nu:X^\nu\longrightarrow X
\]
from the disjoint union of the normalizations of the irreducible components,
a dense open $U\subseteq X$ over which $\nu$ is an isomorphism, the reduced
closed complement $i:Z\hookrightarrow X$, and the exact sequence
\[
 0\longrightarrow F\longrightarrow\nu_*\nu^*F
 \longrightarrow i_*C_F\longrightarrow0.
\]
Here $C_F$ is the unique sheaf of $\Lambda$-modules on $Z$ whose direct image
is the cokernel of $F\to\nu_*\nu^*F$.
The datum includes $\dim Z<\dim X$, exactness of $\nu_*$, the identification
$H^q(X,\nu_*\nu^*F)=H^q(X^\nu,\nu^*F)$, and the resulting long exact
cohomology sequence.
\deps{Finite normalization over a field, the adjunction unit for a finite
surjection, support of its cokernel away from the isomorphism locus, and
finite-morphism exactness.}
\end{definition}

\begin{proposition}[Existence of normalization descent]
\label{prop:normalization-descent}
Every triple $(k,X,F)$ satisfying
Definition~\ref{def:normalization-descent} admits a normalization descent
datum.  If $\dim X=d$, its closed term is proper of dimension at most $d-1$.
\deps{Finiteness of normalization for schemes of finite type over a field,
lying over, the dense normal locus, geometric-stalk injectivity of
$F\to\nu_*\nu^*F$, and closed-immersion reconstruction.}
\end{proposition}

\begin{proposition}[High-degree reduction through normalization]
\label{prop:normalization-high-degree}
Let $k$ be separably closed, let $X$ be reduced, proper, and
$d$-dimensional over $k$, and let $F$ be an etale sheaf of $\Lambda$-modules
on $X$.  Suppose $\operatorname{ProperCD}_N(e)$ holds for every $e<d$.  For
a normalization descent datum for $(X,F)$, the closed term satisfies
\[
 H^r(Z,C_F)=0\qquad(r>2d-2).
\]
The normalization sequence induces an isomorphism
\[
 H^r(X,F)\xrightarrow{\ \sim\ }
 H^r(X^\nu,\nu^*F)\qquad(r>2d-1).
\]
\deps{Proposition~\ref{prop:normalization-descent}, closed-immersion
cohomology, the lower-dimensional proper-cd assertions, and the long exact
cohomology sequence of the normalization datum.}
\end{proposition}

\begin{definition}[Rational-function graph datum]
\label{def:graph-datum}
Let $k$ be a separably closed field on which $N$ is invertible, and let $X$
be normal, integral, proper, and $d$-dimensional over $k$, with $d>1$.
Choose a nonconstant $t\in k(X)$.  Let $X_t$ be the reduced closure
of its graph in $X\times_k\mathbf P^1_k$, and denote the projections by
\[
 X\xleftarrow{\ b\ }X_t\xrightarrow{\ h\ }\mathbf P^1_k.
\]
A rational-function graph datum includes the following properties:
\begin{enumerate}
\item $b$ and $h$ are proper and $b$ is surjective;
\item $b$ is an isomorphism over an open containing all codimension-one
      points of $X$; its complement $i:Z\hookrightarrow X$ satisfies
      $\dim Z\leq d-2$;
\item every fibre of $b$ has dimension at most one and every fibre of $h$
      has dimension at most $d-1$;
\item for every etale sheaf $F$ of $\Lambda$-modules on $X$, the adjunction
      unit and its cone are fixed:
      \[
       F\longrightarrow Rb_*b^*F\longrightarrow Q_{t,F}\longrightarrow F[1].
      \]
\end{enumerate}
\deps{A nonconstant rational function and the closure of its graph in a
proper product.}
\end{definition}

\begin{lemma}[Codimension-one extension of a rational function]
\label{lem:codimension-one-rational-map}
Let $k$ be a field, let $X$ be normal and integral over $k$, and let
$t\in k(X)$.  At every codimension-one
point $x\in X$, the rational map $[1:t]:X\dashrightarrow\mathbf P^1$ extends
to $\Spec\mathcal O_{X,x}$.  Equivalently, either $t$ or $t^{-1}$ belongs to
$\mathcal O_{X,x}$.
\deps{A noetherian normal local domain of dimension one is a discrete
valuation ring and $\mathbf P^1$ satisfies the valuative criterion for
properness.}
\end{lemma}

\begin{proposition}[Existence and geometry of the graph datum]
\label{prop:graph-datum}
Let $k$ be a separably closed field on which $N$ is invertible.  Every normal
integral proper $k$-scheme $X$ of dimension $d>1$ admits a rational-function
graph datum.  Its isomorphism locus contains every
codimension-one point.  Two choices of rational function have a common
refinement given by the closure of their joint graph, and the induced
comparison maps satisfy the cocycle equation after a common three-function
refinement.
\deps{Noether normalization for the existence of a nonconstant function,
Lemma~\ref{lem:codimension-one-rational-map}, properness of closed subschemes
of $X\times\mathbf P^1$, fibre-dimension estimates, and joint graph
closures.}
\end{proposition}

\begin{theorem}[Proper relative curves have amplitude two]
\label{thm:proper-relative-curve-cd}
Let $v:Y\to S$ be proper and suppose that every geometric fibre has
dimension at most one.  For every sheaf $F$ of $\Lambda$-modules on $Y$,
\[
 R^qv_*F=0\qquad(q>2).
\]
\deps{The bounded-below finite-torsion proper-base-change theorem, the
geometric-stalk formula, the cohomological-dimension-two theorem for an
arbitrary torsion sheaf on a proper curve over a separably closed field, and
geometric-stalk conservativity.}
\end{theorem}

\begin{theorem}[Range and support of the graph cone]
\label{thm:graph-cone-support}
Let $(X_t,b,h,Z)$ be a rational-function graph datum for $X/k$, and let $F$
be an arbitrary etale sheaf of $\Lambda$-modules on $X$.  Then
\[
 \mathcal H^q(Q_{t,F})=0\quad(q\notin\{0,1,2\}),
 \qquad
 \operatorname{Supp}\mathcal H^q(Q_{t,F})\subseteq Z.
\]
In degrees $0,1,2$ the cohomology sheaves are respectively the cokernel of
$F\to b_*b^*F$ and the sheaves $R^qb_*b^*F$ for $q=1,2$.
\deps{Proper relative-curve amplitude for $b$, injectivity of the adjunction
unit for a proper surjection, the fact that $b$ is an isomorphism off $Z$,
and the long exact cohomology sequence of the graph triangle.}
\end{theorem}

\begin{theorem}[Hypercohomology bound for the graph cone]
\label{thm:graph-cone-hypercohomology}
Let $(X_t,b,h,Z)$ be a $d$-dimensional rational-function graph datum for
$X/k$, let $F$ be an arbitrary etale sheaf of $\Lambda$-modules on $X$, and
assume $\operatorname{ProperCD}_N(e)$ for every $e<d$.  Then
\[
 H^r(X,Q_{t,F})=0\qquad(r>2d-2).
\]
\deps{The three-row truncation spectral sequence for $Q_{t,F}$, closed
support on the proper scheme $Z$ of dimension at most $d-2$, and the
lower-dimensional proper-cd assertions.}
\end{theorem}

\begin{proposition}[High-degree comparison with the graph closure]
\label{prop:graph-high-degree-comparison}
Let $(X_t,b,h,Z)$, $F$, and the lower-dimensional proper-cd hypotheses be as
in Theorem~\ref{thm:graph-cone-hypercohomology}.  The unit in the graph
triangle induces an isomorphism
\[
 H^r(X,F)\xrightarrow{\ \sim\ }
 H^r(X_t,b^*F)\qquad(r>2d).
\]
\deps{Theorem~\ref{thm:graph-cone-hypercohomology} and the long exact
cohomology sequence of the graph triangle.}
\end{proposition}

\begin{theorem}[Higher direct images of the graph projection]
\label{thm:graph-projection-bound}
Let $(X_t,b,h,Z)$ be a $d$-dimensional rational-function graph datum for
$X/k$, let $F$ be an arbitrary etale sheaf of $\Lambda$-modules on $X$, and
assume $\operatorname{ProperCD}_N(e)$ for every $e<d$.  Then
\[
 R^qh_*b^*F=0\qquad(q>2d-2).
\]
For every geometric point $\bar y\to\mathbf P^1_k$, the stalk in degree $q$
is
\[
 H^q((X_t)_{\bar y},b^*F|_{(X_t)_{\bar y}}),
\]
and the fibre is proper of dimension at most $d-1$.
\deps{Bounded-below proper base change, the fibre-dimension statement in the
graph datum, $\operatorname{ProperCD}_N(d-1)$, and geometric-stalk
conservativity.}
\end{theorem}

\begin{theorem}[Leray bound over the projective line]
\label{thm:graph-p1-leray}
Let $(X_t,b,h,Z)$, $F$, and the lower-dimensional proper-cd hypotheses be as
in Theorem~\ref{thm:graph-projection-bound}.  Then
\[
 H^r(X_t,b^*F)=0\qquad(r>2d).
\]
\deps{The graph-projection bound, cohomological dimension two for
$\mathbf P^1$ with arbitrary $N$-torsion coefficients, and finite
convergence of the Leray spectral sequence.}
\end{theorem}

\begin{proposition}[Strong proper-cd induction step]
\label{prop:proper-cd-step}
Let $d>1$.  If $\operatorname{ProperCD}_N(e)$ holds for every $e<d$, then
$\operatorname{ProperCD}_N(d)$ holds.
\deps{Invariance under reduction, normalization descent and its
high-degree reduction of Proposition~\ref{prop:normalization-high-degree},
existence of a rational-function graph datum,
Proposition~\ref{prop:graph-high-degree-comparison}, and the projective-line
Leray bound.}
\end{proposition}

\begin{theorem}[Proper-cd base cases]
\label{thm:proper-cd-base}
The assertions $\operatorname{ProperCD}_N(0)$ and
$\operatorname{ProperCD}_N(1)$ hold.
\deps{Field torsion cohomological dimension zero, finite direct-image
exactness, normalization of curves, strict-trait cohomological dimension
one, and the proper-curve calculation.}
\end{theorem}

\begin{theorem}[Strong geometric proper cohomological dimension]
\label{thm:proper-cd}
For every $d\geq0$, the assertion $\operatorname{ProperCD}_N(d)$ holds.
The resulting zero identifications are compatible with extensions of the
separably closed base field and with common refinements of the chosen graph
data.
\deps{The proper-cd base cases, the strong proper-cd induction step, strong
induction on $d$, and the common-refinement cocycle for graph closures.}
\end{theorem}

\begin{corollary}[Relative proper cohomological-dimension bound]
\label{cor:relative-proper-cd}
Let $p:Z\to T$ be proper, suppose every geometric fibre has dimension at
most $d$, and let $F$ be an arbitrary sheaf of $\Lambda$-modules on $Z$.
Then
\[
 R^qp_*F=0\qquad(q>2d).
\]
This is a statement about arbitrary $N$-torsion sheaves; no constructibility
or finiteness hypothesis is imposed.
\deps{Theorem~\ref{thm:ambient-proper-bc}, its arbitrary-sheaf geometric
stalk formula, Theorem~\ref{thm:proper-cd}, and conservativity of geometric
stalks.}
\end{corollary}

\begin{theorem}[Finite cohomology on a proper geometric fibre]
\label{thm:proper-fibre-finite}
Let $k$ be separably closed, let $X$ be proper of finite type over $k$, and
let $F$ be a constructible finite $\Lambda$-sheaf on $X$.  Then every
$H^q(X,F)$ is a finite $\Lambda$-module.
\deps{The proper constructibility theorem
\ref{thm:cech-proper} applied to $X\to\Spec k$ and the description of a
constructible finite sheaf on a geometric point as a finite module.}
\end{theorem}

\begin{corollary}[Proper geometric-fibre finiteness and dimension bound]
\label{thm:fibre-artin-bound}
Let $k$ be separably closed, let $X$ be proper of finite type over $k$ with
$\dim X\leq d$, and let $F$ be a constructible finite $\Lambda$-sheaf.  Then
every $H^q(X,F)$ is finite and
\[
 H^q(X,F)=0\qquad(q>2d).
\]
\deps{Finite cohomology on a proper geometric fibre and

\end{corollary}

\begin{corollary}[Proper geometric-stalk bound]
\label{cor:proper-stalk-bound}
Let $p:Z\to T$ be proper and of finite presentation, with geometric fibres
of dimension at most $d$, and let $F$ be a constructible finite
$\Lambda$-sheaf on $Z$.  For every geometric point $\bar t\to T$,
\[
 (R^qp_*F)_{\bar t}\xrightarrow{\ \sim\ }
 H^q(Z_{\bar t},F|_{Z_{\bar t}}).
\]
These stalks are finite, and $R^qp_*F=0$ for $q>2d$.
\deps{The bounded-below proper-base-change theorem, the proper
geometric-fibre theorem, exact and conservative forgetting from
$\Lambda$-module sheaves to abelian sheaves, and conservativity of geometric
stalks.}
\end{corollary}

\begin{definition}[The proper Artin--finiteness predicate]
Let $p:Z\to T$ be proper and of finite presentation, and suppose every

\]
means that for every constructible finite sheaf $F$ on $Z$,

 \qquad
 R^qp_*F\text{ is constructible}\quad(0\leq q\leq2d).

\deps{Constructible higher direct images and a uniform geometric-fibre

\end{definition}

\begin{theorem}[Proper Artin--finiteness]

\begin{theorem}[Finite cohomological dimension for compact support]
\label{thm:finite-cd-shriek}
Let $f:X\to Y$ be separated and of finite presentation between qcqs
noetherian finite-dimensional schemes.  Choose a compactification
\[
 X\xhookrightarrow{j}\overline X\xrightarrow{p}Y
\]
and an integer $d_p$ bounding the dimensions of the geometric fibres of
$p$.  Then
\[
 R^qf_!F=0\qquad(q>2d_p)
\]
for every sheaf $F$ of $\Lambda$-modules.  The integer $c(f)=2d_p$ is a finite
cohomological bound for this compactification model.  If bounds $c(f)$ and
$c(g)$ have been chosen for two composable maps, then $c(f)+c(g)$ is a valid
bound for $gf$.
\deps{The compactification formula $Rf_!=Rp_*j_!$, exactness of $j_!$,
Corollary~\ref{cor:relative-proper-cd}, ambient composition, and the
Grothendieck spectral sequence.}
\end{theorem}

\begin{theorem}[Constructible preservation for compact support]
\label{thm:shriek-preserves}
Let $f:X\to Y$ be separated and of finite presentation between noetherian
finite-dimensional schemes.  The ambient functor restricts to an exact
triangulated functor
\[
 Rf_!:\Dbc(X_{\mathrm{et}},\Lambda)\longrightarrow
       \Dbc(Y_{\mathrm{et}},\Lambda).
\]
If $K\in D_c^{[a,b]}$ and $c(f)$ is a chosen cohomological bound, then
\[
 Rf_!K\in D_c^{[a,b+c(f)]}.
\]
\deps{The choice-independent ambient functor of
Theorem~\ref{thm:ambient-shriek}, its composition law from
Theorem~\ref{thm:ambient-shriek-composition}, the quasi-finite seed,
affine-line constructibility, affine-space and affine-morphism assembly,
finite affine Cech assembly, Theorem~\ref{thm:finite-cd-shriek}, finite
truncation triangles, and Proposition~\ref{prop:hypercohomology}.}
\end{theorem}

\begin{corollary}[Proper direct image from compact support]
\label{cor:proper-star-shriek}
Let $f:X\to Y$ be proper and of finite presentation between noetherian
finite-dimensional $\Lambda$-admissible schemes.  The identity compactification
gives a canonical isomorphism of ambient functors
\[
 Rf_!\xrightarrow{\ \sim\ }Rf_*.
\]
Its bounded-constructible restriction is
\[
 Rf_*:\Dbc(X_{\mathrm{et}},\Lambda)\longrightarrow
       \Dbc(Y_{\mathrm{et}},\Lambda)
\].
\deps{The ambient proper normalization of

restriction of Theorem~\ref{thm:shriek-preserves}.}
\end{corollary}

\begin{proposition}[Intrinsic restriction criterion]
\label{prop:restriction}
Let $f:X\to Y$ be a morphism between noetherian finite-dimensional
$\Lambda$-admissible schemes.  Assume that $f$ satisfies both conditions:
\begin{enumerate}
\item $R^qf_*F$ is constructible for every constructible finite sheaf $F$
      and every $q\geq0$;
\item $\operatorname{ccd}_\Lambda(f)\leq c$ for some finite $c$.
\end{enumerate}
Then the ambient functor restricts uniquely to
\[
 Rf_*:\Dbc(X_{\mathrm{et}},\Lambda)\longrightarrow
       \Dbc(Y_{\mathrm{et}},\Lambda),
\]
and maps $D_c^{[a,b]}$ into $D_c^{[a,b+c]}$.
\deps{The bounded hypercohomology spectral sequence and its finite
constructible abutment filtration.}
\end{proposition}

\begin{theorem}[Proper bounded-constructible output]
\label{thm:proper-output}

\textbf{Source and target.}

\textbf{Conclusion.} For every $K\in D_c^{[a,b]}$,

\deps{Proper Artin--finiteness, the intrinsic restriction criterion, and the

\end{theorem}

\begin{theorem}[Tensor comparison in the finite-Tor range]
\label{thm:tensor-comparison}
\textbf{Hypotheses.} Let $p:Z\to T$ be as in

$K\in\Dctf(Z_{\mathrm{et}},\Lambda)$, and let
$M\in\mathcal A_{\mathrm{et}}(T,\Lambda)$ be placed in degree zero.

\textbf{Map.} The adjunction and the tensor product define

 \operatorname{PF}_{p,K,M}:
 (Rp_*K)\otimes^L_{\Lambda_T}M
 \longrightarrow

\]

\textbf{Conclusion.} The map $\operatorname{PF}_{p,K,M}$ is an isomorphism,
naturally in $K$ and $M$, and is compatible with composition of proper
morphisms.
\deps{Finite cohomological dimension, a bounded $K$-flat representative of
$K$, the underived projection morphism, truncation in a finite rectangle,
and derived composition.}
\end{theorem}

If $K$ has Tor-amplitude $[a,b]$, then $Rp_*K$ has Tor-amplitude contained

\section{Uniformity in the coefficient level}

\begin{definition}[Derived change of finite coefficients]
\label{def:coefficient-change}
Let $f:X\to Y$, let $\rho:\Lambda\to\Lambda'$ be a homomorphism of finite
commutative rings, and assume that $X$ and $Y$ are admissible for annihilating
integers of both rings.  For $\tau\in\{\mathrm{et},\mathrm{proet}\}$ define
\[
 \rho_X^*K:=K\otimes^L_{\Lambda_X}\Lambda'_X:
 D(X_\tau,\Lambda)\longrightarrow D(X_\tau,\Lambda').
\]
Adjunction and the projection morphism define the canonical
change-of-coefficients map
\[
 \chi_{f,\rho,K}:
 \rho_Y^*Rf_{*,\Lambda}K
 \longrightarrow
 Rf_{*,\Lambda'}\rho_X^*K.
\]
\deps{Derived tensor product, the two ambient direct-image functors, and the
coefficient homomorphism on the constant ring sheaves.}
\end{definition}

\begin{theorem}[Finite coefficient change for proper direct image]
\label{thm:coefficient-change}
Let $p:Z\to T$ satisfy the hypotheses of Theorem~\ref{thm:proper-output} and
let $\rho:\Lambda\to\Lambda'$ satisfy
Definition~\ref{def:coefficient-change} with both coefficient rings
admissible on $Z$ and $T$.  For $K\in\Dctf(Z_{\mathrm{et}},\Lambda)$,
\[
 \chi_{p,\rho,K}:
 \Lambda'_T\otimes^L_{\Lambda_T}Rp_{*,\Lambda}K
 \xrightarrow{\ \sim\ }
 Rp_{*,\Lambda'}
   \bigl(\Lambda'_Z\otimes^L_{\Lambda_Z}K\bigr)
\]
is an isomorphism.  For
$\Lambda\to\Lambda'\to\Lambda''$, the direct comparison equals the
composite of the two comparisons after inserting the associator for derived
tensor product.
\deps{The tensor comparison in the finite-Tor range, scalar restriction, and
associativity of derived tensor product.}
\end{theorem}

\begin{theorem}[Uniform finite-level proper output]
\label{thm:uniform-level}
Let $\Lambda_n=\mathcal O/\varpi^n$, and let $p:Z\to T$ be proper and of
finite presentation between noetherian finite-dimensional schemes on which
$\ell$ is invertible.  Suppose its geometric fibres have dimension at most
$d$.
Let
\[
 K_n\in\Dctf(Z_{\mathrm{et}},\Lambda_n),\qquad n\geq1,
\]
have one common cohomological interval $[a,b]$ and compatible reduction
isomorphisms
\[
 K_n\otimes^L_{\Lambda_n}\Lambda_m
 \xrightarrow{\ \sim\ }K_m
 \qquad(m\leq n),
\]
satisfying the identity and tower equations.

Then the output complexes
\[
 L_n:=Rp_{*,\Lambda_n}K_n
\]
belong to $D_c^{[a,b+2d]}(T_{\mathrm{et}},\Lambda_n)$, have finite
Tor-amplitude, and carry compatible reduction isomorphisms
\[
 L_n\otimes^L_{\Lambda_n}\Lambda_m
 \xrightarrow{\ \sim\ }L_m.
\]
The interval $[a,b+2d]$ and all comparison maps are independent of $n$.
\deps{Proper Artin--finiteness with its coefficient-independent bound,
finite-Tor output, finite coefficient change, and the tower coherence of
derived tensor product.}
\end{theorem}

\section{Transport to the pro-etale site}

\begin{theorem}[Direct-image site comparison]
\label{thm:site-rf}
Let $f:X\to Y$ be quasi-compact and quasi-separated and let
$K\in D^+(X_{\mathrm{et}},\Lambda)$.  The square of topoi gives a canonical
map
\[
 \epsilon_Y^{-1}Rf_{*,\mathrm{et}}K
 \longrightarrow
 Rf_{*,\mathrm{proet}}\epsilon_X^{-1}K.
\]
It is an isomorphism, is natural in $K$, and is compatible with identity
morphisms and, for every composable $g:Y\to W$, with the composition
comparison $c_{g,f}$.
\deps{The section formula for etale-to-pro-etale inverse image, the
$K$-injective construction of both direct images, continuity on pro-etale
objects, and bounded-below truncation.}
\end{theorem}

\begin{corollary}[Pro-etale bounded-constructible output]
\label{cor:proet-output}
Let $p:Z\to T$ be proper and of finite presentation between noetherian
finite-dimensional $\Lambda$-admissible schemes, with geometric fibres of
dimension at most $d$.
Then
\[
 Rp_*:\Dbc(Z_{\mathrm{proet}},\Lambda)\longrightarrow
       \Dbc(T_{\mathrm{proet}},\Lambda),
 \qquad
 Rp_*:\Dctf(Z_{\mathrm{proet}},\Lambda)\longrightarrow
       \Dctf(T_{\mathrm{proet}},\Lambda)
\]
are exact triangulated functors.  They carry $D_c^{[a,b]}$ into
$D_c^{[a,b+2d]}$, commute with finite coefficient change, and preserve
compatible finite-level towers with the same uniform bound.
\deps{Finite etale--pro-etale equivalence, direct-image site comparison,
proper bounded-constructible output, finite-Tor output,
Theorem~\ref{thm:coefficient-change}, compatibility of
$\epsilon^{-1}$ with derived scalar extension, and uniform finite-level
output.}
\end{corollary}

\begin{theorem}[Proper finite-coefficient direct-image theorem]
Let $\tau\in\{\mathrm{et},\mathrm{proet}\}$ and let $p:Z\to T$ be a proper
finite-presentation morphism between noetherian finite-dimensional
$\Lambda$-admissible schemes whose geometric fibres have dimension at most
$d$.  The following data and conclusions are simultaneously available:
\begin{enumerate}
\item the ambient exact triangulated functor
      $Rp_*:D(Z_\tau,\Lambda)\to D(T_\tau,\Lambda)$ defined by
      $K$-injective replacement;
\item for every morphism $q:T\to S$ between noetherian finite-dimensional
      $\Lambda$-admissible schemes, the coherent composition isomorphism
      $R(qp)_*\cong Rq_*Rp_*$;
\item the restrictions to $\Dbc$ and $\Dctf$;
\item the amplitude bound $[a,b]\mapsto[a,b+2d]$;
\item for every homomorphism $\rho:\Lambda\to\Lambda'$ satisfying
      Definition~\ref{def:coefficient-change}, the finite-Tor
      change-of-coefficients isomorphism;
\item for the principal tower $\Lambda_n=\mathcal O/\varpi^n$ and every
      compatible uniformly bounded finite-Tor tower $(K_n)$, the coherent
      uniformly bounded output tower $(Rp_{*,\Lambda_n}K_n)$;
\item etale--pro-etale compatibility.
\end{enumerate}
\deps{Theorems~\ref{thm:composition}, \ref{thm:proper-output},
\ref{thm:tensor-comparison}, Corollary~\ref{cor:ctf-output},
Theorems~\ref{thm:coefficient-change}, \ref{thm:uniform-level}, and
\ref{thm:site-rf}, and Corollary~\ref{cor:proet-output}.}
\end{theorem}
