\begin{center}
{\large\bfseries The $\Qlb$-Coefficient Triangulated 2-Colimit\par}
\medskip
{\normalsize Common-stage morphisms, cones, scalars, retracts, and Weil objects\par}
\end{center}

\begin{abstract}
This note constructs algebraically closed $\ell$-adic coefficients without
using an infinite coefficient field as a primitive derived coefficient
ring.  Finite coefficient categories are assembled by an explicit pair
category: morphisms are represented over a common finite extension, and
equality, addition, composition, shifts, cones, and scalar multiplication
are all defined at a further common stage.  The Karoubi envelope gives the
idempotent-complete ordinary triangulated 2-colimit.  Separate statements
give its universal property, its comparison with a literal algebraic
$\Qlb$-sheaf under exact descent hypotheses, and the parallel strict
2-colimit of bounded Weil complexes.
\end{abstract}

\section{The coherent diagram of finite coefficient fields}

Fix an algebraic closure $\Qlb$ of $\Ql$ and a scheme $X$.  For every
finite intermediate field $E$, the category $\Dbc(X,E)$ is the rational
Karoubi category constructed from $\mathcal O_E$ in the preceding note.

\begin{definition}[The filtered coefficient index]\label{def:finext}
Let $\mathsf{FinExt}_\ell$ be the poset category whose objects are finite
extensions
\[
 \Ql\subset E\subset\Qlb
\]
and whose unique arrow $E\to F$ exists precisely when $E\subset F$.  For
$E,F$, their compositum $EF\subset\Qlb$ is the smallest subfield containing
both.  It is finite over $\Ql$ and receives arrows from $E$ and $F$.
\end{definition}

\begin{proposition}[Filteredness]\label{prop:finext-filtered}
The category $\mathsf{FinExt}_\ell$ is filtered: it is nonempty, every pair
of objects maps to a common object, and every pair of parallel arrows is
equalized.
\depends{Definition~\ref{def:finext}; finiteness of the compositum of two
finite coefficient-field extensions.}
\end{proposition}

\begin{definition}[Finite-stage categories and transitions]
\label{def:finite-stages}
For $E\in\mathsf{FinExt}_\ell$, put
\[
 \mathcal C_E=\Dbc(X,E).
\]
For an inclusion $E\subset F$, let
\[
 T_{F/E}=\operatorname{Ext}_{F/E}:
 \mathcal C_E\longrightarrow\mathcal C_F
\]
be the exact scalar-extension functor.  Retain a specified unit isomorphism
\[
 \iota_E:T_{E/E}\xRightarrow{\sim}\operatorname{id}_{\mathcal C_E}
\]
and, for $E\subset F\subset G$, a specified associator
\[
 a_{G,F,E}:T_{G/F}T_{F/E}\xRightarrow{\sim}T_{G/E}.
\]
The unit diagrams and the pentagon for $E\subset F\subset G\subset H$ are
part of this transition datum.  We use the normalized convention
\[
 T_{E/E}=\operatorname{id}_{\mathcal C_E},\qquad
 \iota_E=\operatorname{id}
\]
literally; the unit components of the associators are the corresponding
identity maps.

Retain the shift comparison of the exact functor $T_{F/E}$,
\[
 \sigma_{F/E,K}:T_{F/E}(K[1])
 \xrightarrow{\sim}(T_{F/E}K)[1].
\]
These comparisons are natural in $K$, agree with the identity when $E=F$,
and commute with every associator $a_{G,F,E}$.
\end{definition}

\begin{theorem}[Finite-stage admissibility and coherence]
\label{thm:finite-stage-coherence}
Assume that $X$ is quasi-compact, quasi-separated, noetherian of finite
Krull dimension, that $\ell$ is invertible on $X$, and that finite
constructible complexes on the selected pro-etale site satisfy the local
perfectness and finite-cohomological-dimension theorems.  Assume moreover
that the compact-monodromy lattice condition of the preceding note is
imposed on the lisse objects under consideration.  Then each
$\mathcal C_E$ is an $E$-linear idempotent-complete ordinary triangulated
category.  Every $T_{F/E}$ is exact and $E$-linear, where the target is
viewed as $E$-linear through $E\subset F$.  The maps $\iota_E$ and
$a_{G,F,E}$ are exact natural isomorphisms satisfying the unit and pentagon
equations.
\depends{The finite-extension preservation and coherence theorems for
integral completion, rational localization, and Karoubi completion.}
\end{theorem}

\section{The explicit pair category}

\begin{definition}[Objects and common stages]\label{def:pair-objects}
An object representative is a pair $(E,K)$ with
$E\in\mathsf{FinExt}_\ell$ and $K\in\mathcal C_E$.  For two coefficient
fields $E,F$, let
\[
 \mathsf I(E,F)=
 \left\{M\in\mathsf{FinExt}_\ell:E\subset M,\ F\subset M\right\},
\]
ordered by inclusion.  This is a filtered poset.
\end{definition}

\begin{definition}[Common-stage morphism representative]
\label{def:morphism-representative}
For objects $(E,K)$ and $(F,L)$, a morphism representative is a pair
$(M,u)$ with $M\in\mathsf I(E,F)$ and
\[
 u:T_{M/E}K\longrightarrow T_{M/F}L
 \quad\text{in }\mathcal C_M.
\]
If $M\subset N$, its transport to $N$ is the morphism
\[
 u_N:=
 a_{N,M,F,L}\circ T_{N/M}(u)\circ a_{N,M,E,K}^{-1}:
 T_{N/E}K\longrightarrow T_{N/F}L,
\]
where $a_{N,M,E,K}$ denotes the component at $K$ of
$a_{N,M,E}$.
\end{definition}

\begin{proposition}[Transport identity and tower equations]
\label{prop:transport-tower}
Under the unit identifications, transport from $M$ to itself is the identity:
\[
 u_M=u.
\]
For $M\subset N\subset P$, direct and iterated transport agree:
\[
 (u_N)_P=u_P:
 T_{P/E}K\longrightarrow T_{P/F}L.
\]
Transport is additive and sends zero and identity morphisms to zero and
identity morphisms after the corresponding unit identifications.
\depends{Definition~\ref{def:morphism-representative}; the unit, naturality,
and pentagon equations in Theorem~\ref{thm:finite-stage-coherence}.}
\end{proposition}

\begin{definition}[Equality of representatives]\label{def:representative-equality}
Representatives $(M,u)$ and $(N,v)$ from $(E,K)$ to $(F,L)$ are
\emph{equivalent}, written $(M,u)\sim(N,v)$, if there exists
$P\in\mathsf I(M,N)$ such that
\[
 u_P=v_P:T_{P/E}K\longrightarrow T_{P/F}L
 \quad\text{in }\mathcal C_P.
\]
\end{definition}

\begin{lemma}[Reflexivity of common-stage equality]
\label{lem:equality-reflexive}
Every representative $(M,u)$ satisfies $(M,u)\sim(M,u)$.
\depends{Proposition~\ref{prop:transport-tower}.}
\end{lemma}

\begin{lemma}[Symmetry of common-stage equality]
\label{lem:equality-symmetric}
If $(M,u)\sim(N,v)$, then $(N,v)\sim(M,u)$.
\depends{Definition~\ref{def:representative-equality}; symmetry of equality
in the finite-stage Hom set.}
\end{lemma}

\begin{lemma}[Transitivity of common-stage equality]
\label{lem:equality-transitive}
If $(M,u)\sim(N,v)$ and $(N,v)\sim(Q,w)$, then $(M,u)\sim(Q,w)$.
If the two equalities are witnessed over $P_1$ and $P_2$, any finite coefficient field
containing their compositum witnesses the resulting equality.
\depends{Propositions~\ref{prop:finext-filtered} and
\ref{prop:transport-tower}; transitivity of equality at a finite stage.}
\end{lemma}

\begin{proposition}[Common-stage equality is an equivalence relation]
\label{prop:equality-equivalence}
The relation $\sim$ is an equivalence relation, and transporting an equality
to a larger finite stage preserves it.
\depends{Lemmas~\ref{lem:equality-reflexive}--\ref{lem:equality-transitive};
Proposition~\ref{prop:transport-tower}.}
\end{proposition}

\begin{definition}[Hom colimits]\label{def:pair-hom}
Define, with the filtered colimit taken in the category of abelian groups,
\[
\begin{aligned}
 &\Hom_{\Pair}((E,K),(F,L))\\
 &\quad=
 \operatorname*{colim}_{M\in\mathsf I(E,F)}
 \Hom_{\mathcal C_M}(T_{M/E}K,T_{M/F}L).
\end{aligned}
\]
Equivalently, this is the set of representatives of
Definition~\ref{def:morphism-representative} modulo the equivalence relation
of Definition~\ref{def:representative-equality}.  Its zero element is the
class of $(M,0)$ at any common stage.
\end{definition}

\begin{definition}[Addition and negatives]\label{def:pair-addition}
Let $[M,u]$ and $[N,v]$ be two morphisms with the same source and target.
Choose $P\supset M,N$ and define
\[
 [M,u]+[N,v]=[P,u_P+v_P],
 \qquad -[M,u]=[M,-u].
\]
\end{definition}

\begin{definition}[Identities and composition]\label{def:pair-composition}
The identity of $(E,K)$ is
\[
 [E,\operatorname{id}_{T_{E/E}K}],
\]
where the unit isomorphism identifies $T_{E/E}K$ with $K$.  Suppose
\[
 [M,u]:(E,K)\longrightarrow(F,L),
 \qquad
 [N,v]:(F,L)\longrightarrow(G,R).
\]
Choose a finite coefficient field $P$ containing $M$ and $N$.  Their composite is
\[
 [N,v]\circ[M,u]=[P,v_P\circ u_P],
\]
where
\[
 u_P:T_{P/E}K\to T_{P/F}L,\qquad
 v_P:T_{P/F}L\to T_{P/G}R.
\]
\end{definition}

\begin{proposition}[Addition is representative-independent]
\label{prop:addition-well-defined}
The sum in Definition~\ref{def:pair-addition} is independent of the common
stage and of the two representatives.  It makes every pair Hom set an
abelian group, and transport maps are homomorphisms.
\depends{Propositions~\ref{prop:transport-tower} and
\ref{prop:equality-equivalence}; finite-stage additive category laws.}
\end{proposition}

\begin{proposition}[Composition is representative-independent]
\label{prop:composition-well-defined}
The composite in Definition~\ref{def:pair-composition} is independent of
the selected common field $P$.  Replacing either factor by an equivalent
representative gives an equivalent composite.  Composition is biadditive.
\depends{Propositions~\ref{prop:transport-tower},
\ref{prop:equality-equivalence}, and
\ref{prop:addition-well-defined}; functoriality of every transition functor.}
\end{proposition}

\begin{proposition}[Associativity and unit laws]
\label{prop:composition-laws}
For three composable pair morphisms $f,g,h$,
\[
 h\circ(g\circ f)=(h\circ g)\circ f.
\]
The class $[E,\operatorname{id}_{T_{E/E}K}]$ is both a left and a right
identity for $(E,K)$.
\depends{Proposition~\ref{prop:composition-well-defined}; associativity and
identity laws at a common finite stage; transition unit coherence.}
\end{proposition}

\begin{theorem}[Well-defined additive category]\label{thm:pair-category}
The preceding Hom groups, addition, identities, and composition make
$\Pair$ an additive category with objects $(E,K)$.  A zero object is
$(\Ql,0)$.  If $M$ contains $E$ and $F$, a biproduct of $(E,K)$ and $(F,L)$
is
\[
 \bigl(M,T_{M/E}K\oplus T_{M/F}L\bigr),
\]
with the stagewise inclusions and projections.
\depends{Propositions~\ref{prop:addition-well-defined},
\ref{prop:composition-well-defined}, and \ref{prop:composition-laws};
Definitions~\ref{def:pair-hom}--\ref{def:pair-composition}.}
\end{theorem}

\begin{definition}[Finite-stage inclusion]\label{def:stage-inclusion}
For each $E$, define
\[
 j_E:\mathcal C_E\longrightarrow\Pair,
 \qquad K\longmapsto(E,K),\qquad
 u\longmapsto[E,u].
\]
For $E\subset F$, define a natural isomorphism
\[
 \gamma_{F/E}:j_FT_{F/E}\xRightarrow{\sim}j_E
\]
whose component is represented at $F$ by
$\operatorname{id}_{T_{F/E}K}$, viewed as a morphism in both directions.
\end{definition}

\begin{proposition}[Stage inclusions and transition coherence]
\label{prop:stage-inclusions}
Each $j_E$ is additive.  The maps $\gamma_{F/E}$ are natural isomorphisms,
and for $E\subset F\subset G$ their two composites agree after inserting
$a_{G,F,E}$.  Thus $(\Pair,(j_E),(\gamma_{F/E}))$ is a cocone on the
coherent finite-stage diagram.
\depends{Theorem~\ref{thm:pair-category};
Definition~\ref{def:stage-inclusion}; transition unit and pentagon
coherence.}
\end{proposition}

\begin{proposition}[Finite common-stage principle]\label{prop:finite-stage-principle}
Let $(E_i,K_i)$, $1\leq i\leq r$, be finitely many objects of $\Pair$.
For every finite $M$ containing all $E_i$, the components of
$\gamma_{M/E_i}$ give specified isomorphisms
\[
 j_M(T_{M/E_i}K_i)\xrightarrow{\sim}j_{E_i}(K_i)=(E_i,K_i).
\]
After replacing the objects by these isomorphic objects at $M$, every
finite collection of morphisms and composites among them has
representatives over one further finite coefficient field $N\supset M$.  A finite list
of asserted equalities holds in $\Pair$ if and only if all of the resulting
equalities hold after transport to some finite $P\supset N$.
\depends{Proposition~\ref{prop:finext-filtered};
Definition~\ref{def:representative-equality}; Theorem~\ref{thm:pair-category};
Proposition~\ref{prop:stage-inclusions}.}
\end{proposition}

\section{Shifts, cones, and the triangulated structure}

\begin{definition}[Shift]\label{def:pair-shift}
On objects define
\[
 (E,K)[1]=(E,K[1]).
\]
For $f=[M,u]:(E,K)\to(F,L)$ define
\[
 f[1]=
 \left[M,\,
 \sigma_{M/F,L}^{-1}\circ u[1]\circ\sigma_{M/E,K}
 \right],
\]
whose representative has the fully typed source and target
\[
 T_{M/E}(K[1])\longrightarrow T_{M/F}(L[1]).
\]
The inverse shift is defined in the same way with $[-1]$.
\end{definition}

\begin{proposition}[Shift representative calculus]\label{prop:pair-shift-calculus}
If $(M,u)\sim(N,v)$, then the shifted representatives in
Definition~\ref{def:pair-shift} are equivalent.  Consequently $[1]$ and
$[-1]$ are well-defined additive endofunctors of $\Pair$.  They preserve
identities and composition, and the stagewise inverse-shift isomorphisms
induce natural isomorphisms
\[
 [-1][1]\xRightarrow{\sim}\operatorname{id}_{\Pair},
 \qquad
 [1][-1]\xRightarrow{\sim}\operatorname{id}_{\Pair}.
\]
For every $E$, the functor $j_E$ commutes with shift through the comparison
represented by $\operatorname{id}_{K[1]}$ at stage $E$.
\depends{Definition~\ref{def:pair-shift};
Proposition~\ref{prop:transport-tower}; naturality and tower coherence of
the maps $\sigma_{F/E,K}$; the shift functor laws in every $\mathcal C_E$.}
\end{proposition}

\begin{definition}[A common-stage cone]\label{def:pair-cone}
Let $f=[M,u]:(E,K)\to(F,L)$.  In the triangulated category $\mathcal C_M$,
choose a distinguished cone triangle
\[
 T_{M/E}K\xrightarrow{u}T_{M/F}L
 \xrightarrow{i_u}C(u)\xrightarrow{p_u}(T_{M/E}K)[1].
\]
The corresponding cone object in $\Pair$ is $(M,C(u))$.  The second pair
arrow is represented by
\[
 i_u:T_{M/F}L\longrightarrow T_{M/M}C(u)=C(u),
\]
and the third pair arrow is represented by
\[
 \sigma_{M/E,K}^{-1}\circ p_u:
 T_{M/M}C(u)=C(u)\longrightarrow T_{M/E}(K[1]).
\]
\end{definition}

\begin{proposition}[Cone compatibility with coefficient extension]
\label{prop:cone-extension}
Let $(M,u)$ represent $f$ and let $M\subset N$.  Exactness of $T_{N/M}$
gives an isomorphism of distinguished triangles between the scalar
extension of
\[
 T_{M/E}K\to T_{M/F}L\to C(u)\to T_{M/E}K[1]
\]
and a cone triangle of
$u_N:T_{N/E}K\to T_{N/F}L$.  In particular,
$T_{N/M}C(u)$ and $C(u_N)$ are isomorphic in $\mathcal C_N$.
\depends{Definition~\ref{def:pair-cone}; exactness and shift comparison of
$T_{N/M}$ in Theorem~\ref{thm:finite-stage-coherence}.}
\end{proposition}

\begin{proposition}[Cone compatibility with equivalent representatives]
\label{prop:cone-representatives}
If $(M,u)\sim(N,v)$ represent the same pair morphism, then their cone
triangles become isomorphic over a finite coefficient field $P$ witnessing
$u_P=v_P$.  The isomorphism may be selected with identity components on
the common source and target.
\depends{Propositions~\ref{prop:cone-extension} and
\ref{prop:finite-stage-principle}; the finite-stage morphism-of-triangles
axiom.}
\end{proposition}

\begin{proposition}[Independence of the finite-stage cone choice]
\label{prop:cone-choice}
Any two distinguished cone triangles selected for the same finite-stage
morphism $u$ are isomorphic by an isomorphism of triangles whose first two
components are identities.  Consequently the class of a cone triangle in
$\Pair$ depends only on the pair morphism.
\depends{The finite-stage uniqueness of cones up to a noncanonical
isomorphism; Proposition~\ref{prop:cone-representatives}.}
\end{proposition}

\begin{definition}[Stagewise distinguished triangle]
\label{def:stagewise-triangle}
A triangle in $\Pair$ is \emph{distinguished} if it is isomorphic to a
triangle that, after representing its three objects and three arrows over
one finite coefficient field $M$, is distinguished in $\mathcal C_M$.  Equivalently, it
is isomorphic to a common-stage cone triangle from
Definition~\ref{def:pair-cone}.
\end{definition}

\begin{proposition}[TR1: cone and identity triangles]
\label{prop:pair-tr1}
Every pair morphism $f:A\to B$ occurs as the first arrow of a distinguished
triangle
\[
 A\xrightarrow{f}B\longrightarrow C(f)\longrightarrow A[1].
\]
Every triangle isomorphic to a distinguished triangle is distinguished,
and for every $A$ the triangle
\[
 A\xrightarrow{\operatorname{id}_A}A\longrightarrow0\longrightarrow A[1]
\]
is distinguished.
\depends{Proposition~\ref{prop:cone-choice}; Definitions~\ref{def:pair-cone}
and \ref{def:stagewise-triangle}; finite-stage TR1.}
\end{proposition}

\begin{proposition}[TR2: rotation]\label{prop:pair-tr2}
A triangle
\[
 A\xrightarrow{f}B\xrightarrow{g}C\xrightarrow{h}A[1]
\]
is distinguished if and only if its rotation
\[
 B\xrightarrow{g}C\xrightarrow{h}A[1]
 \xrightarrow{-f[1]}B[1]
\]
is distinguished.
\depends{Proposition~\ref{prop:finite-stage-principle};
Definition~\ref{def:stagewise-triangle}; finite-stage TR2 and exact shift
comparisons.}
\end{proposition}

\begin{proposition}[TR3: morphisms of triangles]\label{prop:pair-tr3}
Given two distinguished triangles and a commutative square between their
first arrows, there is a third vertical arrow completing it to a morphism of
triangles in $\Pair$.
\depends{Proposition~\ref{prop:finite-stage-principle}; finite-stage TR3.}
\end{proposition}

\begin{proposition}[TR4: octahedral axiom]\label{prop:pair-tr4}
For composable morphisms $A\xrightarrow{f}B\xrightarrow{g}C$, cone triangles
for $f$, $g$, and $gf$ can be selected together with arrows forming an
octahedral diagram in $\Pair$; all faces required by TR4 commute and the
remaining cone triangle is distinguished.
\depends{Propositions~\ref{prop:finite-stage-principle} and
\ref{prop:cone-choice}; finite-stage TR4.}
\end{proposition}

\begin{theorem}[Stagewise triangulation]\label{thm:pair-triangulated}
The shift of Definition~\ref{def:pair-shift} and the distinguished triangles
of Definition~\ref{def:stagewise-triangle} make $\Pair$ an ordinary
triangulated category.  Every stage inclusion
\[
 j_E:\mathcal C_E\longrightarrow\Pair
\]
is exact, and every $\gamma_{F/E}$ is an exact natural isomorphism.
\depends{Proposition~\ref{prop:pair-shift-calculus};
Propositions~\ref{prop:pair-tr1}--\ref{prop:pair-tr4};
Proposition~\ref{prop:stage-inclusions}.}
\end{theorem}

\section{The \texorpdfstring{$\Qlb$}{Qbar-l}-linear structure}

\begin{definition}[Scalar multiplication at a common stage]
\label{def:qbar-scalar}
Let $\alpha\in\Qlb$ and $f=[M,u]:(E,K)\to(F,L)$.  The field
$\Ql(\alpha)$ is finite.  Choose a finite coefficient field $N$ containing $M$ and
$\alpha$, and define
\[
 \alpha f=[N,\alpha\,u_N],
\]
where $\alpha\,u_N$ uses the $N$-linear Hom group of $\mathcal C_N$.
\end{definition}

\begin{proposition}[Scalar stage independence]
\label{prop:scalar-stage}
If $N$ and $N'$ both contain the stage of $f$ and the scalar $\alpha$, the
representatives $[N,\alpha u_N]$ and $[N',\alpha u_{N'}]$ are equivalent.
Passing first to a larger stage before multiplying by $\alpha$ gives the
same pair morphism.
\depends{Definition~\ref{def:qbar-scalar}; Proposition~\ref{prop:transport-tower};
semilinearity of finite-stage transition functors.}
\end{proposition}

\begin{proposition}[Scalar representative independence]
\label{prop:scalar-representative}
If $[M,u]=[M',u']$, then
$[N,\alpha u_N]=[N',\alpha u'_{N'}]$ for any stages at which the two
expressions are defined.  Thus scalar multiplication is well-defined on
pair Hom groups.
\depends{Propositions~\ref{prop:scalar-stage} and
\ref{prop:equality-equivalence}.}
\end{proposition}

\begin{proposition}[Field laws and bilinearity]\label{prop:scalar-laws}
For $\alpha,\beta\in\Qlb$ and composable pair morphisms $f,g$,
\[
 (\alpha+\beta)f=\alpha f+\beta f,
 \quad (\alpha\beta)f=\alpha(\beta f),
 \quad 1f=f,
\]
\[
 (\alpha g)\circ f=\alpha(g\circ f)
   =g\circ(\alpha f).
\]
\depends{Proposition~\ref{prop:scalar-representative}; the field laws and
bilinearity in a finite stage containing $\alpha$ and $\beta$.}
\end{proposition}

\begin{theorem}[$\Qlb$-linear triangulated structure]
\label{thm:qbar-linear}
Thus $\Pair$ is a $\Qlb$-linear triangulated category.  The functor $j_E$
is $E$-linear, and the action of $\alpha\in E$ agrees with the original
$E$-linear action on $\mathcal C_E$.
\depends{Theorem~\ref{thm:pair-triangulated};
Propositions~\ref{prop:scalar-stage}--\ref{prop:scalar-laws}.}
\end{theorem}

\begin{proposition}[Scalar coherence with shifts and cones]
\label{prop:scalar-cones}
For every $\alpha\in\Qlb$,
$(\alpha f)[1]=\alpha f[1]$.  If $\alpha\neq0$, multiplication by $\alpha$
identifies a cone triangle of $f$ with a cone triangle of $\alpha f$ after
the standard rescaling of one structure map.
\depends{Theorems~\ref{thm:pair-triangulated} and
\ref{thm:qbar-linear}; the corresponding finite-stage statements.}
\end{proposition}

\section{Karoubi triples and the algebraically closed category}

\begin{definition}[Karoubi triples and their morphisms]
\label{def:karoubi-triples}
Define
\[
 \Dbc(X,\Qlb)_{\mathrm{2colim}}=\Kar(\Pair).
\]
An object is a triple $(E,K,e)$ in which $(E,K)\in\Pair$ and
\[
 e\in\operatorname{End}_{\Pair}(E,K),\qquad e^2=e.
\]
The idempotent may be represented by a map
\[
 e_M:T_{M/E}K\longrightarrow T_{M/E}K
\]
at a finite stage $M\supset E$, with $e_M^2=e_M$ after passage to a further
finite stage if necessary.

A morphism
\[
 u:(E,K,e)\longrightarrow(F,L,d)
\]
is a morphism $u:(E,K)\to(F,L)$ in $\Pair$ satisfying
\[
 u=d\,u\,e.
\]
The identity of $(E,K,e)$ is $e$.  Composition is inherited from $\Pair$.
\end{definition}

\begin{proposition}[Karoubi morphism equations and composition]
\label{prop:karoubi-composition}
For $u:(P,e)\to(Q,d)$, the condition $u=d\,u\,e$ is equivalent to the pair
of equations
\[
 d\,u=u,\qquad u\,e=u.
\]
If $u=d\,u\,e$ and $v=c\,v\,d$, then
\[
 v\,u=c\,(v\,u)\,e.
\]
Thus Karoubi morphisms are closed under composition, and $e$ is the identity
of $(P,e)$.
\depends{Definition~\ref{def:karoubi-triples}; idempotence of $c,d,e$.}
\end{proposition}

\begin{definition}[Shift and distinguished triangles of retracts]
\label{def:karoubi-triangles}
Define
\[
 (E,K,e)[1]=(E,K[1],e[1]).
\]
A triangle of Karoubi triples is distinguished if it is isomorphic, in the
category of triangles, to a direct summand of the image of a distinguished
triangle in $\Pair$.
\end{definition}

\begin{theorem}[Idempotent-complete triangulation]
\label{thm:karoubi-triangulated}
Definition~\ref{def:karoubi-triangles} gives
$\Dbc(X,\Qlb)_{\mathrm{2colim}}$ the canonical structure of an
idempotent-complete ordinary triangulated category.  It is $\Qlb$-linear,
and the canonical functor $\Pair\to\Kar(\Pair)$ is exact and
$\Qlb$-linear.  For every finite $E$, the composite
\[
 J_E:\mathcal C_E\xrightarrow{j_E}\Pair
       \longrightarrow\Dbc(X,\Qlb)_{\mathrm{2colim}}
\]
is exact and $E$-linear.
\depends{Theorems~\ref{thm:pair-triangulated} and
\ref{thm:qbar-linear}; Definitions~\ref{def:karoubi-triples} and
\ref{def:karoubi-triangles}; Proposition~\ref{prop:karoubi-composition};
the triangulated Karoubi-envelope theorem.}
\end{theorem}

\begin{definition}[Finite-stage constructible sheaves]
\label{def:finite-constructible-sheaves}
Let $\Cons(X,E)$ be the full subcategory of $\mathsf{Mod}(E_X)$ consisting
of sheaves $\mathcal F$ for which there is a finite partition
\[
 X=\coprod_{\alpha=1}^m X_\alpha
\]
by locally closed constructible subschemes such that
$\mathcal F|_{X_\alpha}$ is locally constant of finite-dimensional
$E$-vector spaces on the selected pro-etale site and, on every connected
component, its continuous monodromy image is bounded.  Equivalently, that
image is contained in a compact subgroup of $\operatorname{GL}_r(E)$, or
stabilizes an $\mathcal O_E$-lattice in $E^r$.  Morphisms are $E_X$-linear
sheaf morphisms.  These objects are called admissible constructible
$E$-sheaves.
\end{definition}

\begin{theorem}[Finite-stage constructible realization]
\label{thm:constructible-realization}
Assume that $X$ satisfies the hypotheses of
Theorem~\ref{thm:finite-stage-coherence}.  The category $\Cons(X,E)$ is an
abelian subcategory of $\mathsf{Mod}(E_X)$, closed under kernels,
cokernels, and extensions.  The inclusion induces an exact equivalence
\[
 \operatorname{Real}_E:D^b(\Cons(X,E))
 \xrightarrow{\sim}\mathcal C_E=\Dbc(X,E).
\]
Here $D^b(\Cons(X,E))$ is the Verdier localization of the homotopy category
of bounded complexes of constructible sheaves at quasi-isomorphisms.
\depends{Definition~\ref{def:finite-constructible-sheaves}; finite-stage
constructibility and finite cohomological dimension; the bounded
constructible realization theorem on the selected pro-etale site.}
\end{theorem}

\begin{proposition}[Finite-stage hearts and scalar extension]
\label{prop:finite-hearts}
The standard bounded $t$-structure on $\mathcal C_E$ has heart
$\mathcal A_E=\Cons(X,E)$.  The functor $T_{F/E}$ is $t$-exact and induces
an exact functor
\[
 T^{\heartsuit}_{F/E}:\mathcal A_E\longrightarrow\mathcal A_F,
 \qquad \mathcal F\longmapsto\mathcal F\otimes_EF.
\]
There are natural isomorphisms
\[
 H^i(T_{F/E}K)\xrightarrow{\sim}
 T^{\heartsuit}_{F/E}H^i(K)
\]
compatible with the transition units and associators.
\depends{The finite-stage constructible realization theorem; flatness of
$F/E$; exactness and coherence of finite scalar extension.}
\end{proposition}

\begin{theorem}[The filtered pair heart is abelian]\label{thm:pair-heart}
Form $\mathcal A^{\mathrm{pair}}$ from the coherent exact diagram
$(\mathcal A_E,T^{\heartsuit}_{F/E})$ by the common-stage construction of
Section~2, taking every Hom colimit in $\mathsf{Ab}$.  Then
$\mathcal A^{\mathrm{pair}}$ is abelian.  If a morphism is represented by
\[
 u:T^{\heartsuit}_{M/E}\mathcal F\longrightarrow
 T^{\heartsuit}_{M/F}\mathcal G,
\]
its kernel and cokernel are represented by $\ker(u)$ and
$\operatorname{coker}(u)$ in $\mathcal A_M$.  These objects and their
structure arrows are independent of the representative after passage to a
common finite stage.  Every $\mathcal A_E\to\mathcal A^{\mathrm{pair}}$ is
exact.  Hence $\mathcal A^{\mathrm{pair}}$ is idempotent-complete and
\[
 \mathcal A^{\mathrm{pair}}\xrightarrow{\sim}
 \Kar(\mathcal A^{\mathrm{pair}})
\]
is an exact equivalence of abelian categories.
\depends{Proposition~\ref{prop:finite-hearts}; filteredness of
$\mathsf{FinExt}_\ell$; exactness of every $T^{\heartsuit}_{F/E}$; the
common-stage equality and composition calculus of Section~2.}
\end{theorem}

\begin{definition}[Cohomology of a retract]\label{def:retract-cohomology}
Let $(E,K,e)$ be a Karoubi triple.  Represent $e$ at a finite stage $M$ by
\[
 e_M:T_{M/E}K\longrightarrow T_{M/E}K.
\]
Using Proposition~\ref{prop:finite-hearts}, $H^i(e_M)$ represents a
well-defined idempotent
\[
 H^i_{\mathrm{pair}}(e)\in
 \End_{\mathcal A^{\mathrm{pair}}}(E,H^i(K)).
\]
Define
\[
 H^i(E,K,e)=\bigl(E,H^i(K),H^i_{\mathrm{pair}}(e)\bigr)
 \quad\text{in }\Kar(\mathcal A^{\mathrm{pair}}).
\]
This records the image of the idempotent without selecting a splitting.
\end{definition}

\begin{proposition}[Cohomological functors]\label{prop:retract-cohomology}
The assignments $H^i$ of Definition~\ref{def:retract-cohomology} are
well-defined additive functors.  A distinguished triangle in
$\Dbc(X,\Qlb)_{\mathrm{2colim}}$ gives a long exact cohomology sequence in
$\Kar(\mathcal A^{\mathrm{pair}})$, where exactness is interpreted through
the abelian structure of Theorem~\ref{thm:pair-heart}.  The construction is
independent of the finite stage at which $e$ is represented.
\depends{Theorems~\ref{thm:karoubi-triangulated} and
\ref{thm:pair-heart}; Proposition~\ref{prop:finite-hearts}; stagewise
cohomology long exact sequences.}
\end{proposition}

\section{The 2-colimit universal property}

\begin{definition}[Exact pseudocone]\label{def:pseudocone}
Let $\mathcal T$ be an ordinary triangulated category.  An exact pseudocone
from the finite-stage diagram to $\mathcal T$ consists of exact functors
\[
 F_E:\mathcal C_E\longrightarrow\mathcal T
\]
and natural isomorphisms
\[
 c_{F/E}:F_FT_{F/E}\xRightarrow{\sim}F_E
 \qquad(E\subset F),
\]
A selected exact-functor structure on $F_E$ is an isomorphism
\[
 \theta_{E,K}:F_E(K[1])\xrightarrow{\sim}F_E(K)[1].
\]
Each $c_{F/E}$ is required to be an isomorphism of exact functors: for every
$K\in\mathcal C_E$,
\[
 \theta_{E,K}\circ c_{F/E,K[1]}
 =c_{F/E,K}[1]\circ\theta_{F,T_{F/E}K}
   \circ F_F(\sigma_{F/E,K}).
\]
The comparisons also satisfy the unit condition that $c_{E/E}$ agrees with
$F_E(\iota_E)$ and, for
$E\subset F\subset G$, the two natural transformations from
$F_GT_{G/F}T_{F/E}$ to $F_E$ agree:
\[
 c_{G/E}\circ F_G(a_{G,F,E})
 =c_{F/E}\circ(c_{G/F}T_{F/E}).
\]
A morphism of pseudocones is a family of exact natural transformations
commuting with all $c_{F/E}$; exact means that every component family
commutes with the selected shift comparisons $\theta_E$.
\end{definition}

\begin{proposition}[Functor induced by a pseudocone]
\label{prop:pseudocone-functor}
Let $(F_E,c_{F/E})$ be an exact pseudocone.  Define
$F:\Pair\to\mathcal T$ on objects by $F(E,K)=F_EK$.  For a representative
\[
 u:T_{M/E}K\longrightarrow T_{M/F}L
\]
define
\[
 F[M,u]=
 c_{M/F,L}\circ F_M(u)\circ c_{M/E,K}^{-1}:
 F_EK\longrightarrow F_FL.
\]
This morphism is independent of the representative and common stage.  The
assignment preserves identities and composition and is additive.
\depends{Definition~\ref{def:pseudocone}; the pseudocone unit and tower
equations; Propositions~\ref{prop:transport-tower},
\ref{prop:composition-well-defined}, and \ref{prop:composition-laws}.}
\end{proposition}

\begin{proposition}[Exactness of the induced pseudocone functor]
\label{prop:pseudocone-exact}
The functor of Proposition~\ref{prop:pseudocone-functor} commutes with shift
through the selected exact-functor comparison and sends every stagewise
distinguished triangle to a distinguished triangle in $\mathcal T$.
\depends{Theorem~\ref{thm:pair-triangulated}; exactness of every $F_E$ and
of every pseudocone comparison.}
\end{proposition}

\begin{theorem}[Universal property before Karoubi completion]
\label{thm:pair-universal}
For every ordinary triangulated category $\mathcal T$, restriction along
the $j_E$ gives an equivalence between
\begin{enumerate}
\item exact functors $\Pair\to\mathcal T$, with exact natural
transformations, namely transformations commuting with the selected shift
comparisons;
\item exact pseudocones $(F_E,c_{F/E})$ as in
Definition~\ref{def:pseudocone}.
\end{enumerate}
The induced functor is the exact functor of
Propositions~\ref{prop:pseudocone-functor} and
\ref{prop:pseudocone-exact}.
\depends{Propositions~\ref{prop:pseudocone-functor},
\ref{prop:pseudocone-exact}, and \ref{prop:stage-inclusions};
Definition~\ref{def:pseudocone}.}
\end{theorem}

\begin{theorem}[Universal property after Karoubi completion]
\label{thm:karoubi-universal}
If $\mathcal T$ is idempotent-complete, restriction along the $J_E$ gives
an equivalence between exact functors
\[
 \Dbc(X,\Qlb)_{\mathrm{2colim}}\longrightarrow\mathcal T
\]
and exact pseudocones to $\mathcal T$.  The extension of a functor from
$\Pair$ sends a Karoubi triple $(E,K,e)$ to the image of the idempotent
$\widetilde F(e)$ in $\mathcal T$, where
$\widetilde F:\Pair\to\mathcal T$ is the functor induced by the
pseudocone and
\[
 \widetilde F(e)\in
 \End_{\mathcal T}\bigl(\widetilde F(E,K)\bigr)
 =\End_{\mathcal T}\bigl(F_E(K)\bigr).
\]
\depends{Theorem~\ref{thm:pair-universal}; the universal property of the
Karoubi envelope; idempotent completeness of $\mathcal T$.}
\end{theorem}

\section{Comparison with a literal algebraic coefficient sheaf}

\begin{definition}[The literal algebraic $\Qlb$-sheaf]
\label{def:literal-qbar}
For each finite $E\subset\Qlb$, let $E_X$ be the ordinary pro-etale
coefficient sheaf.  Define the sheaf of ordinary commutative rings
\[
 \mathcal E_X=\operatorname*{colim}_{E\in\mathsf{FinExt}_\ell}E_X
\]
in the category of sheaves of rings.  Let
$j_{E,X}:E_X\to\mathcal E_X$ be the structure map.  Define
\[
D_{\mathrm{lit}}(X,\Qlb)=
 D(\mathsf{Mod}(\mathcal E_X)).
\]
\end{definition}

\begin{proposition}[Finite-stage coefficient sheaves inside $\mathcal E_X$]
\label{prop:literal-coefficient-stages}
For every finite $E/\Ql$ there is a canonical identification
\[
 E_X=a\!\left(U\longmapsto\operatorname{Cont}(|U|,E)\right).
\]
For $E\subset F$, the map $E_X\to F_X$ is a monomorphism of ring sheaves,
and every structure map $j_{E,X}:E_X\to\mathcal E_X$ is monic.  The sheaf
$\mathcal E_X$ is the filtered union of these sub-ring sheaves: every local
section is locally represented at a finite coefficient stage.
\depends{The continuous-function coefficient-sheaf theorem; exactness of
filtered colimits of sheaves of modules; monicity of finite coefficient-field
inclusions.}
\end{proposition}

\begin{definition}[Finite-stage literal subcategory]
\label{def:literal-finite-stage}
Let $\mathcal C_E^{\mathrm{lit}}$ be the full subcategory of
$D(\mathsf{Mod}(E_X))$ whose objects are bounded and have constructible
cohomology sheaves.  At this point this is only a full subcategory; its
closure under the triangulated operations is part of the next theorem.
\end{definition}

\begin{theorem}[Finite-stage literal realization and closure]
\label{thm:literal-realization}
Assume that $X$ satisfies the hypotheses of
Theorem~\ref{thm:finite-stage-coherence} and the integral--literal lattice
and Hom-localization hypotheses of the preceding coefficient note.  Then
$\mathcal C_E^{\mathrm{lit}}$ is an idempotent-complete triangulated
subcategory of $D(\mathsf{Mod}(E_X))$, and there is an exact equivalence
\[
 \rho_E:\mathcal C_E\xrightarrow{\sim}\mathcal C_E^{\mathrm{lit}}
\]
for every finite $E/\Ql$.  These equivalences may be equipped with exact
coherence isomorphisms
\[
 r_{F/E}:\rho_FT_{F/E}\xRightarrow{\sim}
 \bigl(\rho_E(-)\otimes^L_{E_X}F_X\bigr)
 \qquad(E\subset F)
\]
whose identity and triple-extension diagrams commute.
\depends{The finite-stage integral--literal comparison; literal
constructible closure under shifts, cones, and retracts; compatibility of
integral scalar extension with rational localization and literal scalar
extension.}
\end{theorem}

\begin{definition}[Literal realization and finite-stage descent]
\label{def:literal-descent}
Under Theorem~\ref{thm:literal-realization}, fix one coherent family
$(\rho_E,r_{F/E})$.  For $K\in\mathcal C_E$, put
\[
 L_E(K)=\rho_E(K)\otimes^L_{E_X}\mathcal E_X
 \quad\in D_{\mathrm{lit}}(X,\Qlb).
\]
An object of the literal derived category has \emph{finite-stage descent}
if it is isomorphic to $L_E(K)$ for some finite $E$ and
$K\in\mathcal C_E$.  Let $\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$ be
the full subcategory of such objects.
\depends{Theorem~\ref{thm:literal-realization}; derived extension of scalars
for ordinary sheaves of rings.}
\end{definition}

\begin{definition}[The continuous-function coefficient sheaf]
\label{def:continuous-qbar}
Give $\Qlb$ its $\ell$-adic topology and let
\[
 \mathcal E_X^{\mathrm{cont}}
 =a\!\left(U\longmapsto\operatorname{Cont}(|U|,\Qlb)\right),
\]
where $a$ denotes sheafification in ordinary commutative ring sheaves.
Finite-stage inclusion of values defines
\[
 \kappa_X:\mathcal E_X\longrightarrow\mathcal E_X^{\mathrm{cont}}.
\]
\end{definition}

\begin{theorem}[Algebraic versus continuous coefficient sheaves]
\label{thm:algebraic-continuous}
The morphism $\kappa_X$ is an isomorphism if every continuous map
$|U|\to\Qlb$ is, after a covering of $U$ in the selected pro-etale site,
valued in a single finite extension $E/\Ql$.  Conversely, if $\kappa_X$ is
an isomorphism, every local section of
$\mathcal E_X^{\mathrm{cont}}$ has such a local finite-stage description.
The literal comparison below is formulated for $\mathcal E_X$ and does not
require this additional condition.
\depends{Definitions~\ref{def:literal-qbar} and
\ref{def:continuous-qbar};
Proposition~\ref{prop:literal-coefficient-stages}; the sheaf universal
property of a filtered colimit.}
\end{theorem}

\begin{definition}[Literal comparison hypotheses]\label{def:literal-hypotheses}
The literal comparison hypotheses for $X$ are the following statements.
\begin{enumerate}[label=\textup{(L\arabic*)}]
\item $X$ is quasi-compact, quasi-separated, noetherian of finite Krull
dimension, and $\ell$ is invertible on $X$.
\item Every $L_E$ is exact on $\mathcal C_E$, and for $E\subset F$ there
is a coherent natural isomorphism $L_FT_{F/E}\xrightarrow{\sim}L_E$.
\item For all finite $E,F$, all $K\in\mathcal C_E$, all
$R\in\mathcal C_F$, and all $r\in\mathbf Z$, the canonical map
\[
\begin{aligned}
 &\operatorname*{colim}_{M\in\mathsf I(E,F)}
 \Hom_{\mathcal C_M}
   (T_{M/E}K,T_{M/F}R[r])\\
 &\qquad\longrightarrow
 \Hom_{D_{\mathrm{lit}}(X,\Qlb)}(L_EK,L_FR[r])
\end{aligned}
\]
is an isomorphism.
\item The essential image of the finite-stage objects is closed under the
finite direct sums and shifts used above.
\end{enumerate}
\end{definition}

\begin{proposition}[Exact literal extension and transition comparison]
\label{prop:literal-exact}
The $E_X$-algebra $\mathcal E_X$ is flat: it is the filtered colimit of the
finite free $E_X$-algebras $M_X$, where $M$ ranges over finite extensions
containing $E$.  Consequently
\[
 L_E=\bigl(-\otimes^L_{E_X}\mathcal E_X\bigr)\rho_E:
 \mathcal C_E\longrightarrow D_{\mathrm{lit}}(X,\Qlb)
\]
is exact.  The maps $r_{F/E}$ and tensor associativity induce exact natural
isomorphisms
\[
 \lambda_{F/E}:L_FT_{F/E}\xRightarrow{\sim}L_E
\]
satisfying the unit and triple-extension equations.
\depends{Theorem~\ref{thm:literal-realization}; flatness of filtered
colimits of modules; cofinality of finite coefficient fields containing $E$; finite
local freeness of $M_X$ over $E_X$ for finite $M/E$; associativity and unit
coherence for derived scalar extension.}
\end{proposition}

\begin{proposition}[Perfect internal-Hom base change]
\label{prop:literal-internal-hom-base-change}
Let $M\subset N$ be finite coefficient fields and let $P,Q$ be locally
perfect constructible $M_X$-complexes.  The canonical map
\[
 R\mathcal Hom_{M_X}(P,Q)\otimes^L_{M_X}N_X
 \longrightarrow
 R\mathcal Hom_{N_X}
   (P\otimes^L_{M_X}N_X,Q\otimes^L_{M_X}N_X)
\]
is an isomorphism in $D(\mathsf{Mod}(N_X))$, naturally in $P,Q$ and
compatible with a further extension $N\subset P'$.  It applies to
$P=\rho_M(T_{M/E}K)$ and $Q=\rho_M(T_{M/F}R)$.
\depends{Dualizability of locally perfect complexes; tensor--internal-Hom
adjunction; coherence of finite-stage realizations.}
\end{proposition}

\begin{definition}[Rectified internal-Hom system]
\label{def:rectified-internal-hom}
For $M\in\mathsf I(E,F)$ put
\[
 P_M=\rho_M(T_{M/E}K),\qquad Q_M=\rho_M(T_{M/F}R),
\]
and let $I_M^\bullet$ be a complex of $\mathcal E_X$-modules representing
\[
 R\mathcal Hom_{M_X}(P_M,Q_M)
 \otimes^L_{M_X}\mathcal E_X.
\]
For $M\subset N$, the perfect base-change isomorphism and the coherence
maps $r_{N/M}$ give a morphism in
$D(\mathsf{Mod}(\mathcal E_X))$
\[
 \tau_{N/M}:I_M\longrightarrow I_N.
\]
A \emph{rectification} consists of representatives $I_M^\bullet$ and chain
maps $\tau_{N/M}^\bullet$ for which
\[
 \tau_{M/M}^\bullet=\operatorname{id},\qquad
 \tau_{P/N}^\bullet\tau_{N/M}^\bullet=\tau_{P/M}^\bullet
\]
hold literally.  For a rectification define
\[
 \operatorname*{colim}^{\mathsf{Cpx}}_{M\in\mathsf I(E,F)}I_M^\bullet
\]
degreewise in the Grothendieck abelian category
$\mathsf{Mod}(\mathcal E_X)$, and then view the resulting complex in
$D(\mathsf{Mod}(\mathcal E_X))$.
\end{definition}

\begin{theorem}[Existence and invariance of rectification]
\label{thm:internal-hom-rectification}
The coherent maps $\tau_{N/M}$ admit a rectification.  Any two
rectifications have a common refinement connected to both by levelwise
quasi-isomorphisms, and the induced degreewise filtered colimit maps are
quasi-isomorphisms.  Hence the object
\[
 \operatorname*{colim}^{\mathsf{Cpx}}_{M\in\mathsf I(E,F)}I_M^\bullet
 \quad\in D(\mathsf{Mod}(\mathcal E_X))
\]
is canonically isomorphic after passage to any common rectification; no
uniqueness beyond this common-refinement compatibility is asserted.
\depends{Proposition~\ref{prop:literal-internal-hom-base-change}; exactness
of filtered colimits in $\mathsf{Mod}(\mathcal E_X)$; the strictification
theorem for a filtered coherent diagram of complexes.}
\end{theorem}

\begin{theorem}[Locally perfect internal Hom over the coefficient union]
\label{thm:literal-internal-hom-union}
Let $M/\Ql$ be finite and let $P,Q$ be locally perfect constructible
$M_X$-complexes.  The canonical tensor--Hom morphism
\[
 R\mathcal Hom_{M_X}(P,Q)\otimes^L_{M_X}\mathcal E_X
 \longrightarrow
 R\mathcal Hom_{\mathcal E_X}
 \bigl(P\otimes^L_{M_X}\mathcal E_X,
       Q\otimes^L_{M_X}\mathcal E_X\bigr)
\]
is an isomorphism in $D(\mathsf{Mod}(\mathcal E_X))$.  It is natural in
$P,Q$ and commutes with every finite coefficient extension $M\subset N$.
For $P=P_M$ and $Q=Q_M$, these isomorphisms identify the maps
$\tau_{N/M}$ with the identity transition of the displayed literal
internal Hom after transport through $r_{N/M}$.
\depends{Proposition~\ref{prop:literal-internal-hom-base-change}; local
perfectness and dualizability of $P$; flatness of
$\mathcal E_X=\operatorname*{colim}_{N\supset M}N_X$ over $M_X$;
tensor--internal-Hom adjunction.}
\end{theorem}

\begin{theorem}[Filtered descent of the internal Hom]
\label{thm:literal-internal-hom-descent}
Assume (L1) and that every finite-stage constructible complex under
consideration is locally perfect.  For finite $E,F$,
$K\in\mathcal C_E$, and $R\in\mathcal C_F$, the canonical map
\[
 \operatorname*{colim}^{\mathsf{Cpx}}_{M\in\mathsf I(E,F)}I_M^\bullet
 \longrightarrow
 R\mathcal Hom_{\mathcal E_X}(L_EK,L_FR)
\]
is an isomorphism in $D(\mathsf{Mod}(\mathcal E_X))$.
\depends{Proposition~\ref{prop:literal-exact};
Theorems~\ref{thm:internal-hom-rectification} and
\ref{thm:literal-internal-hom-union}; exactness of degreewise filtered
colimits in $\mathsf{Mod}(\mathcal E_X)$.}
\end{theorem}

\begin{proposition}[Uniform constructibility of the rectified internal Homs]
\label{prop:rectified-internal-hom-uniform}
Assume (L1), finite-stage local perfectness, and finite-stage preservation
of constructibility and finite Tor dimension by internal Hom.  Fix finite
$E,F$, $K\in\mathcal C_E$, and $R\in\mathcal C_F$.  There is one interval
$[a,b]$, independent of $M\in\mathsf I(E,F)$, such that every
$I_M^\bullet$ has constructible finite-Tor cohomology and
\[
 H^r(I_M^\bullet)=0\qquad(r\notin[a,b]).
\]
The same interval applies to every common rectification and to its
degreewise filtered colimit.
\depends{Theorem~\ref{thm:finite-stage-coherence};
Propositions~\ref{prop:literal-internal-hom-base-change} and
\ref{prop:literal-exact}; local perfectness of $P_M,Q_M$; invariance of Tor
amplitude under finite flat coefficient extension.}
\end{proposition}

\begin{theorem}[Derived global sections of a rectified filtered diagram]
\label{thm:rgamma-filtered}
Assume that $X$ is quasi-compact and quasi-separated and that cohomology
of constructible finite-Tor sheaves on quasi-compact opens vanishes above
one fixed degree.  Let $\mathsf I$ be filtered and let
$(C_i^\bullet)_{i\in\mathsf I}$ be a rectified diagram of
$\mathcal E_X$-module complexes such that all $H^r(C_i^\bullet)$ are
constructible of finite Tor dimension and vanish outside one interval
$[a,b]$ independent of $i$.  There is a canonical morphism
\[
 \theta_C:
 \operatorname*{colim}^{\mathsf{Cpx}}_{i\in\mathsf I}
 R\Gamma(X,C_i^\bullet)
 \longrightarrow
 R\Gamma\!\left(X,
   \operatorname*{colim}^{\mathsf{Cpx}}_{i\in\mathsf I}C_i^\bullet
 \right)
 \quad\text{in }D(\Qlb).
\]
On the left, choose one compatible K-injective rectification and take the
degreewise filtered colimit of its global-section complexes; exactness of
filtered colimits makes this object and $\theta_C$ independent of that
choice.  The morphism $\theta_C$ is an isomorphism.
\depends{Exactness of filtered colimits of sheaves and of $\Qlb$-modules;
the uniform cohomological-dimension bound; bounded hypercohomology for the
common interval $[a,b]$; K-injective rectification of a filtered diagram.}
\end{theorem}

\begin{theorem}[Global Hom descent in every shift]
\label{thm:hom-descent}
Assume (L1), finite-stage local perfectness, finite-stage preservation of
constructibility and finite Tor dimension by internal Hom, and the uniform
cohomological-dimension bound of Theorem~\ref{thm:rgamma-filtered}.  Then
for every $r\in\mathbf Z$ the map in (L3) is an isomorphism.  Thus (L3)
holds simultaneously in all shifts.
\depends{Theorems~\ref{thm:literal-internal-hom-descent} and
\ref{thm:rgamma-filtered};
Proposition~\ref{prop:rectified-internal-hom-uniform}; the derived
internal-Hom representation of shifted morphisms.}
\end{theorem}

\begin{proposition}[Closure under sums and shifts]
\label{prop:literal-basic-closure}
The essential image of the objects $L_E(K)$ is closed under finite direct
sums and shifts.  More precisely, after replacing two stages by a common
$M$,
\[
 L_E(K)\oplus L_F(R)\cong
 L_M(T_{M/E}K\oplus T_{M/F}R),
 \qquad
 L_E(K)[r]\cong L_E(K[r]).
\]
Hence (L4) follows from Proposition~\ref{prop:literal-exact}.
\depends{Proposition~\ref{prop:literal-exact}; exactness and additivity of
the finite-stage transition functors.}
\end{proposition}

\begin{definition}[The literal comparison functor]
\label{def:literal-functor}
Under the literal comparison hypotheses, define
\[
 \operatorname{Lit}_X:\Pair\longrightarrow
 D_{\mathrm{lit}}(X,\Qlb),
 \qquad (E,K)\longmapsto L_E(K)
\]
on objects.  For a representative
\[
 u:T_{M/E}K\longrightarrow T_{M/F}R,
\]
define $\operatorname{Lit}_X[M,u]$ to be the composite
\[
 L_EK\xrightarrow{\lambda_{M/E,K}^{-1}}
 L_M(T_{M/E}K)\xrightarrow{L_M(u)}
 L_M(T_{M/F}R)\xrightarrow{\lambda_{M/F,R}}L_FR.
\]
\end{definition}

\begin{definition}[Literal $\Qlb$-scalar action]
\label{def:literal-scalars}
For $\alpha\in\Qlb$, choose a finite coefficient field $E_\alpha$ containing it.  The
constant section $\alpha\in E_{\alpha,X}$ and the structure map
$E_{\alpha,X}\to\mathcal E_X$ define a central multiplication endomorphism
$m_{\alpha,M}$ of every $\mathcal E_X$-module complex $M$.  For a morphism
$a:M\to N$ in $D_{\mathrm{lit}}(X,\Qlb)$ set
\[
 \alpha a=m_{\alpha,N}\circ a=a\circ m_{\alpha,M}.
\]
This is independent of $E_\alpha$ and gives every literal Hom group its
natural $\Qlb$-vector-space structure.
\end{definition}

\begin{proposition}[Scalar compatibility of literal realization]
\label{prop:literal-scalars}
For every $\alpha\in\Qlb$ and every $f\in\Hom_{\Pair}(P,Q)$,
\[
 \operatorname{Lit}_X(\alpha f)=
 \alpha\operatorname{Lit}_X(f).
\]
Thus literal realization is $\Qlb$-linear as a functor to the ambient
literal derived category.
\depends{Definitions~\ref{def:literal-scalars} and
\ref{def:literal-functor}; Proposition~\ref{prop:scalar-stage}; coherence
of the maps $\lambda_{F/E}$.}
\end{proposition}

\begin{proposition}[Ambient triangle compatibility of literal realization]
\label{prop:literal-functor}
Definition~\ref{def:literal-functor} is independent of the representative
and common stage, and it preserves identities, addition, composition,
and shifts.  It sends every stagewise distinguished triangle in $\Pair$ to
a distinguished triangle in the ambient triangulated category
$D_{\mathrm{lit}}(X,\Qlb)$.  Its objectwise essential image is contained in
$\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$.
\depends{Proposition~\ref{prop:literal-exact}; the unit and
triple-extension equations for $\lambda$; Propositions
\ref{prop:transport-tower}, \ref{prop:composition-laws}, and
\ref{prop:pair-shift-calculus}; Proposition~\ref{prop:literal-scalars}.}
\end{proposition}

\begin{theorem}[Full faithfulness of literal realization]
\label{thm:literal-fully-faithful}
Under (L3), the map induced by $\operatorname{Lit}_X$ on every Hom group is
the isomorphism displayed in (L3) for $r=0$.  Therefore
$\operatorname{Lit}_X$ is fully faithful.
\depends{Definition~\ref{def:pair-hom}; Definition~\ref{def:literal-functor};
hypothesis (L3).}
\end{theorem}

\begin{proposition}[Cone closure of finite-stage descent]
\label{prop:literal-cone-closure}
Under (L2) and (L3), $\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$ is a full
triangulated subcategory of $D_{\mathrm{lit}}(X,\Qlb)$.  Explicitly, for a
morphism $a:L_EK\to L_FR$, (L3) supplies a common stage $M$ and
$u:T_{M/E}K\to T_{M/F}R$ inducing $a$; then
\[
 \operatorname{Cone}(a)\cong L_M(\operatorname{Cone}(u)).
\]
The isomorphism class of this cone object, and hence its membership in
$\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$, is independent of the
selected representative of $a$; no canonical cone isomorphism is asserted.
\depends{Theorem~\ref{thm:hom-descent};
Proposition~\ref{prop:literal-functor}; exactness of $L_M$ and uniqueness of
cones up to isomorphism.}
\end{proposition}

\begin{proposition}[Essential surjectivity onto descended objects]
\label{prop:literal-essential-surjectivity}
Every object of $\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$ is isomorphic
to $\operatorname{Lit}_X(E,K)=L_E(K)$ for some finite $E/\Ql$ and
$K\in\mathcal C_E$.  Thus $\operatorname{Lit}_X$ is essentially
surjective onto $\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$.
\depends{Definitions~\ref{def:literal-descent} and
\ref{def:literal-functor}.}
\end{proposition}

\begin{theorem}[Pair category versus literal descent]
\label{thm:pair-literal}
Under the literal comparison hypotheses, the exact functor
$\operatorname{Lit}_X:\Pair\to D_{\mathrm{lit}}(X,\Qlb)$ corestricts to
the now-triangulated subcategory
$\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)$.  This corestriction is exact,
fully faithful, and essentially surjective, and gives an exact equivalence
\[
 \Pair\xrightarrow{\sim}
 \mathcal D_{\mathrm{desc}}(X,\mathcal E_X).
\]
\depends{Proposition~\ref{prop:literal-functor};
Theorem~\ref{thm:literal-fully-faithful};
Propositions~\ref{prop:literal-cone-closure} and
\ref{prop:literal-essential-surjectivity};
Definition~\ref{def:literal-descent}.}
\end{theorem}

\begin{corollary}[Literal $\Qlb$ comparison]
\label{cor:literal-comparison}
Under the literal comparison hypotheses, $\operatorname{Lit}_X$ extends to
an exact $\Qlb$-linear equivalence
\[
 \Dbc(X,\Qlb)_{\mathrm{2colim}}
 \xrightarrow{\sim}
 \Kar\bigl(\mathcal D_{\mathrm{desc}}(X,\mathcal E_X)\bigr).
\]
Thus the explicit 2-colimit agrees with the literal algebraic coefficient
category exactly for finite-stage descended objects and their formal direct
summands; no assertion is made about arbitrary
$\mathcal E_X$-module complexes.
\depends{Theorems~\ref{thm:pair-literal} and
\ref{thm:karoubi-triangulated};
Proposition~\ref{prop:literal-scalars}; functoriality of Karoubi completion.}
\end{corollary}

\section{Descent of coefficient-compatible functors}

Let $(\mathcal D_E,S_{F/E},b_{G,F,E})$ be another coherent diagram of
ordinary triangulated categories indexed by $\mathsf{FinExt}_\ell$.

\begin{definition}[Coefficient-compatible family]\label{def:compatible-family}
A coefficient-compatible family consists of exact functors
\[
 \Phi_E:\mathcal C_E\longrightarrow\mathcal D_E
\]
and natural isomorphisms
\[
 \beta_{F/E}:S_{F/E}\Phi_E
 \xRightarrow{\sim}\Phi_FT_{F/E}.
\]
Write
\[
 \theta^\Phi_{E,K}:\Phi_E(K[1])\xrightarrow{\sim}\Phi_E(K)[1],
 \qquad
 \tau^S_{F/E,Q}:S_{F/E}(Q[1])\xrightarrow{\sim}(S_{F/E}Q)[1]
\]
for the selected shift comparisons.  Require $\beta_{F/E}$ to be an
isomorphism of exact functors; explicitly, for every $K\in\mathcal C_E$,
the two maps from $S_{F/E}\Phi_E(K[1])$ to
$(\Phi_FT_{F/E}K)[1]$ are equal:
\[
\begin{aligned}
 &\theta^\Phi_{F,T_{F/E}K}\circ
   \Phi_F(\sigma_{F/E,K})\circ\beta_{F/E,K[1]}\\
 &\qquad=
 \beta_{F/E,K}[1]\circ\tau^S_{F/E,\Phi_EK}\circ
 S_{F/E}(\theta^\Phi_{E,K}).
\end{aligned}
\]
For $E\subset F\subset G$, require equality of the two transformations
from $S_{G/F}S_{F/E}\Phi_E$ to $\Phi_GT_{G/E}$:
\[
 (\Phi_Ga_{G,F,E})\circ
 (\beta_{G/F}T_{F/E})\circ(S_{G/F}\beta_{F/E})
 =\beta_{G/E}\circ(b_{G,F,E}\Phi_E).
\]
The analogous unit equation is required at $E=F$.
\end{definition}

\begin{theorem}[Descent of a compatible family]
\label{thm:functor-descent}
A coefficient-compatible family induces an exact functor between the two
pair categories.  On objects it sends $(E,K)$ to $(E,\Phi_EK)$.  On a
common-stage representative
\[
 u:T_{M/E}K\longrightarrow T_{M/F}L
\]
it sends $u$ to
\[
 S_{M/E}\Phi_EK
 \xrightarrow{\beta_{M/E}}
 \Phi_MT_{M/E}K
 \xrightarrow{\Phi_M(u)}
 \Phi_MT_{M/F}L
 \xrightarrow{\beta_{M/F}^{-1}}
 S_{M/F}\Phi_FL.
\]
It extends uniquely to the Karoubi completions.  Compatible natural
transformations, adjunction units and counits, and equations descend by the
same common-stage construction.
\depends{Definition~\ref{def:compatible-family};
Theorems~\ref{thm:pair-category}, \ref{thm:pair-triangulated}, and
\ref{thm:karoubi-universal}.}
\end{theorem}

\section{Strict Weil complexes at finite stages}

Let $X_0$ be of finite type over $\mathbf F_q$, let
\[
 X=X_0\times_{\mathbf F_q}\overline{\mathbf F}_q,
\]
and let $\FrX:X\to X$ be geometric Frobenius.  For every finite coefficient
field $E$, the coefficient sheaf has a specified isomorphism
$\FrX^{-1}E_X\xrightarrow{\sim}E_X$, so coefficient-linear pullback is an
exact autoequivalence
\[
 \FrX^*:\Cons(X,E)\longrightarrow\Cons(X,E).
\]

\begin{definition}[Constructible Weil sheaf]\label{def:weil-sheaf}
A constructible Weil $E$-sheaf on $X_0$ is a pair
$(\mathcal F,\varphi)$ with $\mathcal F\in\Cons(X,E)$ and an isomorphism
\[
 \varphi:\FrX^*\mathcal F\xrightarrow{\sim}\mathcal F.
\]
A morphism
$u:(\mathcal F,\varphi)\to(\mathcal G,\psi)$ is a constructible sheaf map
$u:\mathcal F\to\mathcal G$ satisfying
\[
 u\circ\varphi=\psi\circ \FrX^*u.
\]
Let $\Cons_{\mathrm W}(X_0,E)$ denote this category.
\end{definition}

\begin{proposition}[The abelian Weil category]\label{prop:weil-abelian}
The category $\Cons_{\mathrm W}(X_0,E)$ is abelian.  Kernels and cokernels
are taken on the underlying constructible sheaves, with the induced
Frobenius isomorphisms.  The forgetful functor
\[
 \operatorname{For}_E:\Cons_{\mathrm W}(X_0,E)
 \longrightarrow\Cons(X,E)
\]
is exact and conservative.
\depends{Definition~\ref{def:weil-sheaf}; exactness of $\FrX^*$; abelian
closure of constructible sheaves.}
\end{proposition}

\begin{definition}[Strict finite-stage Weil derived category]
\label{def:strict-weil-derived}
A strict bounded Weil complex is a bounded cochain complex in
$\Cons_{\mathrm W}(X_0,E)$.  Thus every differential commutes with the
degreewise Frobenius isomorphisms.  An equivariant quasi-isomorphism is a
chain map whose underlying map of constructible sheaf complexes is a
quasi-isomorphism.  Define the ordinary Verdier localization
\[
 \mathsf K^b_{\mathrm W}(X_0,E)
 =\mathsf{Hot}\bigl(\mathsf{Cpx}^b(\Cons_{\mathrm W}(X_0,E))\bigr),
\]
and then
\[
 \mathcal W_E=D^b(\Cons_{\mathrm W}(X_0,E))
 =\mathsf K^b_{\mathrm W}(X_0,E)
   [\{\text{equivariant quasi-isomorphisms}\}^{-1}].
\]
Its shifts and distinguished triangles are induced by strict shifts and
strict mapping cones.
\end{definition}

\begin{theorem}[Finite-stage strict triangulation]
\label{thm:strict-weil-triangulated}
The category $\mathcal W_E$ is an $E$-linear ordinary triangulated category.
Forgetting Frobenius induces an exact conservative functor
\[
 \operatorname{For}_E:\mathcal W_E\longrightarrow\mathcal C_E.
\]
It is $E$-linear and $t$-exact, and its functor on hearts is the faithful
exact forgetful functor of Proposition~\ref{prop:weil-abelian}.
Its image carries a derived isomorphism $\FrX^*K\xrightarrow{\sim}K$, but
$\mathcal W_E$ is defined by strict complexes in an abelian category, not
as an unqualified category of equivariant objects in a bare triangulated
category.
\depends{Proposition~\ref{prop:weil-abelian};
Definition~\ref{def:strict-weil-derived}; Verdier localization of a bounded
derived category; Theorem~\ref{thm:constructible-realization}.}
\end{theorem}

\begin{definition}[Strict scalar extension]\label{def:strict-scalar-extension}
For $E\subset F$, first define an exact functor on the abelian Weil hearts
\[
 T^{\mathrm W,\heartsuit}_{F/E}:
 \Cons_{\mathrm W}(X_0,E)\longrightarrow
 \Cons_{\mathrm W}(X_0,F),
 \qquad
 (\mathcal F,\varphi)\longmapsto
 (\mathcal F\otimes_{E_X}F_X,\varphi_F).
\]
Using the constant-coefficient identifications
$\FrX^*E_X\cong E_X$ and $\FrX^*F_X\cong F_X$, define $\varphi_F$ by
\[
 \FrX^*(\mathcal F\otimes_{E_X}F_X)
 \xrightarrow{\sim}\FrX^*\mathcal F\otimes_{E_X}F_X
 \xrightarrow{\varphi\otimes1}\mathcal F\otimes_{E_X}F_X.
\]
Termwise application to bounded complexes induces the exact derived functor
\[
 T^{\mathrm W}_{F/E}:\mathcal W_E\longrightarrow\mathcal W_F.
\]
The finite-stage tensor unit and associator give
$\iota_E^{\mathrm W}$ and $a_{G,F,E}^{\mathrm W}$.  Adopt the normalized
convention
\[
 T^{\mathrm W}_{E/E}=\operatorname{id}_{\mathcal W_E},
 \qquad \iota_E^{\mathrm W}=\operatorname{id}
\]
literally, including on the heart.
\end{definition}

\begin{theorem}[Strict transition coherence]
\label{thm:strict-transition-coherence}
Every $T^{\mathrm W}_{F/E}$ is exact and $t$-exact.  It is $E$-linear in the
typed sense
\[
 T^{\mathrm W}_{F/E}(\alpha u)
 =\operatorname{incl}_{F/E}(\alpha)T^{\mathrm W}_{F/E}(u)
 \qquad(\alpha\in E),
\]
where $\operatorname{incl}_{F/E}:E\hookrightarrow F$.  Retain the natural shift
comparison
\[
 \sigma^{\mathrm W}_{F/E,K}:
 T^{\mathrm W}_{F/E}(K[1])
 \xrightarrow{\sim}(T^{\mathrm W}_{F/E}K)[1].
\]
The shift comparisons commute with the unit and associator.  The unit and
composition isomorphisms are morphisms of strict Weil complexes and satisfy
the unit and pentagon equations.  The squares
\[
\begin{CD}
 \mathcal W_E @>{T^{\mathrm W}_{F/E}}>> \mathcal W_F\\
 @V{\operatorname{For}_E}VV @VV{\operatorname{For}_F}V\\
 \mathcal C_E @>{T_{F/E}}>> \mathcal C_F
\end{CD}
\]
commute up to specified coherent exact natural isomorphisms, compatible
with the displayed shift comparisons and with the standard $t$-structures.
\depends{Definition~\ref{def:strict-scalar-extension};
Theorems~\ref{thm:strict-weil-triangulated} and
\ref{thm:finite-stage-coherence}; tensor associativity and Frobenius
pullback compatibility.}
\end{theorem}

\begin{definition}[The strict Weil pair heart]
\label{def:weil-pair-heart}
Let $\mathcal A_E^{\mathrm W}=\Cons_{\mathrm W}(X_0,E)$.  Define
$\mathcal A^{\mathrm W,pair}$ to have objects $(E,\mathcal F)$ with
$\mathcal F\in\mathcal A_E^{\mathrm W}$ and Hom groups
\[
 \Hom_{\mathcal A^{\mathrm W,pair}}
 ((E,\mathcal F),(F,\mathcal G))
 =\operatorname*{colim}_{M\in\mathsf I(E,F)}
 \Hom_{\mathcal A_M^{\mathrm W}}
 (T^{\mathrm W,\heartsuit}_{M/E}\mathcal F,
  T^{\mathrm W,\heartsuit}_{M/F}\mathcal G),
\]
where the colimit is in $\mathsf{Ab}$.  A representative is a map at one
common stage $M$.  For $M\subset N$, transport is conjugation of
$T^{\mathrm W,\heartsuit}_{N/M}(u)$ by the two heart associators.
Representatives are equal when their transports agree at a further common
stage.  The identity is represented by the identity at the normalized
stage $E$; addition and composition are performed after transport to one
common stage.
\end{definition}

\begin{theorem}[The strict Weil pair heart is abelian]
\label{thm:weil-pair-heart}
The exact functors on hearts
\[
 \Cons_{\mathrm W}(X_0,E)\longrightarrow
 \Cons_{\mathrm W}(X_0,F)
\]
induced by $T^{\mathrm W}_{F/E}$ form a coherent filtered diagram of
abelian categories.  Its common-stage pair category
$\mathcal A^{\mathrm W,pair}$ of
Definition~\ref{def:weil-pair-heart} is abelian; kernels and cokernels of a
represented equivariant morphism are its strict finite-stage kernel and
cokernel.  The stage inclusions are exact and
$\mathcal A^{\mathrm W,pair}\to\Kar(\mathcal A^{\mathrm W,pair})$ is an
equivalence of abelian categories.
\depends{Proposition~\ref{prop:weil-abelian};
Theorem~\ref{thm:strict-transition-coherence}; the kernel--cokernel
construction of Theorem~\ref{thm:pair-heart}.}
\end{theorem}

\section{The strict Weil 2-colimit}

\begin{definition}[The Weil pair category]\label{def:weil-pair}
The category $\PairW$ has objects $(E,K)$ with $K\in\mathcal W_E$ and Hom
groups, with the colimit taken in $\mathsf{Ab}$,
\[
\begin{aligned}
 &\Hom_{\PairW}((E,K),(F,L))\\
 &\quad=
 \operatorname*{colim}_{M\in\mathsf I(E,F)}
 \Hom_{\mathcal W_M}
 (T^{\mathrm W}_{M/E}K,T^{\mathrm W}_{M/F}L).
\end{aligned}
\]
A representative is therefore a strict equivariant derived morphism at one
common finite stage:
\[
 (M,u),\qquad
 u:T^{\mathrm W}_{M/E}K\longrightarrow T^{\mathrm W}_{M/F}L
 \quad\text{in }\mathcal W_M.
\]
For $M\subset N$, define its transport by
\[
 u_N=
 a^{\mathrm W}_{N,M,F,L}\circ T^{\mathrm W}_{N/M}(u)\circ
 \bigl(a^{\mathrm W}_{N,M,E,K}\bigr)^{-1}:
 T^{\mathrm W}_{N/E}K\longrightarrow T^{\mathrm W}_{N/F}L.
\]
Representatives $(M,u)$ and $(N,v)$ are equal if their transports to some
finite $P\supset M,N$ agree in $\mathcal W_P$.  The identity of $(E,K)$ is
represented by
$\operatorname{id}_{T^{\mathrm W}_{E/E}K}$.  Composition is defined after
transporting two representatives to one common finite stage.
\end{definition}

\begin{proposition}[Strict Weil transport equations]
\label{prop:weil-transport}
For a representative $(M,u)$, the transport formula of
Definition~\ref{def:weil-pair} satisfies
\[
 u_M=u,
 \qquad
 (u_N)_P=u_P\quad(M\subset N\subset P).
\]
It preserves zero, sums, negatives, identities, and composites whenever
the latter are defined at one stage.
\depends{Theorem~\ref{thm:strict-transition-coherence}; the strict unit and
pentagon equations; functoriality and additivity of every
$T^{\mathrm W}_{N/M}$.}
\end{proposition}

\begin{proposition}[Strict Weil equality of representatives]
\label{prop:weil-representative-equality}
The common-stage equality relation of Definition~\ref{def:weil-pair} is
reflexive, symmetric, and transitive, and is preserved by transport to any
larger finite stage.
\depends{Proposition~\ref{prop:weil-transport}; filteredness of
$\mathsf{FinExt}_\ell$.}
\end{proposition}

\begin{proposition}[Strict Weil operation calculus]
\label{prop:weil-representatives}
Addition and composition are independent of all chosen stages and
representatives.  Composition is associative, and the selected identities
satisfy the left and right unit laws.
\depends{Propositions~\ref{prop:weil-transport} and
\ref{prop:weil-representative-equality}; the abelian-group and category laws
in each finite-stage category $\mathcal W_M$.}
\end{proposition}

\begin{theorem}[The additive Weil pair category]
\label{thm:weil-pair-additive}
The Hom groups, addition, identities, and composition of
Definition~\ref{def:weil-pair} make $\PairW$ an additive category.  If
$M\supset E,F$, a biproduct of $(E,K)$ and $(F,L)$ is
\[
 \bigl(M,T^{\mathrm W}_{M/E}K\oplus T^{\mathrm W}_{M/F}L\bigr),
\]
with the stagewise inclusions and projections.  The finite-stage functors
$\mathcal W_E\to\PairW$ defined below are additive.
\depends{Proposition~\ref{prop:weil-representatives}; filteredness of
$\mathsf{FinExt}_\ell$; additivity and coherence of strict scalar
extension.}
\end{theorem}

\begin{definition}[Strict Weil stage inclusions]
\label{def:weil-stage-inclusion}
For each $E$, define
\[
 j_E^{\mathrm W}:\mathcal W_E\longrightarrow\PairW,
 \qquad K\longmapsto(E,K),\qquad u\longmapsto[E,u].
\]
For $E\subset F$, define
\[
 \gamma^{\mathrm W}_{F/E}:
 j_F^{\mathrm W}T^{\mathrm W}_{F/E}
 \xRightarrow{\sim}j_E^{\mathrm W}
\]
whose component at $K$ is represented over $F$ by
$\operatorname{id}_{T^{\mathrm W}_{F/E}K}$ in both directions.
\end{definition}

\begin{proposition}[The strict Weil additive cocone]
\label{prop:weil-stage-cocone}
Every $j_E^{\mathrm W}$ is additive, every
$\gamma^{\mathrm W}_{F/E}$ is a natural isomorphism, and for
$E\subset F\subset G$ the two comparison maps from
$j_G^{\mathrm W}T^{\mathrm W}_{G/F}T^{\mathrm W}_{F/E}$ to
$j_E^{\mathrm W}$ agree after inserting $a^{\mathrm W}_{G,F,E}$.
Thus $(\PairW,j_E^{\mathrm W},\gamma^{\mathrm W}_{F/E})$ is a coherent
additive cocone.
\depends{Theorem~\ref{thm:weil-pair-additive};
Definition~\ref{def:weil-stage-inclusion}; the strict unit and pentagon
equations of Theorem~\ref{thm:strict-transition-coherence}.}
\end{proposition}

\begin{proposition}[Finite common stage for Weil diagrams]
\label{prop:weil-common-stage}
For finitely many $(E_i,K_i)$ and any finite $M$ containing all $E_i$, the
maps $\gamma^{\mathrm W}_{M/E_i,K_i}$ give isomorphisms
\[
 j_M^{\mathrm W}(T^{\mathrm W}_{M/E_i}K_i)
 \xrightarrow{\sim}(E_i,K_i).
\]
After these replacements, every finite collection of equivariant derived
morphisms and composites has representatives over one further finite
stage.  Every finite list of equalities is detected after one additional
finite extension.
\depends{Propositions~\ref{prop:weil-representatives} and
\ref{prop:weil-stage-cocone}; filteredness of $\mathsf{FinExt}_\ell$.}
\end{proposition}

\begin{definition}[Weil shifts and cone triangles]\label{def:weil-cones}
Define $(E,K)[1]=(E,K[1])$.  If $f=[M,u]:(E,K)\to(F,L)$, define
\[
 f[1]=\left[M,
 (\sigma^{\mathrm W}_{M/F,L})^{-1}\circ u[1]\circ
 \sigma^{\mathrm W}_{M/E,K}\right].
\]
Its representative is a map
\[
 T^{\mathrm W}_{M/E}(K[1])\longrightarrow
 T^{\mathrm W}_{M/F}(L[1]).
\]
Define $[-1]$ analogously.  For the representative
\[
 u:T^{\mathrm W}_{M/E}K\longrightarrow T^{\mathrm W}_{M/F}L,
\]
choose a strict distinguished cone triangle in $\mathcal W_M$,
\[
 T^{\mathrm W}_{M/E}K\xrightarrow{u}T^{\mathrm W}_{M/F}L
 \xrightarrow{i_u} C_{\mathrm W}(u)
 \xrightarrow{p_u}(T^{\mathrm W}_{M/E}K)[1].
\]
The cone object is $(M,C_{\mathrm W}(u))$.  The second pair arrow is
represented by
\[
 i_u:T^{\mathrm W}_{M/F}L\longrightarrow
 T^{\mathrm W}_{M/M}C_{\mathrm W}(u)=C_{\mathrm W}(u),
\]
and the third is represented by
\[
 (\sigma^{\mathrm W}_{M/E,K})^{-1}\circ p_u:
 C_{\mathrm W}(u)\longrightarrow T^{\mathrm W}_{M/E}(K[1]).
\]
A Weil pair triangle is distinguished if it is isomorphic to such a
triangle represented at one finite stage.
\end{definition}

\begin{proposition}[Strict Weil shift calculus]
\label{prop:weil-shift-calculus}
Equivalent representatives have equivalent shifts.  The formulas of
Definition~\ref{def:weil-cones} define additive endofunctors $[1]$ and
$[-1]$ of $\PairW$, preserve identities and composition, and inherit
natural inverse isomorphisms
\[
 [-1][1]\xRightarrow{\sim}\operatorname{id}_{\PairW},
 \qquad [1][-1]\xRightarrow{\sim}\operatorname{id}_{\PairW}.
\]
Every finite-stage inclusion commutes with shift through its specified
strict shift comparison.
\depends{Proposition~\ref{prop:weil-representatives};
Theorem~\ref{thm:strict-transition-coherence}; naturality and tower
coherence of $\sigma^{\mathrm W}$.}
\end{proposition}

\begin{proposition}[Independence of strict Weil cones]
\label{prop:weil-cone-independence}
Strict Weil cone triangles are preserved by finite scalar extension.
Equivalent representatives give isomorphic cone triangles at a common
stage, and two cone choices for one representative give isomorphic
triangles with identity components on the first two terms.
\depends{Definition~\ref{def:weil-cones}; exactness and cone compatibility
of $T^{\mathrm W}_{M/E}$; Proposition~\ref{prop:weil-representatives};
finite-stage uniqueness of cone triangles.}
\end{proposition}

\begin{proposition}[Verdier axioms for the Weil pair category]
\label{prop:weil-verdier-axioms}
The distinguished Weil pair triangles satisfy the following four
statements.
\begin{enumerate}
\item Every equivariant pair morphism has a distinguished cone triangle,
and the identity triangle is distinguished.
\item A Weil pair triangle is distinguished if and only if its signed
rotation is distinguished.
\item Every commutative square between the first arrows of two distinguished
Weil pair triangles extends to a morphism of triangles.
\item Every composable pair of equivariant morphisms admits an octahedral
diagram whose required faces are distinguished.
\end{enumerate}
\depends{Propositions~\ref{prop:weil-common-stage} and
\ref{prop:weil-cone-independence}; the four Verdier axioms in every
$\mathcal W_M$.}
\end{proposition}

\begin{definition}[$\Qlb$-scalars on strict Weil morphisms]
\label{def:weil-scalars}
For $\alpha\in\Qlb$ and a strict Weil morphism represented by $(M,u)$,
choose $N\supset M,\Ql(\alpha)$ and set
\[
 \alpha[M,u]=[N,\alpha u_N].
\]
\end{definition}

\begin{proposition}[Well-defined strict scalar action]
\label{prop:weil-scalars-well-defined}
Definition~\ref{def:weil-scalars} is independent of the field containing
$\alpha$, of the chosen common stage $N$, and of the representative
$(M,u)$.
\depends{Proposition~\ref{prop:weil-representatives}; the typed
$E$-linearity and transition coherence of
Theorem~\ref{thm:strict-transition-coherence}.}
\end{proposition}

\begin{proposition}[Strict scalar field laws]
\label{prop:weil-scalar-laws}
The strict scalar action satisfies the $\Qlb$ field laws, is additive in
the scalar and the morphism, and is bilinear for composition.  Thus every
strict Weil pair Hom group is a $\Qlb$-vector space and composition is
$\Qlb$-bilinear.
\depends{Proposition~\ref{prop:weil-scalars-well-defined}; the field laws
and bilinearity in a finite stage containing all scalars.}
\end{proposition}

\begin{proposition}[Strict scalars, shifts, and cones]
\label{prop:weil-scalar-triangles}
For every $\alpha\in\Qlb$ and every strict Weil morphism $f$,
\[
 (\alpha f)[1]=\alpha f[1].
\]
If $\alpha\ne0$, multiplication by $\alpha$ identifies a strict cone
triangle of $f$ with a strict cone triangle of $\alpha f$ after the
standard rescaling of one structure arrow.  No cone comparison is asserted
for $\alpha=0$.
\depends{Proposition~\ref{prop:weil-shift-calculus};
Definition~\ref{def:weil-cones}; the corresponding finite-stage rescaling
isomorphism of cone triangles.}
\end{proposition}

\begin{theorem}[Strict Weil pair triangulation]
\label{thm:weil-pair-triangulated}
The category $\PairW$ is an ordinary $\Qlb$-linear triangulated category
with the shifts and distinguished triangles of
Definition~\ref{def:weil-cones}.  Every finite-stage inclusion
$j_E^{\mathrm W}:\mathcal W_E\to\PairW$ is exact and $E$-linear.
\depends{Propositions~\ref{prop:weil-representatives} and
\ref{prop:weil-stage-cocone};
Theorem~\ref{thm:weil-pair-additive}; Propositions~\ref{prop:weil-shift-calculus},
\ref{prop:weil-verdier-axioms}, \ref{prop:weil-scalar-laws}, and
\ref{prop:weil-scalar-triangles}.}
\end{theorem}

\begin{proposition}[The pair-level forgetful functor]
\label{prop:weil-pair-forgetful}
The coherent finite-stage forgetful functors induce an exact
$\Qlb$-linear functor
\[
 \operatorname{For}^{\mathrm{pair}}:\PairW\longrightarrow\Pair.
\]
It sends $(E,K)$ to $(E,\operatorname{For}_E K)$.  A representative
$u:T^{\mathrm W}_{M/E}K\to T^{\mathrm W}_{M/F}L$ is sent to the underlying
map in $\mathcal C_M$, conjugated by the specified comparison isomorphisms
\[
 \operatorname{For}_M T^{\mathrm W}_{M/E}
 \xRightarrow{\sim}T_{M/E}\operatorname{For}_E,
 \qquad
 \operatorname{For}_M T^{\mathrm W}_{M/F}
 \xRightarrow{\sim}T_{M/F}\operatorname{For}_F.
\]
The result is independent of the representative, commutes with shift, and
sends strict Weil cone triangles to their underlying cone triangles.
\depends{Theorems~\ref{thm:strict-transition-coherence} and
\ref{thm:weil-pair-triangulated}; the common-stage representative calculus.}
\end{proposition}

\begin{lemma}[Finite-stage zero-idempotent reflection]
\label{lem:finite-weil-zero-idempotent}
Let $e\in\End_{\mathcal W_M}(K)$ satisfy $e^2=e$.  If
$\operatorname{For}_M(e)=0$, then $e=0$.
\depends{Finite-stage conservativity of $\operatorname{For}_M$; the facts
that $1-e$ is idempotent, an idempotent whose image is the identity is
invertible, and an invertible idempotent is the identity.}
\end{lemma}

\begin{proposition}[Pair-level zero-idempotent reflection]
\label{prop:zero-idempotent-reflection}
For every $(E,K)\in\PairW$ and every idempotent
$e\in\End_{\PairW}(E,K)$,
\[
 \operatorname{For}^{\mathrm{pair}}(e)=0
\quad\Longrightarrow\quad e=0.
\]
\depends{Lemma~\ref{lem:finite-weil-zero-idempotent};
Propositions~\ref{prop:weil-pair-forgetful} and
\ref{prop:weil-common-stage}.}
\end{proposition}

\begin{proposition}[Pair-level forgetful conservativity]
\label{prop:weil-pair-conservative}
The exact functor $\operatorname{For}^{\mathrm{pair}}$ is conservative:
if its value on $f$ is an isomorphism in $\Pair$, then $f$ is an
isomorphism in $\PairW$.
\depends{Propositions~\ref{prop:weil-pair-forgetful},
\ref{prop:weil-common-stage}, and \ref{prop:zero-idempotent-reflection};
finite-stage conservativity and the cone criterion for isomorphisms.}
\end{proposition}

\begin{theorem}[Conservativity after Karoubi completion]
\label{thm:weil-karoubi-conservative}
The Karoubi extension
\[
 \Kar(\operatorname{For}^{\mathrm{pair}}):
 \Kar(\PairW)\longrightarrow\Kar(\Pair)
\]
reflects zero objects and is conservative.  For every morphism $u$ in
$\Kar(\PairW)$,
\[
 \Kar(\operatorname{For}^{\mathrm{pair}})(u)\text{ is an isomorphism}
 \quad\Longrightarrow\quad u\text{ is an isomorphism}.
\]
\depends{Propositions~\ref{prop:weil-pair-forgetful},
\ref{prop:zero-idempotent-reflection}, and
\ref{prop:weil-pair-conservative}; the description of
identities and triangles in the Karoubi envelope; the cone criterion for an
isomorphism in a triangulated category.}
\end{theorem}

\begin{definition}[Exact pseudocone on strict Weil stages]
\label{def:weil-pseudocone}
Let $\mathcal T$ be an ordinary triangulated category.  A strict Weil exact
pseudocone consists of exact functors
\[
 G_E:\mathcal W_E\longrightarrow\mathcal T
\]
with selected shift comparisons
$\theta^G_{E,K}:G_E(K[1])\xrightarrow{\sim}G_E(K)[1]$, and exact natural
isomorphisms
\[
 d_{F/E}:G_FT^{\mathrm W}_{F/E}\xRightarrow{\sim}G_E.
\]
Exactness of $d_{F/E}$ means
\[
 \theta^G_{E,K}\circ d_{F/E,K[1]}
 =d_{F/E,K}[1]\circ\theta^G_{F,T^{\mathrm W}_{F/E}K}
  \circ G_F(\sigma^{\mathrm W}_{F/E,K}).
\]
The unit condition identifies $d_{E/E}$ with the normalized identity, and
for $E\subset F\subset G$ require
\[
 d_{G/E}\circ G_G(a^{\mathrm W}_{G,F,E})
 =d_{F/E}\circ(d_{G/F}T^{\mathrm W}_{F/E}).
\]
A morphism of such pseudocones is a family of exact natural transformations
commuting with every $d_{F/E}$.
\end{definition}

\begin{theorem}[Universal property of the strict Weil pair category]
\label{thm:weil-pair-universal}
Restriction along the $j_E^{\mathrm W}$ gives an equivalence between
\begin{enumerate}
\item exact functors $\PairW\to\mathcal T$ and exact natural
transformations;
\item strict Weil exact pseudocones to $\mathcal T$ and their morphisms.
\end{enumerate}
For a pseudocone $(G_E,d_{F/E})$, the induced functor sends $(E,K)$ to
$G_EK$ and sends a representative
$u:T^{\mathrm W}_{M/E}K\to T^{\mathrm W}_{M/F}L$ to
\[
 d_{M/F,L}\circ G_M(u)\circ d_{M/E,K}^{-1}:
 G_EK\longrightarrow G_FL.
\]
\depends{Definition~\ref{def:weil-pseudocone};
Propositions~\ref{prop:weil-representatives},
\ref{prop:weil-stage-cocone}, and \ref{prop:weil-shift-calculus};
Theorem~\ref{thm:weil-pair-triangulated}.}
\end{theorem}

\begin{theorem}[Universal property of the strict Weil Karoubi colimit]
\label{thm:weil-karoubi-universal}
If $\mathcal T$ is idempotent-complete, restriction along the composites
\[
 \mathcal W_E\xrightarrow{j_E^{\mathrm W}}\PairW
 \longrightarrow\Kar(\PairW)
\]
gives an equivalence between exact functors
$\Kar(\PairW)\to\mathcal T$ and strict Weil exact pseudocones to
$\mathcal T$.  For the pair functor $\widetilde G$ induced above, the
extension sends a triple $(E,K,e)$ to the image of
\[
 \widetilde G(e)\in\End_{\mathcal T}(G_EK).
\]
\depends{Theorem~\ref{thm:weil-pair-universal}; the universal property of
the triangulated Karoubi envelope.}
\end{theorem}

\begin{definition}[Strict algebraically closed Weil category]
\label{def:qbar-weil}
Define
\[
 \operatorname{WeilD}^b_c(X_0,\Qlb)_{\mathrm{2colim}}
 =\Kar(\PairW).
\]
An object is a triple $(E,K,e)$ with $K\in\mathcal W_E$
and $e^2=e$ in the Weil pair category.  A morphism
\[
 u:(E,K,e)\longrightarrow(F,L,d)
\]
is a common-stage equivariant morphism satisfying
\[
 u=d\,u\,e.
\]
The identity is $e$, and distinguished triangles are direct summands of
stagewise strict distinguished triangles.
For every finite $E$, denote the composite stage inclusion by
\[
 J_E^{\mathrm W}:\mathcal W_E\xrightarrow{j_E^{\mathrm W}}\PairW
 \longrightarrow\Kar(\PairW).
\]
\end{definition}

\begin{definition}[Cohomology of a strict Weil retract]
\label{def:weil-retract-cohomology}
Let $(E,K,e)$ be a strict Weil Karoubi triple and represent $e$ at a finite
stage $M$ by
\[
 e_M:T^{\mathrm W}_{M/E}K\longrightarrow T^{\mathrm W}_{M/E}K.
\]
The $t$-exactness of strict scalar extension identifies $H^i(e_M)$ with a
representative of a well-defined idempotent
\[
 H^i_{\mathrm W,pair}(e)\in
 \End_{\mathcal A^{\mathrm W,pair}}(E,H^iK).
\]
Define
\[
 H^i(E,K,e)=
 (E,H^iK,H^i_{\mathrm W,pair}(e))
 \quad\in\Kar(\mathcal A^{\mathrm W,pair}).
\]
\end{definition}

\begin{proposition}[Strict Weil cohomological functors]
\label{prop:weil-retract-cohomology}
Definition~\ref{def:weil-retract-cohomology} is independent of the
representative of $e$ and defines an additive functor for every $i$.  A
distinguished triangle in $\Kar(\PairW)$ gives a long exact sequence in the
abelian category $\Kar(\mathcal A^{\mathrm W,pair})$.
\depends{Theorem~\ref{thm:weil-pair-heart}; $t$-exactness in
Theorem~\ref{thm:strict-transition-coherence}; finite-stage cohomology long
exact sequences; the triangulated Karoubi-envelope theorem.}
\end{proposition}

\begin{theorem}[Strict Weil 2-colimit theorem]
\label{thm:strict-weil-colimit}
The category
$\operatorname{WeilD}^b_c(X_0,\Qlb)_{\mathrm{2colim}}$ is an
idempotent-complete $\Qlb$-linear ordinary triangulated category.  It has
the universal property of Theorem~\ref{thm:weil-karoubi-universal} for
exact pseudocones on the strict finite-stage Weil categories.  The finite-stage
forgetful functors induce an exact functor
\[
 \operatorname{For}_{\Qlb}:
 \operatorname{WeilD}^b_c(X_0,\Qlb)_{\mathrm{2colim}}
 \longrightarrow\Dbc(X,\Qlb)_{\mathrm{2colim}}.
\]
This functor is $\Qlb$-linear and conservative.  For $(E,K,e)$, its $i$th
cohomology object is the object of
Definition~\ref{def:weil-retract-cohomology}.
\depends{Theorem~\ref{thm:weil-pair-triangulated};
Definition~\ref{def:qbar-weil}; the triangulated and universal properties
of Karoubi completion; Theorems~\ref{thm:weil-karoubi-universal} and
\ref{thm:weil-karoubi-conservative};
Proposition~\ref{prop:weil-retract-cohomology}.}
\end{theorem}

\begin{corollary}[The coefficient chain used by Weil theory]
The exact coefficient functors are
\[
 \Dbc(X,\Zl)\longrightarrow\Dbc(X,\Ql)
 \xrightarrow{J_{\Ql}}\Dbc(X,\Qlb)_{\mathrm{2colim}},
\]
where the first arrow is rational localization and Karoubi completion and
the second is the ordinary finite-stage inclusion.  For schemes over
$\mathbf F_q$, the distinct strict Weil stage inclusion is
\[
 \mathcal W_{\Ql}\xrightarrow{\ J^{\mathrm W}_{\Ql}\ }
 \operatorname{WeilD}^b_c(X_0,\Qlb)_{\mathrm{2colim}}.
\]
Together with the finite-stage forgetful functors, these arrows form the
commutative square
\[
\begin{CD}
 \mathcal W_{\Ql} @>{J^{\mathrm W}_{\Ql}}>>
 \operatorname{WeilD}^b_c(X_0,\Qlb)_{\mathrm{2colim}}\\
 @V{\operatorname{For}_{\Ql}}VV @VV{\operatorname{For}_{\Qlb}}V\\
 \mathcal C_{\Ql} @>{J_{\Ql}}>>
 \Dbc(X,\Qlb)_{\mathrm{2colim}}.
\end{CD}
\]
\depends{The finite rationalization theorem; Theorems~\ref{thm:karoubi-triangulated}
and \ref{thm:strict-weil-colimit}.}
\end{corollary}

\section{Reference points}

The finite-stage coefficient categories are those used in the classical
theory of weights and in Fourier--Deligne theory.  The construction above
is the explicit filtered bicategorical colimit of these ordinary
triangulated categories, followed by the ordinary Karoubi envelope.  The
literal comparison isolates the precise Hom-descent and finiteness
hypotheses required to replace this explicit category by modules over a
single sheaf of algebraic coefficients.
