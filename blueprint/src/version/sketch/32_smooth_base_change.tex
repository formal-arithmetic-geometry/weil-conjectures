\begin{center}
{\large\bfseries Smooth Base Change\par}
\medskip
{\normalsize Local acyclicity, smooth exchange, and completed coefficients\par}
\end{center}

\begin{abstract}
This note decomposes smooth base change into definitions, local geometric
statements, and assembly theorems. The finite-torsion result is first
formulated in an ambient bounded-below derived category. Its restriction to
constructible objects is made only after the necessary boundedness and
constructibility statement. The final sections transport the theorem to the
pro-etale, adic, rational, and algebraically closed coefficient categories.
Every category is an ordinary triangulated category.
\end{abstract}

\section{Finite coefficients and derived direct image}

Fix a positive integer $N$ and a finite commutative ring $\Lambda$ annihilated
by $N$. All schemes in the finite-coefficient theorem are schemes on which
$N$ is invertible.

\begin{definition}[Small etale site and coefficient sheaf]
For a scheme $T$, the small etale site $T_{\et}$ has etale morphisms $U\to T$
as objects and jointly surjective etale families as coverings. The sheaf
$\Lambda_T$ assigns to $U$ the locally constant maps $|U|\to\Lambda$. Write
$\Mod(T_{\et},\Lambda)$ for the abelian category of sheaves of
$\Lambda_T$-modules.
\end{definition}

\begin{definition}[Ambient derived categories]
Let $\operatorname{Ch}(T_{\et},\Lambda)$ be the category of cochain complexes
in $\Mod(T_{\et},\Lambda)$, let
$K(T_{\et},\Lambda)$ be its quotient by chain homotopy, and let
$K_{\mathrm{ac}}(T_{\et},\Lambda)$ be the full thick triangulated
subcategory of acyclic complexes. Define the ordinary Verdier quotient
\[
 D(T_{\et},\Lambda)
 =K(T_{\et},\Lambda)/K_{\mathrm{ac}}(T_{\et},\Lambda).
\]
Equivalently, it is the localization of complexes at quasi-isomorphisms.
Its full subcategory $\Dplus(T_{\et},\Lambda)$ consists of objects $K$ for
which $H^i(K)=0$ for all sufficiently small $i$.
\end{definition}

\begin{proposition}[Grothendieck property and adapted resolutions]
The category $\Mod(T_{\et},\Lambda)$ is Grothendieck abelian: it has all
small limits and colimits, filtered colimits are exact, it has a generator,
and it has enough injectives. A complex $I$ is $K$-injective if
\[
 \operatorname{Hom}_{K(T_{\et},\Lambda)}(A,I)=0
\]
for every acyclic complex $A$. Every complex admits a quasi-isomorphism to a
$K$-injective complex. A quasi-isomorphism between $K$-injective complexes is
a homotopy equivalence.
\end{proposition}

\begin{definition}[Constructible sheaves and complexes]
Assume $T$ is noetherian. A sheaf $F\in\Mod(T_{\et},\Lambda)$ is
constructible if there is a finite partition $T=\coprod_{\alpha=1}^rT_\alpha$
by locally closed subschemes such that $F|_{T_\alpha}$ is finite locally
constant. Here finite locally constant means locally constant with finite
stalks, or equivalently trivialized by an etale cover with a finite
$\Lambda$-module as fibre.

The category $\Dbc(T_{\et},\Lambda)$ is the full triangulated subcategory of
$D(T_{\et},\Lambda)$ consisting of complexes $K$ such that every $H^i(K)$ is
constructible and $H^i(K)=0$ outside one finite interval.
\end{definition}

\begin{definition}[Derived pullback and direct image]
For $u:S\to T$, inverse image of sheaves and
$u^{-1}\Lambda_T\cong\Lambda_S$ give an exact functor
\[
 u^*:\Mod(T_{\et},\Lambda)\longrightarrow\Mod(S_{\et},\Lambda).
\]
Its derived functor is denoted $Lu^*$ and is computed termwise.

The direct image
\[
 u_*:\Mod(S_{\et},\Lambda)\longrightarrow\Mod(T_{\et},\Lambda),
 \qquad (u_*F)(V)=F(V\times_TS),
\]
is left exact. Choose a $K$-injective resolution of a representative of $K$
and define
\[
 Ru_*K=u_*I(K),\qquad
 Ru_*:D(S_{\et},\Lambda)\longrightarrow D(T_{\et},\Lambda).
\]
It restricts to $\Dplus(S_{\et},\Lambda)\to\Dplus(T_{\et},\Lambda)$.
There is an adjunction $Lu^*\dashv Ru_*$, with unit $\eta_u$ and counit
$\epsilon_u$. The resulting functor and adjunction are independent, up to a
unique derived natural isomorphism, of the selected resolution functor.
\dep{K-injective resolutions and preservation of homotopy equivalences by
direct image on K-injective complexes.}
\end{definition}

\begin{definition}[Constructible direct-image finiteness]
A morphism $u:S\to T$ has property $\mathrm{CF}_\Lambda$ if:
\begin{enumerate}
\item $R^qu_*F$ is constructible for every constructible $\Lambda$-sheaf
      $F$ and every $q\geq0$;
\item there is an integer $c(u,\Lambda)$ such that $R^qu_*F=0$ for
      $q>c(u,\Lambda)$ and every constructible $F$.
\end{enumerate}
\end{definition}

\begin{proposition}[Restriction of direct image]
If $u$ has property $\mathrm{CF}_\Lambda$, then the ambient functor restricts
to
\[
 Ru_*:\Dbc(S_{\et},\Lambda)\longrightarrow\Dbc(T_{\et},\Lambda).
\]
If $K$ has cohomology in $[a,b]$, then $Ru_*K$ has cohomology in
$[a,b+c(u,\Lambda)]$.
\dep{The spectral sequence $R^pu_*H^q(K)\Rightarrow H^{p+q}(Ru_*K)$ and the
two clauses of $\mathrm{CF}_\Lambda$.}
\end{proposition}

\begin{corollary}[A geometric sufficient case]
Let $k$ be a field of characteristic prime to $N$. If $S,T$ are of finite
type over $k$ and $u:S\to T$ is of finite type, then $u$ has
$\mathrm{CF}_\Lambda$. Thus $Ru_*$ is defined between the displayed bounded
constructible categories.
\dep{Constructibility of higher direct images and finite etale
cohomological dimension over a field.}
\end{corollary}

\section{The smooth square and its exchange map}

\begin{definition}[Smooth Cartesian square]
A smooth Cartesian square is a Cartesian diagram
\[
\begin{CD}
 X' @>{v}>> X\\
 @V{f'}VV @VV{f}V\\
 Y' @>{g}>> Y
\end{CD}
\qquad (X'=X\times_YY')
\]
together with the chosen Cartesian identification and the assertion that
$g$ is smooth. The main ambient theorem assumes that $f$ is quasi-compact
and quasi-separated. Since smoothness is stable under base change, $v$ is
smooth.
\end{definition}

\begin{definition}[Derived exchange morphism]
For any Cartesian square as above, define
\[
 \BC_{f,g}(K):Lg^*Rf_*K\longrightarrow Rf'_*Lv^*K
\]
as the composite
\[
\begin{aligned}
 Lg^*Rf_*K
 &\xrightarrow{\eta_{f'}}Rf'_*Lf'^*Lg^*Rf_*K\\
 &\xrightarrow{\sim}Rf'_*Lv^*Lf^*Rf_*K
 \xrightarrow{Rf'_*Lv^*(\epsilon_f)}Rf'_*Lv^*K.
\end{aligned}
\]
It is a natural transformation on $D(X_{\et},\Lambda)$. For a sheaf $F$,
the degree-$q$ map is
\[
 \BC^q_{f,g}(F):g^*R^qf_*F\longrightarrow R^qf'_*v^*F.
\]
This definition contains no isomorphism assertion.
\end{definition}

\begin{lemma}[Identity and base-change composition]
For $g=\operatorname{id}_Y$, the exchange map is the identity. If
$Y''\xrightarrow{h}Y'\xrightarrow{g}Y$, then, after the canonical
associativity identifications, the exchange for $g\circ h$ is
\[
 Lh^*Lg^*Rf_*
 \xrightarrow{Lh^*\BC_{f,g}}Lh^*Rf'_*Lv^*
 \xrightarrow{\BC_{f',h}}Rf''_*L(v\circ v')^*.
\]
\dep{The two triangle identities for the pullback--direct-image
adjunctions.}
\end{lemma}

\begin{lemma}[Direct-image composition]
For $X\xrightarrow{f}Y\xrightarrow{p}Z$ quasi-compact and
quasi-separated, the exchange map for $pf$ agrees, under
$R(pf)_*\cong Rp_*Rf_*$, with the composite of the exchange maps for $p$
and $f$.
\dep{Pseudofunctoriality of derived direct image and naturality of the
unit--counit construction.}
\end{lemma}

\section{Strict local fibres and local acyclicity}

\begin{definition}[Strict localization and Milnor tube]
Let $a:Z\to S$, let $\bar z\to Z$ be geometric, and write
$\bar s=a(\bar z)$. Put
\[
 Z_{(\bar z)}=\Spec(\mathcal O^{\sh}_{Z,\bar z}),\qquad
 S_{(\bar s)}=\Spec(\mathcal O^{\sh}_{S,\bar s}).
\]
For a geometric point $\bar t\to S_{(\bar s)}$, the Milnor tube, or variety
of vanishing cycles, is
\[
 M_a(\bar z,\bar t)
 =\bar t\times_{S_{(\bar s)}}Z_{(\bar z)}.
\]
For $K\in D(Z_{\et},\Lambda)$, restriction induces
\[
 \operatorname{sp}_{a,K,\bar z,\bar t}:
 K_{\bar z}=R\Gamma(Z_{(\bar z)},K)
 \longrightarrow R\Gamma(M_a(\bar z,\bar t),K).
\]
\end{definition}

\begin{definition}[Local $n$-acyclicity]
Let $n\in\mathbf Z$. The morphism $a$ is locally $n$-acyclic at $\bar z$
relative to $K$ if, for every $\bar t\to S_{(\bar s)}$, the specialization
map is an isomorphism on $H^q$ for $q\leq n$ and is injective on
$H^{n+1}$. It is locally $n$-acyclic relative to $K$ if this holds at every
geometric point. It is locally acyclic relative to $K$ if the untruncated
specialization map is an isomorphism for every $(\bar z,\bar t)$.
\end{definition}

\begin{definition}[Universal local acyclicity]
The morphism $a:Z\to S$ is universally locally $n$-acyclic relative to $K$
if, for every $S'\to S$, the base change
$Z\times_SS'\to S'$ is locally $n$-acyclic relative to the pullback of $K$.
Universal local acyclicity, abbreviated ULA, is the same condition without
truncation.
\end{definition}

\begin{definition}[Allowed residue primes]
Let $\mathcal P$ be a set of primes invertible on $S$. A finite
$\mathcal P$-group is a finite abelian group whose prime divisors lie in
$\mathcal P$. The morphism $a$ is $\mathcal P$-locally $n$-acyclic if it is
locally $n$-acyclic relative to every constant sheaf $\underline A_Z$ with
$A$ a finite $\mathcal P$-group. The universal and untruncated versions are
defined in the same way.
\end{definition}

\begin{definition}[Degree-one asphericity]
An ind-$\mathcal P$-group sheaf is a filtered union of sheaves of finite
groups whose orders have prime divisors in $\mathcal P$. For a sheaf of
groups $G$, let $R^1w_*G$ denote the pointed sheaf of isomorphism classes of
$G$-torsors along $w$.

The morphism $a:Z\to S$ is locally $1$-aspherical for $\mathcal P$ if the
following condition holds after replacing $a$ by the induced morphism
$Z_{(\bar z)}\to S_{(a(\bar z))}$ for every geometric $\bar z\to Z$. For
every etale $i:S'\to S$, every etale morphism of finite presentation
$i':S''\to S'$, and the induced Cartesian diagram
\[
\begin{CD}
 Z'' @>{j'}>> Z'\\
 @V{a''}VV @VV{a'}V\\
 S'' @>{i'}>> S',
\end{CD}
\]
the base-change maps
\[
 (a')^*R^qi'_*G\longrightarrow R^qj'_*(a'')^*G
\]
are isomorphisms for $q=0,1$ and every ind-$\mathcal P$-group sheaf $G$ on
$S''$. Universally locally $1$-aspherical means that this condition holds
after every base change on $S$.
\end{definition}

\begin{lemma}[Cohomological form for constant coefficients]
For a finite $\mathcal P$-group $A$, local acyclicity relative to
$\underline A_Z$ is equivalent to
\[
 H^q(M_a(\bar z,\bar t),A)=
 \begin{cases}A,&q=0,\\0,&q\ne0,\end{cases}
\]
for every $(\bar z,\bar t)$, with the degree-zero identification induced by
constant sections.
\dep{The specialization map and cohomology of the strictly henselian local
scheme $Z_{(\bar z)}$.}
\end{lemma}

\begin{lemma}[Stability and locality]
The following hold for $\mathcal P$-local $n$-acyclicity, local
$1$-asphericity, and local acyclicity.
\begin{enumerate}
\item Their universal versions are stable under arbitrary base change.
\item They are local for the etale topology on source and target.
\item A composition of universally locally acyclic morphisms is universally
      locally acyclic.
\item An etale morphism is universally locally acyclic.
\end{enumerate}
\dep{Strict-henselization base change, the Cartesian description of Milnor
tubes, and the Leray spectral sequence for a composite.}
\end{lemma}

\section{The open-immersion theorem}

\begin{definition}[Open-immersion exchange]
Let $j:U\hookrightarrow Z$ be a quasi-compact open immersion and let
$a:Z'\to Z$. Form the Cartesian square
\[
\begin{CD}
 U' @>{a_U}>> U\\
 @V{j'}VV @VV{j}V\\
 Z' @>{a}>> Z.
\end{CD}
\]
The open-immersion exchange is
\[
 \BC_{j,a}(F):La^*Rj_*F\longrightarrow Rj'_*La_U^*F.
\]
\end{definition}

\begin{theorem}[Truncated open-immersion base change]
Assume $a$ is universally $\mathcal P$-locally $n$-acyclic. Let $F$ be a
sheaf on $U_{\et}$ whose geometric stalks are torsion $\mathcal P$-groups.
Then
\[
 a^*R^qj_*F\xrightarrow{\ \sim\ }R^qj'_*a_U^*F
\]
for $0\leq q\leq n$, and the same base-change map is injective for
$q=n+1$. These maps are natural in $F$ and compatible with composition of
open immersions and with further base change.
\dep{The strict-local stalk formula for $R^qj_*$, the Milnor-tube criterion,
and geometric-stalk detection.}
\end{theorem}

\begin{corollary}[Derived open-immersion base change]
If $a$ is universally $\mathcal P$-locally acyclic and
$K\in\Dplus(U_{\et})$ has $\mathcal P$-torsion cohomology, then
\[
 La^*Rj_*K\xrightarrow{\ \sim\ }Rj'_*La_U^*K
\]
in $\Dplus(Z'_{\et})$.
\dep{Truncated open-immersion base change in every degree and bounded-below
truncation.}
\end{corollary}

\begin{proposition}[Extension by zero and arbitrary base change]
For the same square, extension by zero is exact and has a canonical natural
isomorphism
\[
 a^*j_!F\xrightarrow{\ \sim\ }j'_!a_U^*F.
\]
The same formula holds for complexes. No local-acyclicity hypothesis is
needed.
\dep{The sectionwise definition of $j_!$ and $U'=U\times_ZZ'$.}
\end{proposition}

\section{From open immersions to arbitrary qcqs direct images}

\begin{definition}[Quasi-projective factorization datum]
Let $f:X\to Y$ be quasi-projective and separated. A factorization datum is
an open immersion $j:X\hookrightarrow\overline X$ and a projective morphism
$p:\overline X\to Y$ satisfying $f=pj$. After $Y'\to Y$, write $j'$, $p'$,
and $\overline g:\overline X'\to\overline X$ for their base changes.
\end{definition}

\begin{lemma}[Quasi-projective direct image]
Let $g:Y'\to Y$ be universally $\mathcal P$-locally acyclic. If proper base
change holds for $p$ and derived open-immersion base change holds for $j$,
then
\[
 Lg^*Rf_*K\xrightarrow{\ \sim\ }Rf'_*Lv^*K
\]
for every bounded-below complex $K$ with $\mathcal P$-torsion cohomology.
\dep{$Rf_*\cong Rp_*Rj_*$, proper base change, open-immersion base change,
and exchange compatibility with composite direct images.}
\end{lemma}

\begin{definition}[Finite affine Cech nerve]
Let $f:X\to Y$ be quasi-compact and quasi-separated and choose a finite
affine cover $X=\bigcup_{i=1}^rU_i$. For $p\geq0$, put
\[
 X_p=\coprod_{(i_0,\ldots,i_p)}
 U_{i_0}\cap\cdots\cap U_{i_p},
\]
with its face maps, degeneracy maps, and augmentation
$a_\bullet:X_\bullet\to X$. Write $f_p=f\circ a_p$.
\end{definition}

\begin{theorem}[Finite Cech cohomological descent]
For a torsion sheaf $F$ on $X_{\et}$, the augmented cosimplicial complex
associated with the finite affine Cech nerve resolves $F$ and gives
\[
 E_1^{p,q}=R^q(f_p)_*(F|_{X_p})
 \Longrightarrow R^{p+q}f_*F.
\]
In each fixed total degree the induced filtration is finite. The
construction commutes with base change and carries the termwise exchange
maps to the exchange map on the abutment.
\dep{Exactness of the finite Cech resolution, quasi-separatedness for the
intersections, and the spectral sequence of a filtered total complex.}
\end{theorem}

\begin{lemma}[Finite affine refinement in bounded degree]
Each intersection $U_{i_0}\cap\cdots\cap U_{i_p}$ is quasi-compact, hence has
a finite affine open covering. For every integer $n\geq0$, choosing these
covers for the intersections with $p\leq n+1$ and taking their Cech
resolutions gives a finite multicomplex of affine schemes which computes
$R^qf_*F$ for $q\leq n$. Its construction can be made simultaneously after
base change, and its comparison map is induced by the exchange maps on the
affine pieces.
\dep{Quasi-separatedness of $X$, finite affine covers of quasi-compact open
subschemes, bounded-degree Cech descent, and comparison of the associated
finite filtrations.}
\end{lemma}

\begin{lemma}[Strict-local reduction]
To show that $\BC^q_{f,g}(F)$ is an isomorphism, it is enough to pull it back
to every strict localization $Y'_{(\bar y')}$ at a geometric point of $Y'$.
There it is the restriction map on cohomology of the corresponding
strict-local base changes of $X$.
\dep{Conservativity of geometric stalks and continuity over pointed etale
neighbourhoods.}
\end{lemma}

\begin{theorem}[Affine finite-type approximation]
Let $Y$ be affine and $X\to Y$ affine and quasi-compact. There is a
cofiltered system of affine schemes $X_\lambda$ of finite presentation over
$Y$, with affine transition maps, such that
\[
 X\cong\varprojlim_\lambda X_\lambda.
\]
Every finitely presented etale object, finite locally constant sheaf, and
constructible sheaf on $X$ descends to $X_\lambda$ for all sufficiently
large $\lambda$, including all selected morphisms and cocycle identities.
\dep{Approximation of finitely presented algebras and finite-presentation
descent for etale morphisms and finite sheaves.}
\end{theorem}

\begin{theorem}[Continuity of cohomology and exchange]
If $F$ descends to $F_\lambda$ on $X_\lambda$, then
\[
 \varinjlim_{\mu\geq\lambda}H^q(X_\mu,F_\mu)
 \xrightarrow{\ \sim\ }H^q(X,F)
\]
for every $q$. These isomorphisms are natural and identify the filtered
colimit of finite-stage exchange maps with the exchange map on the limit.
\dep{Continuity of the small etale topos for inverse limits with affine
transition maps and exactness of filtered colimits.}
\end{theorem}

\begin{proposition}[Extension from finite type to qcqs]
Suppose smooth base change is known for quasi-projective morphisms of finite
presentation. Then it holds for every quasi-compact and quasi-separated
$f:X\to Y$ and every finite constructible $\Lambda$-sheaf on $X$.
\dep{Strict-local reduction, the finite affine refinement of the Cech nerve
through each required degree, affine finite-type approximation on every
affine piece, continuity of exchange, and comparison of the resulting
finite-range spectral sequences.}
\end{proposition}

\section{Why smooth morphisms supply local acyclicity}

\begin{theorem}[Etale-local normal form]
Let $g:Y'\to Y$ be smooth at $y'\in Y'$. After replacing $Y,Y'$ by affine
neighbourhoods, there are $d\geq0$ and an etale morphism
\[
 e:Y'\longrightarrow\mathbf A^d_Y
\]
such that $g$ is $e$ followed by the projection
$p_d:\mathbf A^d_Y\to Y$.
\dep{The Jacobian criterion for a standard smooth algebra and the fact that
a smooth morphism of relative dimension zero is etale.}
\end{theorem}

\begin{definition}[Universal affine-curve test datum]
A universal affine-curve test datum consists of:
\begin{enumerate}
\item an affine normal integral scheme $S=\Spec(A)$ whose fraction field
      $L$ is algebraically closed;
\item an algebraically closed extension $K/L$ and the dominant map
      $\Spec(K)\to S$;
\item an affine smooth morphism $C\to S$ of relative dimension one;
\item an integer $d$ invertible on $S$ and the constant sheaf
      $\mathbf Z/d\mathbf Z$ on $C_K$.
\end{enumerate}
For an etale $U\to C$, put
\[
 \mathcal H^q_K(U)=H^q(U_K,\mathbf Z/d\mathbf Z).
\]
\end{definition}

\begin{lemma}[Locality of the affine test]
For an affine smooth $u:V\to S$, a qcqs map $h:T\to S$, and a torsion sheaf
$F$ on $T$, the assertion
\[
 u^*R^qh_*F\longrightarrow R^qh_{V,*}(F|_{T\times_SV})
 \quad\text{is an isomorphism}
\]
is local on $V$ for the etale topology, local on $S$ for the Zariski
topology, and compatible with a composite of smooth maps.
\dep{Localization of the small etale site, geometric-stalk conservativity,
and composition coherence of exchange.}
\end{lemma}

\begin{lemma}[Reduction to relative dimension one]
If the affine smooth base-change assertion holds for every affine smooth
morphism of relative dimension one, then it holds for every smooth
morphism.
\dep{The etale-local normal form, the factorization
$\mathbf A^d_S\to\mathbf A^{d-1}_S\to\cdots\to\mathbf A^1_S\to S$,
locality of the affine test, and composition coherence.}
\end{lemma}

\begin{lemma}[Filtered torsion reduction]
Let $h:T\to S$ be quasi-compact and quasi-separated and let $F$ be a sheaf
whose geometric stalks are torsion of orders invertible on $S$. If
$F[m]$ denotes the subsheaf killed by $m$, with $m$ ranging through the
integers invertible on $S$, then
\[
 F=\varinjlim_m F[m].
\]
Inverse image commutes with this filtered colimit, and for every $q$ the
canonical map
\[
 \varinjlim_m R^qh_*F[m]\xrightarrow{\ \sim\ }R^qh_*F
\]
is an isomorphism.
\dep{The stalkwise torsion hypothesis, exactness of filtered colimits of
sheaves, and commutation of higher direct image with filtered colimits for a
qcqs morphism of schemes on the small etale site.}
\end{lemma}

\begin{theorem}[Universal-curve failure criterion]
Fix $q\geq1$ and suppose smooth base change is known in degrees $<q$. If it
fails in degree $q$ for an affine smooth morphism of relative dimension one,
then there is a universal affine-curve test datum for which the sheafification
of
\[
 U\longmapsto\mathcal H^q_K(U)
\]
is nonzero.
\dep{Reduction to a sheaf killed by one integer, finite-type approximation,
normalization, passage to an algebraically closed fraction field, and the
strict-local description of $R^qh_*$.}
\end{theorem}

\begin{lemma}[Degree-zero flat base change]
Let $u:V\to S$ be flat and locally of finite presentation with geometrically
reduced fibres; in particular, $u$ may be smooth. For every morphism
$h:T\to S$ and every sheaf $F$ on $T_{\et}$, the degree-zero exchange map
\[
 u^*h_*F\longrightarrow h_{V,*}(F|_{T\times_SV})
\]
is an isomorphism.
\dep{The description of connected components after flat finitely presented
base change with geometrically reduced fibres, and the section equalizer
defining direct image.}
\end{lemma}

\begin{theorem}[Cohomological dimension of an affine curve]
Let $C$ be a separated affine scheme of finite type and dimension at most
one over an algebraically closed field. For every torsion etale sheaf $F$,
\[
 H^q(C_{\et},F)=0\qquad(q>1).
\]
\end{theorem}

\begin{theorem}[Algebraically closed field-extension invariance]
Let $K/L$ be an extension of algebraically closed fields, let $C$ be a
separated finite-type $L$-scheme of dimension at most one, and let $F$ be a
constructible torsion sheaf whose torsion orders are prime to
$\operatorname{char}(L)$. Then
\[
 H^q(C,F)\xrightarrow{\ \sim\ }H^q(C_K,F|_{C_K})
\]
for every $q\geq0$.
\end{theorem}

\begin{lemma}[Sectioned connected curve reduction]
Let $C\to S$ be affine and smooth of relative dimension one. There is an
etale covering $(U_i\to C)_i$ such that each composite $U_i\to S$ factors as
\[
 U_i\longrightarrow V_i\longrightarrow S,
\]
where $V_i\to S$ is etale and $U_i\to V_i$ is an affine smooth morphism with
geometrically connected fibres and a section $V_i\to U_i$.
\dep{Etale localization of a smooth morphism with a chosen point and the
open representability of the relative connected-component functor.}
\end{lemma}

\begin{theorem}[Spreading a finite etale Galois cover]
Let $S$ be a quasi-compact and quasi-separated integral normal scheme with
generic point $\eta$. Let $C\to S$ be a quasi-compact and quasi-separated
smooth morphism with geometrically connected fibres, and fix a section
$\sigma:S\to C$. Let $Z\to C_\eta$ be a finite etale Galois cover with group
$G$, assume $|G|$ is invertible on $S$, and assume $Z$ has a
$\kappa(\eta)$-rational point above $\sigma(\eta)$. Then there is a finite
etale Galois cover $\widetilde Z\to C$ with group $G$ whose restriction to
$C_\eta$ is the given cover $Z\to C_\eta$.
\dep{The tame extension theorem over a normal noetherian base and its limit
extension to a qcqs normal base.}
\end{theorem}

\begin{lemma}[Terminal vanishing for the curve test]
For every universal affine-curve test datum and every $q>0$, each local
section of $U\mapsto\mathcal H^q_K(U)$ becomes zero after an etale covering
of $U$. Hence the associated sheaf is zero.
\dep{Affine-curve cohomological dimension for $q>1$; for $q=1$, the
sectioned connected curve reduction, field-extension invariance,
interpretation by finite torsors, and spreading of the resulting finite
etale Galois cover.}
\end{lemma}

\begin{theorem}[Relative-dimension-one affine smooth base change]
Let $u:V\to S$ be affine and smooth of relative dimension one, let
$h:T\to S$ be quasi-compact and quasi-separated, and let $F$ be a sheaf on
$T_{\et}$ whose stalks are torsion of orders invertible on $S$. In the
induced Cartesian square, the exchange map
\[
 u^*R^qh_*F\xrightarrow{\ \sim\ }
 R^qh_{V,*}(F|_{T\times_SV})
\]
is an isomorphism for every $q\geq0$.
\dep{Degree-zero flat base change and induction on $q$: a first
failure in degree $q$ produces a nonzero universal-curve failure sheaf,
whereas terminal vanishing makes that sheaf zero.}
\end{theorem}

\begin{lemma}[From affine-line exchange to Milnor tubes]
Suppose relative-dimension-one affine smooth base change holds after every
base change on $S$. For $p:\mathbf A^1_S\to S$, every geometric point
$\bar x$, and every $\bar t\to S_{(p(\bar x))}$, the corresponding exchange
isomorphism and strict-henselian stalk formula identify
\[
 H^q(M_p(\bar x,\bar t),A)
 \quad\text{with}\quad
 \begin{cases}A,&q=0,\\0,&q>0,\end{cases}
\]
for each finite abelian group $A$ of order invertible on $S$.
\dep{Relative-dimension-one affine smooth base change, the strict-local
stalk formula, and vanishing of higher cohomology on a strictly henselian
local scheme for a constant finite sheaf.}
\end{lemma}

\begin{theorem}[Affine-line local acyclicity]
Let $p:\mathbf A^1_S\to S$ and let $\mathcal P$ be a set of primes
invertible on $S$. Then $p$ is universally $\mathcal P$-locally acyclic.
Equivalently, every Milnor tube of every base change of $p$ has constant
$A$-cohomology $A$ in degree zero and zero cohomology in positive degrees,
for every finite $\mathcal P$-group $A$.
\dep{Relative-dimension-one affine smooth base change and the
exchange-to-Milnor-tube lemma, uniformly after arbitrary base change.}
\end{theorem}

\begin{corollary}[Affine-space local acyclicity]
For every $d\geq0$, the projection $\mathbf A^d_S\to S$ is universally
$\mathcal P$-locally acyclic.
\dep{Affine-line local acyclicity and stability under composition and base
change.}
\end{corollary}

\begin{theorem}[Smooth morphisms are universally locally acyclic]
Every smooth morphism $g:Y'\to Y$ is universally
$\mathcal P$-locally acyclic for every set $\mathcal P$ of primes invertible
on $Y$.
\dep{The etale-local normal form, affine-space local acyclicity, etale local
acyclicity, and etale descent of the Milnor-tube condition.}
\end{theorem}

\section{Finite-torsion smooth base change for sheaves}

\begin{theorem}[Quasi-projective finite-torsion theorem]
In the smooth Cartesian square, suppose $f$ is quasi-projective and
separated. Let $F$ be an abelian sheaf on $X_{\et}$ whose geometric stalks
are torsion of orders invertible on $Y$. Then
\[
 g^*R^qf_*F\xrightarrow{\ \sim\ }R^qf'_*v^*F
\]
for every $q\geq0$.
\dep{Smooth ULA, a factorization $f=pj$, open-immersion base change for
$j$, proper base change for $p$, and exchange compatibility with $pj$.}
\end{theorem}

\begin{theorem}[Ambient finite-torsion smooth base change]
Let
\[
\begin{CD}
 X' @>{v}>> X\\
 @V{f'}VV @VV{f}V\\
 Y' @>{g}>> Y
\end{CD}
\]
be Cartesian, with $f$ quasi-compact and quasi-separated and $g$ smooth. If
$F$ is an abelian sheaf on $X_{\et}$ whose geometric stalks are torsion of
orders invertible on $Y$, then
\[
 \BC^q_{f,g}(F):g^*R^qf_*F
 \xrightarrow{\ \sim\ }R^qf'_*v^*F
\]
for every $q\geq0$.
\dep{Filtered torsion reduction, etale and Zariski locality of the affine
test, the smooth local normal form, composition coherence, reduction to
relative dimension one, and relative-dimension-one affine smooth base
change. The quasi-projective compactification and Cech-continuity chain gives
an independent derivation for finite constructible coefficients.}
\end{theorem}

\begin{definition}[Strict-henselian stalk form]
Let $\bar y'\to Y'$ be geometric and let $\bar y$ be its image in $Y$. Put
\[
 X_{(\bar y)}=X\times_YY_{(\bar y)},\qquad
 X'_{(\bar y')}=X'\times_{Y'}Y'_{(\bar y')}.
\]
The stalk of $\BC^q_{f,g}(F)$ at $\bar y'$ is the canonical restriction map
\[
 H^q(X_{(\bar y)},F)
 \longrightarrow H^q(X'_{(\bar y')},v^*F).
\]
\end{definition}

\begin{corollary}[Finite locally constant stalk calculation]
If $L$ is finite locally constant on $X$, the preceding strict-henselian map
is an isomorphism for every $q$. Since geometric stalks are conservative,
these maps recover the global smooth-base-change isomorphism for $L$.
\dep{The strict-local stalk formula and ambient finite-torsion smooth base
change.}
\end{corollary}

\section{Constructible devissage and the derived theorem}

\begin{lemma}[Finite stratification filtration]
Let $F$ be a constructible $\Lambda$-sheaf on a noetherian $X$. After
choosing a finite stratification on which $F$ is finite locally constant,
there is a finite filtration
\[
 0=F_{-1}\subset F_0\subset\cdots\subset F_r=F
\]
whose successive quotients are extensions by zero of finite locally
constant sheaves on individual locally closed strata.
\dep{An ordering of the strata by closed unions and the exact open--closed
localization sequence.}
\end{lemma}

\begin{lemma}[Exchange and short exact sequences]
A short exact sequence $0\to F_1\to F_2\to F_3\to0$ gives a morphism between
the long exact sequences obtained from $g^*R^qf_*$ and
$R^qf'_*v^*$. Every connecting square commutes.
\dep{The distinguished triangle of a short exact sequence, exactness of
pullback, triangulatedness of derived direct image, and naturality of
$\BC_{f,g}$.}
\end{lemma}

\begin{proposition}[Smooth base change for constructible sheaves]
For every constructible $\Lambda$-sheaf $F$ on $X$, the map
\[
 g^*R^qf_*F\xrightarrow{\ \sim\ }R^qf'_*v^*F
\]
is an isomorphism for every $q$.
\dep{Finite stratification, finite locally constant smooth base change,
base change for extension by zero, compatibility with short exact
sequences, and the five lemma.}
\end{proposition}

\begin{definition}[Derived categories with allowable torsion]
Let $\mathcal P$ be the set of primes invertible on $Y$. Define
\[
 D^b_{\mathcal P\text{-tor}}(T_{\et}),\qquad
 D^+_{\mathcal P\text{-tor}}(T_{\et})
\]
as the full triangulated subcategories of the derived category of abelian
sheaves on $T_{\et}$ consisting respectively of bounded and bounded-below
complexes $K$ such that every geometric stalk of every $H^q(K)$ is a torsion
group whose prime divisors belong to $\mathcal P$. The fixed-coefficient
categories with finite $\Lambda$ have exact restriction-of-scalars functors
to these allowable-torsion categories. These functors need not be fully
faithful on derived Hom groups, so the fixed-coefficient and abelian-sheaf
statements are kept as distinct compatible formulations.
\end{definition}

\begin{definition}[Hypercohomology comparison]
For $K\in\Dbc(X_{\et},\Lambda)$, the standard $t$-structure gives
\[
 E_2^{p,q}=R^pf_*H^q(K)\Longrightarrow H^{p+q}(Rf_*K)
\]
and the corresponding sequence after base change. Exact pullback and the
exchange transformation give a morphism between these spectral sequences.
\end{definition}

\begin{lemma}[Finite-range comparison]
If $H^q(K)=0$ outside $[a,b]$ and the comparison on every $E_2^{p,q}$ is an
isomorphism, then the comparison on every abutment $H^n$ is an isomorphism.
\dep{For fixed $n$, the abutment filtration has only finitely many nonzero
successive quotients.}
\end{lemma}

\begin{theorem}[Bounded derived smooth base change]
For every $K\in\Dbc(X_{\et},\Lambda)$,
\[
 \BC_{f,g}(K):Lg^*Rf_*K
 \xrightarrow{\ \sim\ }Rf'_*Lv^*K
\]
in the ambient category $D(Y'_{\et},\Lambda)$.
\dep{Constructible-sheaf smooth base change on the $E_2$ page, exactness of
pullback, and finite-range comparison.}
\end{theorem}

\begin{theorem}[Bounded allowable-torsion derived theorem]
For every $K\in D^b_{\mathcal P\text{-tor}}(X_{\et})$, the exchange map
\[
 Lg^*Rf_*K\xrightarrow{\ \sim\ }Rf'_*Lv^*K
\]
is an isomorphism in $D^b_{\mathcal P\text{-tor}}(Y'_{\et})$ whenever both
sides are bounded, and is always an isomorphism in the ambient derived
category of abelian sheaves.
\dep{Ambient smooth base change for every allowable-torsion cohomology
sheaf, the bounded hypercohomology spectral sequences, and finite-range
comparison.}
\end{theorem}

\begin{lemma}[Bounded-below truncation criterion]
Let $K\in D^+_{\mathcal P\text{-tor}}(X_{\et})$. For fixed $n$, only $H^q(K)$ with
$q\leq n$ can contribute to $H^n(Rf_*K)$. Hence
\[
 H^n(Rf_*\tau_{\leq n}K)\xrightarrow{\ \sim\ }H^n(Rf_*K),
\]
and the analogous statement holds after pullback and for $f'$.
\dep{Left $t$-exactness of a right derived functor and the bounded-below
hypercohomology spectral sequence.}
\end{lemma}

\begin{theorem}[Bounded-below derived smooth base change]
Let $K\in D^+_{\mathcal P\text{-tor}}(X_{\et})$. Then
\[
 \BC_{f,g}(K):Lg^*Rf_*K
 \xrightarrow{\ \sim\ }Rf'_*Lv^*K
\]
in $D^+_{\mathcal P\text{-tor}}(Y'_{\et})$.
\dep{The bounded allowable-torsion theorem for $\tau_{\leq n}K$ in each degree, the
truncation criterion, and detection on cohomology sheaves.}
\end{theorem}

\begin{corollary}[The theorem in bounded constructible categories]
Assume $X,Y,X',Y'$ are noetherian and $f,f'$ have
$\mathrm{CF}_\Lambda$. Then
\[
 Lg^*Rf_*\xRightarrow{\ \sim\ }Rf'_*Lv^*
\]
is a natural isomorphism of exact functors
\[
 \Dbc(X_{\et},\Lambda)\longrightarrow\Dbc(Y'_{\et},\Lambda).
\]
\dep{Preservation of bounded constructibility by pullback and direct image,
together with the ambient derived theorem.}
\end{corollary}

\section{Finite etale--pro-etale comparison}

Throughout this section $T$ and the schemes mapping to it are noetherian, so
constructibility may be expressed by one finite global stratification. For a
non-noetherian scheme the corresponding definition is imposed separately on
each affine open.

\begin{definition}[Comparison morphism of topoi]
For a scheme $T$, let $T_{\proet}$ be the small pro-etale site and let
\[
 \nu_T:T_{\proet}\longrightarrow T_{\et}
\]
be the canonical morphism of topoi. For finite discrete $\Lambda$,
$\nu_T^*\Lambda_{T,\et}\cong\Lambda_{T,\proet}$, and $\nu_T^*$ is exact. A
pro-etale $\Lambda$-sheaf is constructible if a finite locally closed
stratification makes it finite locally constant.
\end{definition}

\begin{theorem}[Finite-lisse descent]
Every finite locally constant $\Lambda$-sheaf on $T_{\proet}$ is isomorphic
to $\nu_T^*L$ for a finite locally constant sheaf $L$ on $T_{\et}$. The
descent is unique up to unique isomorphism and respects morphisms,
compositions, and finite etale trivializations.
\dep{Finite-stage factorization of a pro-etale trivialization, effective
descent on pairwise overlaps, and the triple-overlap cocycle.}
\end{theorem}

\begin{lemma}[Full faithfulness on finite-lisse sheaves]
For finite locally constant etale sheaves $A,B$, pullback induces
\[
 \operatorname{Hom}_{T_{\et}}(A,B)
 \xrightarrow{\ \sim\ }
 \operatorname{Hom}_{T_{\proet}}(\nu_T^*A,\nu_T^*B).
\]
\dep{Finite-lisse descent of morphisms and uniqueness on a common finite
etale trivialization.}
\end{lemma}

\begin{lemma}[Open and closed comparison]
For a complementary pair $j:U\hookrightarrow T$, $i:Z\hookrightarrow T$,
the canonical maps
\[
 \nu_T^*j_!A\xrightarrow{\ \sim\ }j_!\nu_U^*A,
 \qquad
 \nu_T^*i_*B\xrightarrow{\ \sim\ }i_*\nu_Z^*B
\]
are isomorphisms for finite constructible sheaves and are compatible with
composition of immersions.
\dep{The sectionwise description of extension by zero and geometric-stalk
detection on both sites.}
\end{lemma}

\begin{proposition}[Constructible-heart equivalence]
The exact functor $\nu_T^*$ is an equivalence between constructible finite
$\Lambda$-sheaves on $T_{\et}$ and on $T_{\proet}$.
\dep{Finite-lisse descent and full faithfulness on one stratum, open and
closed comparison, and induction on the number of strata.}
\end{proposition}

\begin{theorem}[Derived finite-coefficient comparison]
The exact pullback induces an equivalence of ordinary triangulated
categories
\[
 \nu_T^*:\Dbc(T_{\et},\Lambda)
 \xrightarrow{\ \sim\ }\Dbc(T_{\proet},\Lambda).
\]
It commutes with truncations and derived pullback. For a qcqs morphism
$u:S\to T$ and $K\in D^+(S_{\et},\Lambda)$ with constructible cohomology, the
canonical map
\[
 \nu_T^*Ru_{\et,*}K
 \xrightarrow{\ \sim\ }
 Ru_{\proet,*}\nu_S^*K
\]
is an isomorphism whenever both direct images are defined in the displayed
bounded-below categories.
\dep{The constructible-heart equivalence; the finite-coefficient unit
$K\to R\nu_{T,*}\nu_T^*K$; identification of the constructible essential
image as a weak Serre subcategory; the resulting bounded derived
equivalence; and the qcqs finite-coefficient direct-image comparison.}
\end{theorem}

\begin{corollary}[Finite pro-etale smooth base change]
Conjugating the etale exchange isomorphism by the preceding equivalences
gives
\[
 Lg^*Rf_*K\xrightarrow{\ \sim\ }Rf'_*Lv^*K
\]
for $K\in\Dbc(X_{\proet},\Lambda)$ whenever the finiteness hypotheses needed
to remain in $\Dbc$ hold.
\dep{Derived finite-coefficient comparison and its compatibility with all
four functors in the square.}
\end{corollary}

\section{Adic, rational, and algebraically closed coefficients}

\begin{definition}[Tower category and derived inverse limit]
Let $A_T$ be a sheaf of rings on $T_{\proet}$.  Define
\[
 \mathsf{Tow}(T,A_T)
 =\operatorname{Fun}\bigl(\mathbf N^{\mathrm{op}},
          \operatorname{Mod}(T_{\proet},A_T)\bigr),
\]
where $\mathbf N$ has one arrow $n\to m$ when $n\leq m$.  Thus an object is a
sequence $M_0\leftarrow M_1\leftarrow M_2\leftarrow\cdots$ of $A_T$-module
sheaves.  This is a Grothendieck abelian category.  Its inverse-limit functor
\[
 \varprojlim:\mathsf{Tow}(T,A_T)
 \longrightarrow\operatorname{Mod}(T_{\proet},A_T)
\]
is left exact and is right adjoint to the exact constant-tower functor.
Consequently it preserves injectives.  Using a $K$-injective resolution in
$\mathsf{Tow}(T,A_T)$ defines the ordinary derived functor
\[
 \Rlim:D\bigl(\mathsf{Tow}(T,A_T)\bigr)
 \longrightarrow D(T_{\proet},A_T).
\]
A strict tower of complexes is a complex in $\mathsf{Tow}(T,A_T)$ and hence is
an input to this functor.  Every tower below is the strict tower produced by
the coefficient-reduction maps on chosen $K$-flat or $K$-injective models;
changing the models gives a canonically isomorphic derived inverse limit.
\end{definition}

\begin{definition}[Derived-complete adic category]
Let $A$ be noetherian, complete and separated for an ideal $I$, put
$A_n=A/I^{n+1}$, and assume every $A_n$ is finite of order invertible on the
schemes under consideration. On $T_{\proet}$ put
$A_T=\varprojlim_n(A_n)_T$.  Restriction of scalars along
$A_T\to(A_n)_T$ is the exact functor
\[
 \operatorname{Real}_{T,n}:
 D(T_{\proet},(A_n)_T)\longrightarrow D(T_{\proet},A_T).
\]
For $K\in D(T_{\proet},A_T)$ define
\[
 K_n=K\otimes^L_{A_T}(A_n)_T.
\]
The object $K$ is derived $I$-complete if
\[
 K\xrightarrow{\ \sim\ }
 \Rlim_n\operatorname{Real}_{T,n}(K_n)
\]
in $D(T_{\proet},A_T)$, where $\operatorname{Real}_{T,n}$ is restriction of
scalars.  Write $D_{\mathrm{comp}}(T_{\proet},A_T)$ for the full
triangulated subcategory of derived $I$-complete objects.  Define
$\Dbc(T_{\proet},A)$ to be its full triangulated subcategory consisting of
$K$ for which every
$K_n\in\Dbc(T_{\proet},A_n)$ and one finite cohomological interval works for
all $n$.
\end{definition}

\begin{definition}[Completed inverse image]
Let $u:S\to T$ and let the same constant pair $(A,I)$ be used on both
pro-etale sites.  The completed inverse image is
\[
 Lu^*_{\mathrm{comp}}:
 D_{\mathrm{comp}}(T_{\proet},A_T)
 \longrightarrow D_{\mathrm{comp}}(S_{\proet},A_S),
 \qquad
 K\longmapsto (Lu^*K)^{\wedge}.
\]
Here derived completion means
\[
 (Lu^*K)^{\wedge}
 =\Rlim_n\operatorname{Real}_{S,n}(Lu_n^*K_n).
\]
It is left adjoint to $Ru_*$ after both functors are restricted to
derived-complete objects.
\end{definition}

\begin{definition}[Normalized finite-level tower]
Let $C_n\in D(T_{\proet},A_n)$. A tower $(C_n)_n$ is normalized if, for every
$m\geq n$, its transition morphism induces an isomorphism
\[
 C_m\otimes^L_{A_m}A_n\xrightarrow{\ \sim\ }C_n
\]
and these isomorphisms satisfy the identity and three-index cocycle. For such
a tower all terms are also regarded as $A$-complexes by restriction of
scalars before taking an inverse limit.
\end{definition}

\begin{theorem}[Bounded-above normalized-tower reconstruction]
Let $T$ be a scheme.  Let $(C_n)_n$ be a normalized tower on
$T_{\proet}$, and assume $C_0$ is bounded above.  Put
\[
 C=\Rlim_n\operatorname{Real}_{T,n}C_n\in D(T_{\proet},A).
\]
Then $C$ is bounded above and derived $I$-complete, and the canonical maps
\[
 C\otimes^L_AA_r\xrightarrow{\ \sim\ }C_r
\]
are isomorphisms for every $r$.  If the $C_n$ lie in
$\Dbc(T_{\proet},A_n)$ and one finite cohomological interval works for all
$n$, then $C$ lies in $\Dbc(T_{\proet},A)$.
\dep{The pro-etale site has enough weakly contractible objects.  Evaluation
on such an object is exact and conservative and commutes with derived
inverse limits.  The algebraic input is the bounded-above normalized-tower theorem
for a Noetherian ring: compatible complexes over $A/I^{n+1}$ reconstruct to
a derived-complete $A$-complex whose derived reductions are the given
complexes.}
\end{theorem}

\begin{theorem}[Direct image and derived completion]
Let $u:S\to T$ be a morphism, viewed as a morphism of pro-etale topoi.  The
$A_T$-linear derived direct image is defined on the ambient derived category
by a $K$-injective resolution,
\[
 Ru_*:D(S_{\proet},A_S)\longrightarrow D(T_{\proet},A_T).
\]
For each $n$, write
\[
 Ru_{n,*}:D(S_{\proet},(A_n)_S)
 \longrightarrow D(T_{\proet},(A_n)_T)
\]
for the corresponding finite-coefficient direct image.
It carries derived-complete objects to derived-complete objects; write
$Ru_{*,\mathrm{comp}}$ for this restriction.  For every derived-complete $K$
there is a canonical isomorphism
\[
 Ru_{*,\mathrm{comp}}K
 \xrightarrow{\ \sim\ }
 \Rlim_n\operatorname{Real}_{T,n}(Ru_{n,*}K_n).
\]
Thus no normalization of the direct-image tower is assumed in order to define
$Ru_{*,\mathrm{comp}}$.
\dep{The ordinary Roos comparison theorem
$Ru_*\Rlim C_n\cong\Rlim Ru_*C_n$ for a morphism of topoi; the description
of derived completion as $\Rlim$ of finite reductions; and preservation of
derived-complete objects by derived direct image.}
\end{theorem}

\begin{definition}[Perfect coefficient reduction]
An $A$-module $B$ is perfect if $B[0]$ is quasi-isomorphic to a bounded
complex of finitely generated projective $A$-modules.  The adic
change-of-coefficients theorem below assumes that every $A_n$ is perfect as
an $A$-module.  It does not assume that $A_n$ is perfect as an $A_m$-module
for $m\geq n$.
\end{definition}

\begin{proposition}[Perfect projection formula and coefficient reduction]
Let $u:S\to T$, let $K\in D(S_{\proet},A_S)$, and let $B$ be a perfect
$A$-module.  The projection morphism
\[
 (Ru_*K)\otimes^L_A B
 \longrightarrow Ru_*(K\otimes^L_A B)
\]
is an isomorphism.  In particular, if $A_n$ is perfect over $A$, then
\[
 (Ru_*K)\otimes^L_AA_n
 \xrightarrow{\ \sim\ }Ru_{n,*}(K\otimes^L_AA_n).
\]
\dep{The projection formula for finite free coefficient complexes; thick
closure under shifts and cones; compatibility with finite direct sums and
retracts; and generation of perfect complexes by finite projective modules.}
\end{proposition}

\begin{lemma}[Compatibility of finite-level exchange]
Let $K$ be derived complete.  The reduction maps $K_m\to K_n$ for $m\geq n$
and naturality of the unit--counit construction make the finite-level smooth
exchange maps into a morphism of strict $A$-linear towers after restriction
of scalars.  This statement uses the transition maps of the towers; it does
not assert that $A_n$ is perfect over $A_m$.
\dep{Naturality of adjunction units and counits, and the identity and
three-index cocycles for coefficient reduction.}
\end{lemma}

\begin{proposition}[Finite-level realization of the adic exchange]
In the Cartesian square of Section~2, let $K$ be derived complete and assume
every $A_n$ is perfect over $A$.  There are canonical isomorphisms
\[
 Lg^*_{\mathrm{comp}}Rf_*K
 \cong
 \Rlim_n\operatorname{Real}_{Y',n}
       (Lg_n^*Rf_{n,*}K_n)
\]
and
\[
 Rf'_*Lv^*_{\mathrm{comp}}K
 \cong
 \Rlim_n\operatorname{Real}_{Y',n}
       (Rf'_{n,*}Lv_n^*K_n).
\]
Under these identifications, the adic exchange morphism is the derived
inverse limit of the finite-coefficient exchange morphisms.
\dep{Direct image and derived completion, perfect coefficient reduction,
the inverse-limit formula for completed pullback, and compatibility of
finite-level exchange.}
\end{proposition}

\begin{theorem}[Adic smooth base change]
Assume $X,Y,X',Y'$ are noetherian, $f:X\to Y$ is qcqs, and
$g:Y'\to Y$ is smooth.  Let $X'=X\times_Y Y'$ with projections $v$ and $f'$.
Assume every finite ring $A_n$ has order invertible on these schemes and is
perfect as an $A$-module.  Let $K\in\Dbc(X_{\proet},A)$ and assume each
$K_n$ belongs to the essential image of
$\Dbc(X_{\et},A_n)\to\Dbc(X_{\proet},A_n)$.  Then the unit--counit exchange
map is an isomorphism in the derived-complete category:
\[
 Lg^*_{\mathrm{comp}}Rf_*K
 \xrightarrow{\ \sim\ }
 Rf'_*Lv^*_{\mathrm{comp}}K.
\]
If the four systems
$Rf_{n,*}K_n$, $Lg_n^*Rf_{n,*}K_n$, $Lv_n^*K_n$, and
$Rf'_{n,*}Lv_n^*K_n$ have constructible cohomology and one common finite
cohomological interval works for every $n$, this is an isomorphism in
$\Dbc(Y'_{\proet},A)$.
\dep{Finite-level realization of the adic exchange, finite pro-etale smooth
base change, and conservativity of reductions on derived-complete objects.}
\end{theorem}

\begin{definition}[Ambient and constructible $E$-coefficient categories]
Let $E/\Ql$ be finite, with ring of integers $\mathcal O_E$ and uniformizer
$\varpi$.  Central localization inverts multiplication by $\varpi$ on every
Hom group.  Define
\[
 D_{\mathrm{comp}}(T,E)=
 \operatorname{Kar}\bigl(
   D_{\mathrm{comp}}(T_{\proet},\mathcal O_E)[\varpi^{-1}]
 \bigr)
\]
and
\[
 \Dbc(T,E)=
 \operatorname{Kar}\bigl(
   \Dbc(T_{\proet},\mathcal O_E)[\varpi^{-1}]
 \bigr).
\]
An object before Karoubi completion is an integral derived-complete complex;
a morphism is an integral morphism tensored with $E$.  An object of the
Karoubi envelope is a pair $(K,e)$ with $e^2=e$, a morphism
$(K,e)\to(L,d)$ is a morphism $a:K\to L$ satisfying $a=dae$, and the identity
of $(K,e)$ is $e$.  Both categories are idempotent-complete ordinary
triangulated categories, and the second has its canonical exact fully
faithful functor to the first.

The $\mathcal O_E$-linear functors $Ru_{*,\mathrm{comp}}$ and
$Lu^*_{\mathrm{comp}}$ commute with multiplication by $\varpi$.  Central
localization and splitting idempotents therefore define exact functors
\[
 Ru_{*,\mathrm{comp},E}:D_{\mathrm{comp}}(S,E)
       \longrightarrow D_{\mathrm{comp}}(T,E),
 \qquad
 Lu^*_{\mathrm{comp},E}:D_{\mathrm{comp}}(T,E)
       \longrightarrow D_{\mathrm{comp}}(S,E).
\]
No constructible landing assertion is part of these ambient definitions.
\end{definition}

\begin{definition}[Admissible lattices for a smooth square]
Retain the Cartesian square of Section~2 and put
$A_n=\mathcal O_E/\varpi^{n+1}$.  An integral lattice for
$K_E\in\Dbc(X,E)$ is an object
$L\in\Dbc(X_{\proet},\mathcal O_E)$, an idempotent
\[
 e\in\operatorname{End}_{
 \Dbc(X_{\proet},\mathcal O_E)[\varpi^{-1}]}
 (L[\varpi^{-1}]),
 \qquad e^2=e,
\]
and a chosen isomorphism
$(L[\varpi^{-1}],e)\xrightarrow{\sim}K_E$ in the Karoubi envelope
$\Dbc(X,E)$.  This formulation does not require $e$ to split before Karoubi
completion.

The lattice is \emph{finite-level etale admissible} if, for every $n$, its
reduction
\[
 L_n=L\otimes^L_{\mathcal O_E}A_n
\]
belongs to the essential image of
$\Dbc(X_{\et},A_n)\to\Dbc(X_{\proet},A_n)$.  It is
\emph{square-constructibly admissible} if it is finite-level etale
admissible and the four systems
\[
 Rf_{n,*}L_n,\qquad Lg_n^*Rf_{n,*}L_n,\qquad
 Lv_n^*L_n,\qquad Rf'_{n,*}Lv_n^*L_n
\]
have constructible cohomology and are all concentrated in one finite
cohomological interval independent of $n$.
\end{definition}

\begin{theorem}[$E$-coefficient smooth base change in the ambient category]
Assume $\ell$ is invertible on the four noetherian schemes, $f$ is qcqs, and
$g$ is smooth.  Let $K_E\in\Dbc(X,E)$ admit a finite-level etale admissible
lattice $L$.  The unit--counit exchange morphism is an isomorphism in the
ambient rationalized complete category:
\[
 Lg^*_{\mathrm{comp},E}Rf_{*,\mathrm{comp},E}K_E
 \xrightarrow{\ \sim\ }
 Rf'_{*,\mathrm{comp},E}Lv^*_{\mathrm{comp},E}K_E
 \quad\text{in }D_{\mathrm{comp}}(Y',E).
\]
It is independent of the chosen finite-level etale admissible lattice and is
compatible with every finite extension $E'/E$ for which admissibility is
preserved.

If $L$ is square-constructibly admissible, then all four rationalized terms
obtained from the square belong to the corresponding $\Dbc$ categories, and
the displayed isomorphism is an isomorphism in $\Dbc(Y',E)$.  Without this
four-system hypothesis, no claim that the ambient direct image lands in
$\Dbc(Y',E)$ is made.
\dep{The two-term finite-free resolution
$[\mathcal O_E\xrightarrow{\varpi^{n+1}}\mathcal O_E]$ of
$\mathcal O_E/\varpi^{n+1}$; adic smooth base change in the ambient complete
category; its uniform constructible landing criterion; commensurability of
lattices; central localization; and the triangulated Karoubi-envelope
theorem.}
\end{theorem}

\begin{definition}[Finite-stage pair categories over $\Qlb$]
Fix an algebraic closure $\Qlb$ of $\Ql$, and let finite coefficient fields
always mean finite subextensions of $\Qlb/\Ql$.  For either choice
\[
 \mathcal C_E(T)=D_{\mathrm{comp}}(T,E)
 \quad\text{or}\quad
 \mathcal C_E(T)=\Dbc(T,E),
\]
let $A=\mathcal O_E$ and $B=\mathcal O_M$.  On an integral representative,
finite scalar extension is the derived tensor functor
\[
 L\longmapsto L\otimes^L_{A_T}B_T.
\]
Finite freeness of $B$ over $A$ implies that this object is derived complete
and preserves the uniform constructible bounds when $L$ is constructible.
Multiplication by a uniformizer of $A$ becomes invertible after central
localization at a uniformizer of $B$.  Hence a fraction morphism
$a/\varpi_E^r$ is sent to the localized morphism
$(a\otimes 1)/\varpi_E^r$, and a Karoubi object $(L,e)$ is sent to
$(L\otimes^L_{A_T}B_T,e\otimes1)$.  Write the resulting exact functor as
\[
 T_{M/E}:\mathcal C_E(T)\longrightarrow\mathcal C_M(T).
\]
The tensor unit, associator, and shift comparison give specified natural
isomorphisms for identity extension and for
$T_{P/M}T_{M/E}\cong T_{P/E}$; they satisfy the unit and pentagon equations.

Define the additive pair category $\operatorname{Pair}_{\mathcal C}(T)$ as
follows.  Its objects are pairs $(E,K)$ with $K\in\mathcal C_E(T)$.  A
morphism $(E,K)\to(F,L)$ is represented by a finite common overfield
$M\supset E,F$ and a morphism
\[
 u:T_{M/E}K\longrightarrow T_{M/F}L
 \quad\text{in }\mathcal C_M(T).
\]
Two representatives are equal if their scalar extensions agree in some
further finite overfield.  Identities, addition, and composition are computed
after transporting the finitely many representatives involved to one common
finite stage.  Equivalently,
\[
 \operatorname{Hom}_{\operatorname{Pair}_{\mathcal C}(T)}
 ((E,K),(F,L))
 =\varinjlim_{M\supset E,F}
 \operatorname{Hom}_{\mathcal C_M(T)}
 (T_{M/E}K,T_{M/F}L).
\]
At this point only the additive category has been defined.
\end{definition}

\begin{proposition}[Finite common-stage principle]
Given finitely many objects of $\operatorname{Pair}_{\mathcal C}(T)$, they
become isomorphic to objects represented over one finite coefficient field
$M$.  After enlarging $M$ once more, any prescribed finite collection of
morphisms and composites among them has representatives at that same stage.
A finite collection of equalities between such morphisms holds in the pair
category if and only if all the equalities hold after transport to some
further finite coefficient field.
\dep{Filteredness of finite subextensions of $\Qlb$; the common-stage
equivalence relation on morphism representatives; and the unit and pentagon
coherence of finite scalar extension.}
\end{proposition}

\begin{definition}[Common-stage shifts, cones, and triangles]
Put $(E,K)[1]=(E,K[1])$; on a represented morphism use the shift comparison
isomorphisms of the two scalar-extension functors.  If
$f=[M,u]:(E,K)\to(F,L)$, choose in $\mathcal C_M(T)$ a distinguished triangle
\[
 T_{M/E}K\xrightarrow{u}T_{M/F}L
 \longrightarrow C(u)\longrightarrow (T_{M/E}K)[1].
\]
The associated common-stage cone object is $(M,C(u))$, and the last arrow is
transported to $(E,K)[1]$ through the specified shift comparison.  A triangle
in $\operatorname{Pair}_{\mathcal C}(T)$ is \emph{stagewise distinguished}
if it is isomorphic to a triangle whose objects and arrows are represented
over one finite field $M$ and which is distinguished in $\mathcal C_M(T)$.
\end{definition}

\begin{theorem}[Common-stage triangulation]
Scalar extension carries every selected common-stage cone triangle to an
isomorphic cone triangle.  Equivalent morphism representatives have
isomorphic cone triangles after passage to a common overfield, and the
stagewise distinguished class is independent of every cone choice.

With the preceding shift and distinguished triangles, the following hold:
\begin{enumerate}
\item every morphism has a distinguished cone triangle, identity triangles
      are distinguished, and distinguishedness is invariant under
      isomorphism;
\item a triangle is distinguished if and only if its signed rotation is
      distinguished;
\item every commutative square between the first arrows of two distinguished
      triangles extends to a morphism of triangles;
\item composable morphisms admit a Verdier octahedron whose required faces
      are distinguished.
\end{enumerate}
Consequently TR1--TR4 make $\operatorname{Pair}_{\mathcal C}(T)$ an ordinary
triangulated category, and every finite-stage inclusion is exact.
\dep{Filteredness of finite subextensions of $\Qlb$; coherence and exactness
of finite scalar extension; the finite common-stage principle; finite-stage
uniqueness of cone triangles up to an isomorphism fixing the first two
vertices; and TR1--TR4 in every $\mathcal C_M(T)$.}
\end{theorem}

\begin{definition}[Karoubi completion of the pair category]
For an ordinary triangulated pair category, its Karoubi envelope has objects
$(P,e)$ with $e^2=e$ and morphisms $a:(P,e)\to(Q,d)$ satisfying $a=dae$.
Define $(P,e)[1]=(P[1],e[1])$.  A triangle of Karoubi objects is distinguished
if it is isomorphic, as a triangle, to a direct summand of the image of a
stagewise distinguished triangle.
\end{definition}

\begin{theorem}[Triangulated Karoubi completion]
The shift and distinguished triangles of the preceding definition give
$\operatorname{Kar}(\operatorname{Pair}_{\mathcal C}(T))$ its canonical
structure of an idempotent-complete ordinary triangulated category.  The
canonical functor from the pair category is exact, and every finite-stage
inclusion remains exact after this functor.
\dep{Common-stage triangulation; the equations for morphisms between
Karoubi objects; closure of direct summands of distinguished triangles; and
the triangulated Karoubi-envelope theorem.}
\end{theorem}

\begin{definition}[The two $\Qlb$-coefficient categories]
Apply this construction to the two finite-stage systems above and define
\[
 D_{\mathrm{comp}}(T,\Qlb)
 =\operatorname{Kar}\bigl(
   \operatorname{Pair}_{D_{\mathrm{comp}}}(T)\bigr),
 \qquad
 \Dbc(T,\Qlb)
 =\operatorname{Kar}\bigl(
   \operatorname{Pair}_{\Dbc}(T)\bigr).
\]
The stagewise exact embeddings $\Dbc(T,E)\to D_{\mathrm{comp}}(T,E)$ induce
an exact fully faithful functor
$\Dbc(T,\Qlb)\to D_{\mathrm{comp}}(T,\Qlb)$.  Coherent families of ambient
finite-stage direct and completed inverse images extend first to the
$D_{\mathrm{comp}}$ pair categories and then to their Karoubi envelopes.  For
a morphism $u:S\to T$ this defines exact functors
\[
 Ru_{*,\mathrm{comp},\Qlb}:D_{\mathrm{comp}}(S,\Qlb)
       \longrightarrow D_{\mathrm{comp}}(T,\Qlb),
 \qquad
 Lu^*_{\mathrm{comp},\Qlb}:D_{\mathrm{comp}}(T,\Qlb)
       \longrightarrow D_{\mathrm{comp}}(S,\Qlb).
\]
They restrict to $\Dbc$ pair categories only under the constructible landing
hypotheses stated below.
\end{definition}

\begin{theorem}[$\Qlb$-coefficient smooth base change]
Assume the hypotheses of ambient $E$-coefficient smooth base change hold for
every finite-stage representative under consideration and are preserved by
passage to a common finite overfield.  The finite-stage exchange
isomorphisms define an isomorphism
\[
 Lg^*_{\mathrm{comp},\Qlb}Rf_{*,\mathrm{comp},\Qlb}K
 \xrightarrow{\ \sim\ }
 Rf'_{*,\mathrm{comp},\Qlb}Lv^*_{\mathrm{comp},\Qlb}K
 \quad\text{in }D_{\mathrm{comp}}(Y',\Qlb)
\]
for every $K\in\Dbc(X,\Qlb)$ represented by a finite-level etale admissible
lattice at a finite stage.  The construction is independent of the common
stage and preserves every idempotent defining a Karoubi summand.

If, after passage to one common finite stage, the representative lattice is
square-constructibly admissible, then both sides belong to
$\Dbc(Y',\Qlb)$ and the same arrow is an isomorphism there.  More generally,
if this four-system condition holds for every object of a full triangulated
subcategory $\mathcal A\subset\Dbc(X,\Qlb)$ and is preserved under finite
scalar extension, then the ambient functors restrict to $\Dbc$-valued
functors on $\mathcal A$, and the displayed arrows form a natural
isomorphism between those restricted functors.  No $\Dbc$-valued direct-image
functor on all of $\Dbc(X,\Qlb)$ is asserted without this landing hypothesis.
\dep{Ambient $E$-coefficient smooth base change; preservation of admissible
lattices under finite scalar extension; common-stage triangulation; the
finite common-stage principle; and functoriality of the triangulated
Karoubi envelope.}
\end{theorem}

\section{Coherence package}

\begin{theorem}[Coherence of smooth base change]
At finite, adic, $E$, and $\Qlb$ coefficients, the isomorphisms satisfy:
\begin{enumerate}
\item naturality in the coefficient complex and compatibility with shifts,
      distinguished triangles, and chosen morphisms of triangles;
\item identity for an identity base change;
\item compatibility with a composite $Y''\to Y'\to Y$ of smooth maps;
\item compatibility with $R(pf)_*\cong Rp_*Rf_*$ for a composite direct
      image $X\xrightarrow{f}Y\xrightarrow{p}Z$;
\item compatibility with finite reduction, adic reconstruction, scalar
      extension, common overfields, and splitting idempotents.
\end{enumerate}
\dep{The unit--counit definition, composition lemmas, finite-level
coefficient naturality, and uniqueness after completion and scalar
extension.}
\end{theorem}

\section{Dependency summary and source locators}

The direct ambient chain for an arbitrary allowable-torsion sheaf is
\[
\begin{gathered}
 \text{degree-zero flat base change}
 +\text{ universal-curve failure criterion}
 +\text{ terminal curve vanishing}\\
 \Longrightarrow \text{relative-dimension-one affine smooth base change}
 \Longrightarrow \text{smooth local normal form}\\
 \Longrightarrow \text{ambient qcqs smooth base change}.
\end{gathered}
\]
There is also an independent compactification chain for finite constructible
coefficients:
\[
\begin{gathered}
 \mathbf A^1\text{ local acyclicity}
 \Longrightarrow \text{smooth ULA}
 \Longrightarrow \text{open-immersion base change}\\
 \Longrightarrow \text{quasi-projective base change}
 \Longrightarrow \text{finite affine Cech and continuity assembly}.
\end{gathered}
\]
Constructible and derived forms then follow by finite stratification,
finite-range hypercohomology, and bounded-below truncation. Pro-etale and
adic forms are coefficient transports subject to the comparison and
reconstruction hypotheses stated above.

Primary source locators are SGA~4, Expose~XV, Sections~1--2, for local
acyclicity and smooth local acyclicity, and Expose~XVI, Theorem~1.1 and
Corollary~1.2, for smooth base change. Modern locators in the Stacks Project
include Tags 054L (smooth local normal form), 0F08 (universal-curve
reduction), 03SC (torsion cohomology of curves), 0EYU (smooth base change),
0GJM (local acyclicity), 03S6 (extension by zero and base change), 0EZJ
(spreading the tame Galois cover), and 09AJ.  Tags 09AU, 099M, 099N, and 099P
give bounded normalized-tower reconstruction, completion as $\Rlim$,
compatibility of direct image with completion, and detection on sites with
enough weakly contractible objects.
