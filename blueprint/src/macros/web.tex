% plasTeX's HTML5 template for \hyperref[label]{text} ignores the label and emits
% href="#.", while its template for \hyperlink resolves the label.  Both produce
% the same anchor text, so the web version of \termref uses \hyperlink.
\renewcommand{\termref}[2]{\hyperlink{#1}{#2}}
