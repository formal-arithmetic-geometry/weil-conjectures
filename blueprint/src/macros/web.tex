% Typically the printed version could have more fancy decorations.

% plasTeX has no script math alphabet; render \mathscr as \mathcal on the web.
\newcommand{\mathscr}[1]{\mathcal{#1}}

% Counterpart of macros/print.tex's \nagataUsesAnnotations, which installs a
% \uses wrapper printing an explicit reference for every listed edge.  That
% wrapper is written in expl3, which plasTeX does not implement: it would emit
% its own source into the rendered page and print nothing.  So on the web the
% command does nothing and \uses stays the blueprint package's own.
\newcommand{\nagataUsesAnnotations}{}
