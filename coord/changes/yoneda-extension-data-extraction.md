/-- The map induced on the left endpoint by a map of extension sequences.

def leftMap {V U V' U' : A} {i : ℕ} (E : YonedaExtension A V U i)

namespace YonedaExtension

variable {A : Type u} [Category.{v} A] [Abelian A]

/-- A short extension is split when its short complex has a Mathlib splitting.

Source: Stacks Project, Derived Categories, Tag 06XV, and Homology, Section 010I
(blueprint: lem-coef-yoneda-derived-ext-comparison). -/
def IsSplit {V U : A} (E : YonedaExtension A V U 1) : Prop :=
  Nonempty (E.sequence.sc E.exact.toIsComplex 0).Splitting

/-- Data identifying an extension with the splice of a short exact sequence and
an extension of one smaller positive degree. The kernel object is not retained
among the middle terms of the spliced sequence.

Source: Stacks Project, Derived Categories, Tag 06XU and the Yoneda composition
following Tag 06XV (blueprint: lem-coef-yoneda-derived-ext-comparison). -/
structure Splice {U : A} {n : ℕ} (S : ShortComplex A)
    (F : YonedaExtension A S.X₃ U (n + 1))
    (E : YonedaExtension A S.X₁ U (n + 2)) where
  /-- Identification of the new first middle term. -/
  firstIso : S.X₂ ≅ E.sequence.obj' 1
  /-- Identification of all subsequent terms and their differentials. -/
  tailIso : F.sequence.δ₀ ≅ E.sequence.δ₀.δ₀
  /-- The first arrow agrees with that of the short exact sequence. -/
  first_comm : S.f ≫ firstIso.hom = E.leftIso.inv ≫ E.sequence.map' 0 1
  /-- At the splice the two arrows through the kernel object are composed. -/
  splice_comm : S.g ≫ F.leftIso.inv ≫ F.sequence.map' 0 1 ≫ tailIso.hom.app 0 =
    firstIso.hom ≫ E.sequence.map' 1 2
  /-- The identification fixes the right endpoint. -/
  right_comm : tailIso.hom.app (Fin.last (n + 1)) ≫ E.rightIso.hom = F.rightIso.hom

/-- The Baer-sum construction on actual extensions. First take a termwise direct
sum `T`, then pull back its last arrow along the diagonal to obtain `P`, and push
out its first arrow along addition to obtain `G`. All the other middle terms are
preserved by isomorphisms. In an abelian category a binary product is also a binary
direct sum; its projections specify the termwise direct sum here.

Source: Stacks Project, Homology, Section 010I, Baer sum construction, and Derived
Categories, Tag 06XV (blueprint: lem-coef-yoneda-derived-ext-comparison). -/
def BaerSum {V U : A} {i : ℕ} (E F G : YonedaExtension A V U i) : Prop :=
  ∃ (T : YonedaExtension A (Limits.prod V V) (Limits.prod U U) i)
    (P : YonedaExtension A (Limits.prod V V) U i)
    (p₁ : T.sequence ⟶ E.sequence) (p₂ : T.sequence ⟶ F.sequence)
    (b : P.sequence ⟶ T.sequence) (c : P.sequence ⟶ G.sequence),
    T.leftMap E p₁ = Limits.prod.fst ∧ T.leftMap F p₂ = Limits.prod.snd ∧
    T.rightMap E p₁ = Limits.prod.fst ∧ T.rightMap F p₂ = Limits.prod.snd ∧
    (∀ k : Fin (i + 2), Nonempty (IsLimit (BinaryFan.mk (p₁.app k) (p₂.app k)))) ∧
    P.leftMap T b = 𝟙 _ ∧ P.rightMap T b = Limits.prod.lift (𝟙 U) (𝟙 U) ∧
    IsPullback (b.app ⟨i, by omega⟩) (P.sequence.map' i (i + 1) ≫ P.rightIso.hom)
      (T.sequence.map' i (i + 1) ≫ T.rightIso.hom) (Limits.prod.lift (𝟙 U) (𝟙 U)) ∧
    (∀ k : Fin (i + 2), k.val < i → IsIso (b.app k)) ∧
    P.leftMap G c = Limits.prod.fst + Limits.prod.snd ∧ P.rightMap G c = 𝟙 U ∧
    IsPushout (P.leftIso.inv ≫ P.sequence.map' 0 1) (Limits.prod.fst + Limits.prod.snd)
      (c.app ⟨1, by omega⟩) (G.leftIso.inv ≫ G.sequence.map' 0 1) ∧
    (∀ k : Fin (i + 2), 1 < k.val → IsIso (c.app k))

end YonedaExtension
```
